## ② 왜 필요한가

지금까지 필요한 기능을 **전부 직접 만들었습니다.**
평균 계산, 최댓값 찾기, 중복 제거… 원리를 배우기 위해선 좋았습니다.

하지만 실무에서는 **이미 있는 것을 다시 만들지 않습니다.**

```python
# 직접 만들면 15줄
def get_most_common(items):
    counter = {}
    for item in items:
        counter[item] = counter.get(item, 0) + 1
    return max(counter, key=counter.get)

# 표준 라이브러리를 쓰면 2줄
from collections import Counter
Counter(items).most_common(1)[0][0]
```

파이썬은 **"배터리 포함(batteries included)"** 을 표방합니다.
설치하는 순간 200개가 넘는 모듈이 함께 옵니다.

**중요한 건 외우는 게 아니라 "이런 게 있다"를 아는 것**입니다.
필요할 때 검색해 찾을 수 있으면 충분합니다.

### pathlib — 경로 다루기 (가장 중요)

```python
from pathlib import Path

# 현재 파일 기준 경로 — 실행 위치와 무관!
BASE_DIR = Path(__file__).resolve().parent

# 경로 합치기 — / 연산자
ASSET_DIR = BASE_DIR / "assets"
image_path = ASSET_DIR / "player.png"

print(image_path)                    # C:\프로젝트\assets\player.png
print(image_path.name)               # player.png
print(image_path.stem)               # player
print(image_path.suffix)             # .png
print(image_path.parent)             # C:\프로젝트\assets
print(image_path.exists())           # True / False
```

**주요 메서드**

```python
p = Path("data")

p.mkdir(exist_ok=True)               # 폴더 생성 (있어도 에러 안 남)
p.mkdir(parents=True, exist_ok=True) # 중간 폴더까지 생성

list(p.glob("*.json"))               # data 폴더의 json 파일들
list(p.rglob("*.py"))                # 하위 폴더까지 전부 검색

f = p / "config.json"
f.write_text("{}", encoding="utf-8")  # 쓰기 (한 줄)
content = f.read_text(encoding="utf-8")  # 읽기 (한 줄)
f.unlink(missing_ok=True)            # 삭제
```

### datetime — 날짜와 시간

```python
from datetime import datetime, timedelta, date

now = datetime.now()
print(now)                                   # 2026-08-07 14:30:15.123456
print(now.strftime("%Y-%m-%d %H:%M:%S"))     # 2026-08-07 14:30:15
print(now.year, now.month, now.day)          # 2026 8 7
```

### collections — 특수 컨테이너

```python
from collections import Counter, defaultdict, deque, namedtuple

# ① Counter — 개수 세기
words = ["a", "b", "a", "c", "a", "b"]
c = Counter(words)
print(c)                      # Counter({'a': 3, 'b': 2, 'c': 1})
print(c.most_common(2))       # [('a', 3), ('b', 2)]
print(c["a"])                 # 3
print(c["없는키"])             # 0 (KeyError 안 남!)

# ② defaultdict — 기본값이 있는 딕셔너리
d = defaultdict(list)
d["과일"].append("사과")       # 키가 없어도 자동으로 [] 생성
print(d)                      # defaultdict(<class 'list'>, {'과일': ['사과']})

# ③ deque — 양쪽에서 빠른 추가·삭제
q = deque([1, 2, 3], maxlen=5)
q.appendleft(0)               # 앞에 추가 (리스트보다 훨씬 빠름)
q.append(4)
print(q)                      # deque([0, 1, 2, 3, 4], maxlen=5)
q.append(5)                   # maxlen 초과 → 앞에서 하나 밀려남
print(q)                      # deque([1, 2, 3, 4, 5], maxlen=5)

# ④ namedtuple — 이름 있는 튜플
Point = namedtuple("Point", ["x", "y"])
p = Point(10, 20)
print(p.x, p.y)               # 10 20
```

> **`deque(maxlen=N)`** 은 [Day 020](day-020.md)의 궤적 관리에 딱 맞습니다.
> `trail.pop(0)` 을 직접 할 필요가 없습니다.

### itertools — 반복 도구

```python
from itertools import product, combinations, permutations, cycle, chain

# 모든 조합
print(list(product([1, 2], ["a", "b"])))      # [(1,'a'), (1,'b'), (2,'a'), (2,'b')]
print(list(combinations([1, 2, 3], 2)))       # [(1,2), (1,3), (2,3)]
print(list(permutations([1, 2, 3], 2)))       # [(1,2), (1,3), (2,1), ...]

# 리스트 이어붙이기
print(list(chain([1, 2], [3, 4])))            # [1, 2, 3, 4]

# 무한 반복 (게임 애니메이션에 유용)
colors = cycle(["빨강", "파랑", "초록"])
for _ in range(5):
    print(next(colors), end=" ")              # 빨강 파랑 초록 빨강 파랑
```

### Step 1. pathlib 로 안전한 경로 만들기

```python
"""pathlib 실습 — 실행 위치와 무관한 경로 만들기."""

from pathlib import Path
import json

# ── 이 파일이 있는 폴더를 기준으로 ──
BASE_DIR = Path(__file__).resolve().parent
DATA_DIR = BASE_DIR / "data"
ASSET_DIR = BASE_DIR / "assets"

print("=" * 56)
print(f"현재 파일:   {Path(__file__).name}")
print(f"기준 폴더:   {BASE_DIR}")
print(f"데이터 폴더: {DATA_DIR}")
print("=" * 56)

# ── 폴더 만들기 ──
DATA_DIR.mkdir(exist_ok=True)
ASSET_DIR.mkdir(exist_ok=True)
print(f"폴더 생성 완료 (이미 있어도 에러 없음)")

# ── 파일 쓰기 ──
config_file = DATA_DIR / "config.json"
config = {"volume": 80, "fullscreen": False, "difficulty": "normal"}

config_file.write_text(
    json.dumps(config, ensure_ascii=False, indent=2),
    encoding="utf-8",
)
print(f"\n저장: {config_file.name}")

# ── 파일 정보 ──
print(f"\n[파일 정보]")
print(f"  전체 경로: {config_file}")
print(f"  파일명:    {config_file.name}")
print(f"  확장자 뺀: {config_file.stem}")
print(f"  확장자:    {config_file.suffix}")
print(f"  부모 폴더: {config_file.parent.name}")
print(f"  존재하나:  {config_file.exists()}")
print(f"  크기:      {config_file.stat().st_size} 바이트")

# ── 파일 읽기 ──
loaded = json.loads(config_file.read_text(encoding="utf-8"))
print(f"\n불러온 설정: {loaded}")

# ── 여러 파일 만들고 검색 ──
for i in range(1, 4):
    (DATA_DIR / f"save{i}.json").write_text(
        json.dumps({"slot": i, "score": i * 1000}), encoding="utf-8")

print(f"\n[data 폴더의 json 파일]")
for f in sorted(DATA_DIR.glob("*.json")):
    size = f.stat().st_size
    print(f"  {f.name:<16} {size:>4} 바이트")

print(f"\n[프로젝트 전체 .py 파일]")
for f in sorted(BASE_DIR.rglob("*.py")):
    print(f"  {f.relative_to(BASE_DIR)}")

# ── 정리 ──
for f in DATA_DIR.glob("save*.json"):
    f.unlink()
print(f"\n임시 파일 삭제 완료")
print("=" * 56)
```

### Step 2. 표준 라이브러리 투어

`stdlib_tour.py`

```python
"""자주 쓰는 표준 라이브러리 실습."""

import random
import statistics as st
from collections import Counter, defaultdict, deque
from datetime import datetime, timedelta
from itertools import combinations

print("=" * 56)
print("                표준 라이브러리 투어")
print("=" * 56)

# ══════════ ① Counter — 개수 세기 ══════════
print("\n[① Counter]")

log = ["로그인", "구매", "로그인", "조회", "구매", "로그인", "조회", "구매", "구매"]
counter = Counter(log)

print(f"  전체 이벤트: {len(log)}건")
print(f"  종류별 집계: {dict(counter)}")
print(f"  가장 많은 3개: {counter.most_common(3)}")
print(f"  '구매' 횟수: {counter['구매']}")
print(f"  없는 키:     {counter['탈퇴']}  ← KeyError 안 남")

# ══════════ ② defaultdict — 자동 초기화 ══════════
print("\n[② defaultdict]")

students = [("1반", "홍길동"), ("2반", "김철수"), ("1반", "이영희"), ("2반", "박민수")]
by_class = defaultdict(list)

for cls, name in students:
    by_class[cls].append(name)          # 키가 없어도 자동으로 [] 생성

for cls, names in sorted(by_class.items()):
    print(f"  {cls}: {', '.join(names)} ({len(names)}명)")

# ══════════ ③ deque — 최근 N개만 유지 ══════════
print("\n[③ deque]")

recent = deque(maxlen=5)
for i in range(1, 9):
    recent.append(f"작업{i}")

print(f"  최근 5개만 유지: {list(recent)}")
print(f"  앞에 추가:")
recent.appendleft("긴급작업")
print(f"    {list(recent)}  ← 뒤가 하나 밀려남")

# ══════════ ④ datetime — 날짜 계산 ══════════
print("\n[④ datetime]")

now = datetime.now()
print(f"  현재:      {now.strftime('%Y-%m-%d %H:%M:%S')}")
print(f"  요일:      {now.strftime('%A')}")
print(f"  1주일 뒤:  {(now + timedelta(weeks=1)).strftime('%Y-%m-%d')}")
print(f"  100일 전:  {(now - timedelta(days=100)).strftime('%Y-%m-%d')}")

course_end = datetime(2027, 4, 30)
remaining = course_end - now
print(f"  과정 종료까지: {remaining.days}일")

# ══════════ ⑤ statistics — 통계 ══════════
print("\n[⑤ statistics]")

scores = [85, 92, 78, 90, 65, 88, 92, 71]
print(f"  데이터:     {scores}")
print(f"  평균:       {st.mean(scores):.2f}")
print(f"  중앙값:     {st.median(scores)}")
print(f"  최빈값:     {st.mode(scores)}")
print(f"  표준편차:   {st.stdev(scores):.2f}")

# ══════════ ⑥ random — 난수와 재현성 ══════════
print("\n[⑥ random]")

random.seed(42)
first = [random.randint(1, 100) for _ in range(5)]

random.seed(42)                          # 같은 시드
second = [random.randint(1, 100) for _ in range(5)]
```

```python
"""랭킹 관리 — pathlib, json, datetime, collections 활용."""

import json
from datetime import datetime
from collections import Counter

from .paths import RANKING_FILE, ensure_dirs

MAX_ENTRIES = 10


def load():
    """랭킹을 불러옵니다."""
    ensure_dirs()
    if not RANKING_FILE.exists():
        return []
    try:
        return json.loads(RANKING_FILE.read_text(encoding="utf-8"))
    except json.JSONDecodeError:
        print("[경고] 랭킹 파일이 손상되었습니다.")
        return []


def save(entries):
    """랭킹을 저장합니다."""
    ensure_dirs()
    RANKING_FILE.write_text(
        json.dumps(entries, ensure_ascii=False, indent=2),
        encoding="utf-8",
    )


def add(name, score, kills=0, play_seconds=0):
    """새 기록을 추가하고 상위 MAX_ENTRIES 개만 남깁니다. 순위를 반환합니다."""
    entries = load()
    entries.append({
        "name": name,
        "score": score,
        "kills": kills,
        "seconds": play_seconds,
        "date": datetime.now().strftime("%Y-%m-%d %H:%M"),
    })

    entries.sort(key=lambda e: e["score"], reverse=True)
    entries = entries[:MAX_ENTRIES]
    save(entries)

    for i, e in enumerate(entries, start=1):
        if e["name"] == name and e["score"] == score:
            return i
    return None


def is_new_record(score):
    """랭킹에 들어갈 점수인지 확인합니다."""
    entries = load()
    if len(entries) < MAX_ENTRIES:
        return True
    return score > entries[-1]["score"]


def format_table():
    """랭킹표를 문자열로 만듭니다."""
    entries = load()
    if not entries:
        return "  아직 기록이 없습니다."

    lines = [f"{'순위':<5}{'이름':<12}{'점수':>10}{'처치':>7}{'시간':>8}  {'날짜'}"]
    lines.append("-" * 62)

    medals = {1: "🥇", 2: "🥈", 3: "🥉"}
    for i, e in enumerate(entries, start=1):
        rank = medals.get(i, f"{i}위")
        m, s = divmod(e.get("seconds", 0), 60)
        lines.append(
            f"{rank:<5}{e['name']:<12}{e['score']:>10,}{e.get('kills', 0):>7}"
            f"{m:>5}:{s:02d}  {e.get('date', '-')}"
        )

    lines.append("-" * 62)

    # Counter 로 최다 플레이어 찾기
    names = Counter(e["name"] for e in entries)
    top_name, top_count = names.most_common(1)[0]
    lines.append(f"  총 {len(entries)}건 · 최다 기록: {top_name} ({top_count}회)")

    return "\n".join(lines)


if __name__ == "__main__":
    import random

    print("=== 랭킹 테스트 ===\n")

    for name in ["홍길동", "김철수", "이영희", "홍길동", "박민수"]:
        score = random.randint(1000, 50000)
        kills = random.randint(10, 200)
        seconds = random.randint(60, 600)
        rank = add(name, score, kills, seconds)
        print(f"{name:<8} {score:>7,}점 → {rank}위")

    print(f"\n{format_table()}")
    print(f"\n30000점은 신기록인가? {is_new_record(30000)}")
    print(f"100점은 신기록인가?   {is_new_record(100)}")
```

```python
"""프로젝트의 파이썬 파일 줄 수를 분석합니다."""

from pathlib import Path

BASE_DIR = Path(__file__).resolve().parent

# 제외할 폴더
IGNORE = {"venv", ".venv", "__pycache__", ".git", "node_modules"}


def count_lines(path):
    """파일의 (전체 줄, 코드 줄, 주석 줄, 빈 줄)을 반환합니다."""
    try:
        lines = path.read_text(encoding="utf-8").splitlines()
    except (UnicodeDecodeError, PermissionError):
        return 0, 0, 0, 0

    total = len(lines)
    blank = sum(1 for line in lines if not line.strip())
    comment = sum(1 for line in lines if line.strip().startswith("#"))
    code = total - blank - comment

    return total, code, comment, blank


def should_skip(path):
    """제외 폴더 안의 파일이면 True."""
    return any(part in IGNORE for part in path.parts)


def main():
    results = []

    for py_file in BASE_DIR.rglob("*.py"):
        if should_skip(py_file):
            continue

        total, code, comment, blank = count_lines(py_file)
        results.append({
            "path": py_file.relative_to(BASE_DIR),
            "total": total,
            "code": code,
            "comment": comment,
            "blank": blank,
        })

    if not results:
        print("파이썬 파일을 찾을 수 없습니다.")
        return

    results.sort(key=lambda r: r["total"], reverse=True)

    print("=" * 68)
    print(f"프로젝트: {BASE_DIR.name}")
    print("=" * 68)
    print(f"{'파일':<34}{'전체':>7}{'코드':>7}{'주석':>7}{'빈줄':>7}")
    print("-" * 68)

    for r in results[:5]:
        print(f"{str(r['path']):<34}{r['total']:>7}{r['code']:>7}"
              f"{r['comment']:>7}{r['blank']:>7}")

    if len(results) > 5:
        print(f"  ... 외 {len(results) - 5}개 파일")

    print("-" * 68)
    print(f"{'합계':<34}"
          f"{sum(r['total'] for r in results):>7}"
          f"{sum(r['code'] for r in results):>7}"
          f"{sum(r['comment'] for r in results):>7}"
          f"{sum(r['blank'] for r in results):>7}")
    print(f"파일 {len(results)}개 · "
          f"평균 {sum(r['total'] for r in results) / len(results):.0f}줄")

    # 300줄 넘는 파일 경고
    long_files = [r for r in results if r["total"] > 300]
    if long_files:
        print(f"\n⚠ 300줄이 넘는 파일 {len(long_files)}개 — 분리를 검토하세요")
        for r in long_files:
            print(f"    {r['path']} ({r['total']}줄)")

    print("=" * 68)


if __name__ == "__main__":
    main()
```

- `any(part in IGNORE for part in path.parts)` — 경로의 각 부분을 확인해 `venv` 등을 제외합니다.
- `read_text()` 가 실패할 수 있으므로 `try/except` 로 감쌉니다. (인코딩이 다른 파일)
- **300줄 경고**를 넣어 리팩터링이 필요한 파일을 알려 줍니다. [Day 028](day-028.md)의 기준입니다.
</details>

**3.** 아래 코드는 로그가 두 번씩 출력됩니다. 원인과 해결책을 쓰세요.

```python
import logging


def get_logger():
    logger = logging.getLogger("game")
    logger.setLevel(logging.INFO)

    handler = logging.StreamHandler()
    handler.setFormatter(logging.Formatter("[%(levelname)s] %(message)s"))
    logger.addHandler(handler)

    return logger


log1 = get_logger()
log1.info("첫 번째 메시지")

log2 = get_logger()
log2.info("두 번째 메시지")
```

**출력**

```
[INFO] 첫 번째 메시지
[INFO] 두 번째 메시지
[INFO] 두 번째 메시지
```

<details>
<summary>답 보기</summary>

**원인 — 같은 로거에 핸들러가 중복으로 추가됩니다.**

`logging.getLogger("game")` 은 **같은 이름이면 같은 객체를 반환**합니다. (싱글턴)

```
1회차: logger("game") 생성 → 핸들러 1개 추가
2회차: 같은 logger 반환   → 핸들러 하나 더 추가 (총 2개)
```

핸들러가 2개면 메시지 하나가 **두 번 출력**됩니다.
`get_logger()` 를 10번 부르면 10번 출력됩니다.

**해결 ① 이미 핸들러가 있으면 건너뛰기**

```python
def get_logger(name="game"):
    logger = logging.getLogger(name)

    if logger.handlers:               # 이미 설정됨
        return logger

    logger.setLevel(logging.INFO)
    handler = logging.StreamHandler()
    handler.setFormatter(logging.Formatter("[%(levelname)s] %(message)s"))
    logger.addHandler(handler)

    return logger
```

**해결 ② 모듈 최상단에서 한 번만 설정**

```python
# logger.py
import logging

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s",
    datefmt="%H:%M:%S",
)

log = logging.getLogger("game")       # 모듈 수준 변수
```

```python
# 다른 파일에서
from logger import log

log.info("메시지")                    # 어디서 import 해도 하나만 출력
```

**②번이 더 좋은 이유** — 모듈은 **첫 import 때 한 번만 실행**됩니다([Day 026](day-026.md)).
설정 코드가 자연스럽게 한 번만 돌게 됩니다.

> **교훈** — **"이 함수가 여러 번 호출될 수 있는가?"** 를 항상 생각하세요.
> 호출될 때마다 무언가를 **추가**하는 함수는 중복 방지가 필요합니다.
>
> 같은 문제가 pygame에도 있습니다. 스프라이트를 그룹에 두 번 추가하면 `update()` 가 두 번 호출됩니다.
> (pygame `Group` 은 내부적으로 중복을 막아 주지만, 직접 만든 리스트는 그렇지 않습니다)
</details>
