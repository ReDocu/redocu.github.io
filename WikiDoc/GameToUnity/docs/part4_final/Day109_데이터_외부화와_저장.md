### 3-4. 오버라이드 적용

**왜 필요한가** — SO를 오염시키지 않는다.

```csharp
namespace OneFloor.Data
{
    public class BalanceLoader : MonoBehaviour
    {
        public static BalanceLoader Instance { get; private set; }

        [SerializeField] private GameDatabase database;
        [SerializeField] private string fileName = "balance.json";

        private BalanceConfig config;

        private readonly Dictionary<string, EnemyBalance> enemyOverrides = new();
        private readonly Dictionary<string, WeaponBalance> weaponOverrides = new();
        private readonly Dictionary<int, FloorBalance> floorOverrides = new();

        public PlayerBalance Player => config.player;
        public EconomyBalance Economy => config.economy;

        private void Awake() { Instance = this; }

        public void Initialize()
        {
            config = ConfigLoader.LoadFromStreamingAssets(fileName, new BalanceConfig());

            BuildLookups();

            DebugUtil.Log($"[밸런스] v{config.version} 로드 — " +
                          $"적 {enemyOverrides.Count}, 무기 {weaponOverrides.Count}, " +
                          $"층 {floorOverrides.Count}");
        }

        private void BuildLookups()
        {
            enemyOverrides.Clear();
            weaponOverrides.Clear();
            floorOverrides.Clear();

            foreach (EnemyBalance e in config.enemies)
            {
                if (string.IsNullOrEmpty(e.id)) continue;
                enemyOverrides[e.id] = e;
            }

            foreach (WeaponBalance w in config.weapons)
            {
                if (string.IsNullOrEmpty(w.id)) continue;
                weaponOverrides[w.id] = w;
            }

            foreach (FloorBalance f in config.floors)
                floorOverrides[f.floor] = f;
        }

        // ─── 조회 (SO 기본값 + 오버라이드) ───

        public int GetEnemyHp(EnemyData data)
        {
            if (enemyOverrides.TryGetValue(data.id, out var o) && o.maxHp >= 0)
                return o.maxHp;
            return data.maxHp;
        }

        public float GetEnemySpeed(EnemyData data)
        {
            if (enemyOverrides.TryGetValue(data.id, out var o) && o.moveSpeed >= 0)
                return o.moveSpeed;
            return data.moveSpeed;
        }

        public int GetWeaponDamage(WeaponData data)
        {
            if (weaponOverrides.TryGetValue(data.id, out var o) && o.damage >= 0)
                return o.damage;
            return data.damage;
        }

        public float GetFloorHpMult(FloorData data)
        {
            if (floorOverrides.TryGetValue(data.floorNumber, out var o) && o.hpMult >= 0)
                return o.hpMult;
            return data.hpMultiplier;
        }
    }
}
```

```
   ⚠️ SO를 직접 수정하지 않는다 (Day 78·79)

   data.maxHp = o.maxHp;                       ✗ 원본 오염
   별도 Dictionary에서 조회                     ✔
```

### 3-5. 로더와 폴백

**왜 필요한가** — 파일이 없거나 깨져도 실행돼야 한다.

```csharp
namespace OneFloor.Core
{
    public static class ConfigLoader
    {
        public static T LoadFromStreamingAssets<T>(string fileName, T fallback)
            where T : class
        {
            string path = Path.Combine(Application.streamingAssetsPath, fileName);

            if (!File.Exists(path))
            {
                DebugUtil.LogWarning($"[설정] {fileName} 없음 → 기본값");
                return fallback;
            }

            try
            {
                string json = File.ReadAllText(path);

                if (string.IsNullOrWhiteSpace(json))
                {
                    DebugUtil.LogWarning($"[설정] {fileName} 비어 있음");
                    return fallback;
                }

                T result = JsonUtility.FromJson<T>(json);

                if (result == null)
                {
                    DebugUtil.LogWarning($"[설정] {fileName} 파싱 실패");
                    return fallback;
                }

                DebugUtil.Log($"[설정] {fileName} 로드");
                return result;
            }
            catch (System.Exception e)
            {
                DebugUtil.LogWarning($"[설정] {fileName} 오류: {e.Message}");
                return fallback;
            }
        }
    }
}
```

### 3-7. 밸런스 표

**왜 필요한가** — Day 89의 방식.

```
   ★ 계산

   필요 DPS = 층 총 HP / 노출 시간 / 효율 계수
```

| 층 | 적 수 | 총 HP | 노출 시간 | 필요 DPS | 플레이어 DPS | 판정 |
|:--:|:--:|:--:|:--:|:--:|:--:|---|
| 1 | 2 | 100 | 14s | 7.1 | 30.0 | ▼ 여유 |
| 2 | 3 | 180 | 17s | 10.6 | 30.0 | ▼ 여유 |
| 3 | 5 | 420 | 22s | 19.1 | 30.0 | ✔ 적정 |
| 4 | 4 | 578 | 24s | 24.1 | 30.0 | ✔ 적정 |
| 5 | 6 | 900 | 28s | 32.1 | 30.0 | ✔ 도전 |
| 6 | 1 | 1800 | 35s | 51.4 | 34.0 | ▲ 부족 |

```
   ★ 노출 시간 = 적이 살아 있는 예상 시간

   ① 스폰 간격
   ② 접근 시간
   ③ 전투 시간
```

```
   ★ 판정 기준

   플레이어 DPS > 필요 DPS × 1.5   →  여유
   플레이어 DPS ≈ 필요 DPS × 1.0~1.3 →  적정
   플레이어 DPS < 필요 DPS         →  부족
```

### 3-9. 안전 저장

**왜 필요한가** — Day 59·79의 재현.

```csharp
namespace OneFloor.Save
{
    public static class SaveSystem
    {
        private const string FileName = "save.json";

        private static string Path => System.IO.Path.Combine(
            Application.persistentDataPath, FileName);

        public static bool Save(SaveData data)
        {
            string temp = Path + ".tmp";
            string backup = Path + ".bak";

            try
            {
                string json = JsonUtility.ToJson(data, true);

                File.WriteAllText(temp, json);         // ★ ① 임시

                if (File.Exists(Path))
                {
                    if (File.Exists(backup)) File.Delete(backup);
                    File.Move(Path, backup);           // ★ ② 백업
                }

                File.Move(temp, Path);                 // ★ ③ 교체

                DebugUtil.Log($"[저장] {json.Length} bytes");
                return true;
            }
            catch (System.Exception e)
            {
                DebugUtil.LogWarning($"[저장 실패] {e.Message}");

                if (File.Exists(temp))
                {
                    try { File.Delete(temp); } catch { }
                }

                return false;
            }
        }

        public static SaveData Load()
        {
            SaveData data = TryLoad(Path);
            if (data != null) return data;

            data = TryLoad(Path + ".bak");             // ★ 백업 복구

            if (data != null)
            {
                DebugUtil.LogWarning("[로드] 백업에서 복구");
                return data;
            }

            DebugUtil.Log("[로드] 새 세이브 생성");
            return new SaveData();
        }

        private static SaveData TryLoad(string path)
        {
            if (!File.Exists(path)) return null;

            try
            {
                string json = File.ReadAllText(path);
                if (string.IsNullOrWhiteSpace(json)) return null;

                SaveData data = JsonUtility.FromJson<SaveData>(json);

                if (data == null || !data.Validate()) return null;

                data.Migrate();
                return data;
            }
            catch (System.Exception e)
            {
                DebugUtil.LogWarning($"[로드 실패] {path}: {e.Message}");
                return null;
            }
        }

        public static void Delete()
        {
            if (File.Exists(Path)) File.Delete(Path);
            if (File.Exists(Path + ".bak")) File.Delete(Path + ".bak");
        }

#if UNITY_EDITOR
        [UnityEditor.MenuItem("Tools/세이브 폴더 열기")]
        private static void OpenFolder()
            => UnityEditor.EditorUtility.RevealInFinder(Application.persistentDataPath);
#endif
    }
}
```

```csharp
    private float GetAveragePlayerDps()
    {
        float sum = 0;
        int count = 0;

        foreach (WeaponData w in database.weapons)
        {
            if (w == null) continue;

            int dmg = BalanceLoader.Instance.GetWeaponDamage(w);
            float interval = BalanceLoader.Instance.GetWeaponInterval(w);

            sum += dmg / Mathf.Max(interval, 0.01f);
            count++;
        }

        return count > 0 ? sum / count : 0;
    }
```
