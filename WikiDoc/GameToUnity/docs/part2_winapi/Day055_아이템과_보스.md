### 3-1. 파워업 시스템

**왜 필요한가** — 성장이 없으면 20분 뒤에도 같은 게임이다.

```
   파워 레벨별 탄 배치

   Lv.1        Lv.2         Lv.3          Lv.4
     ▲         ▲ ▲         ▲ ▲ ▲       ╲ ▲ ▲ ▲ ╱
     │         │ │         │ │ │        ╲│ │ │╱
     ●         ●           ●             ●

   1발        2발 평행     3발          5발 확산
```

```cpp
struct WeaponLevel
{
    int    bulletCount;
    double spreadAngle;                        // 확산 각도 (라디안)
    double fireInterval;
    int    damage;
    double speed;
};

const WeaponLevel g_weaponLevels[] = {
    { 1, 0.00, 0.12, 1, 700 },                 // Lv.1
    { 2, 0.00, 0.11, 1, 700 },                 // Lv.2 평행 2발
    { 3, 0.00, 0.10, 1, 720 },                 // Lv.3 평행 3발
    { 3, 0.20, 0.09, 2, 740 },                 // Lv.4 확산 3발
    { 5, 0.30, 0.08, 2, 760 },                 // Lv.5 확산 5발
};
```

```cpp
void Player::Fire()
{
    const WeaponLevel& w = g_weaponLevels[m_powerLevel - 1];

    for (int i = 0; i < w.bulletCount; i++)
    {
        double t = (w.bulletCount > 1)
                 ? (double)i / (w.bulletCount - 1) - 0.5 : 0.0;

        double angle = -3.14159265 / 2 + t * w.spreadAngle;
        double offsetX = (w.spreadAngle < 0.01) ? t * 24.0 : 0.0;   // 평행이면 위치를 벌린다

        m_bulletMgr->RequestFire(
            m_x + offsetX, m_y - 24,
            cos(angle) * w.speed,
            sin(angle) * w.speed,
            TAG_PLAYER_BULLET, w.damage);
    }
}
```

```cpp
{
    m_y += m_fallSpeed * dt;

    // 흡수 (플레이어가 가까우면 끌려온다)
    if (m_player && m_player->IsActive())
    {
        double dx = m_player->GetX() - m_x;
        double dy = m_player->GetY() - m_y;
        double distSq = dx*dx + dy*dy;

        const double MAGNET_R = 120.0;

        if (distSq < MAGNET_R * MAGNET_R)
        {
            double d = sqrt(distSq);
            if (d > 0.1)
            {
                m_x += dx / d * 320.0 * dt;
                m_y += dy / d * 320.0 * dt;
            }
        }
    }

    if (m_y > m_core->GetHeight() + 30) Destroy();
}
```

### 3-3. 보스 — 다단계 체력과 페이즈

**왜 필요한가** — 보스는 일반 적과 완전히 다른 존재감을 가져야 한다.

```
   보스 페이즈 (체력 비율 기준)

   HP 100% ~ 60%   PHASE 1   느린 확산탄
   HP  60% ~ 30%   PHASE 2   원형 나선 + 이동
   HP  30% ~  0%   PHASE 3   전탄막 + 조준탄
   HP       0%     DYING     폭발 연출 후 소멸
```

```cpp
enum BossPhase { BOSS_ENTER, BOSS_P1, BOSS_P2, BOSS_P3, BOSS_DYING, BOSS_DEAD };

class Boss : public PooledObject
{
    void Update(double dt) override
    {
        m_phaseTimer += dt;

        switch (m_phase)
        {
        case BOSS_ENTER:  UpdateEnter(dt);  break;
        case BOSS_P1:     UpdatePhase1(dt); break;
        case BOSS_P2:     UpdatePhase2(dt); break;
        case BOSS_P3:     UpdatePhase3(dt); break;
        case BOSS_DYING:  UpdateDying(dt);  break;
        }

        CheckPhaseTransition();
    }

    void CheckPhaseTransition()
    {
        if (m_phase == BOSS_DYING || m_phase == BOSS_DEAD) return;

        double ratio = (double)m_hp / m_maxHp;

        if      (ratio <= 0.0  && m_phase != BOSS_DYING) ChangePhase(BOSS_DYING);
        else if (ratio <= 0.30 && m_phase <  BOSS_P3)    ChangePhase(BOSS_P3);
        else if (ratio <= 0.60 && m_phase <  BOSS_P2)    ChangePhase(BOSS_P2);
    }

    void ChangePhase(BossPhase p)
    {
        m_phase = p;
        m_phaseTimer = 0.0;
        m_fireTimer = 0.5;                     // 전환 직후 잠시 쉼

        if (p == BOSS_DYING)
        {
            m_deathTimer = 3.0;
            g_bulletMgr->ClearEnemyBullets();  // ★ 화면 정리
        }

        DebugLog("[Boss] 페이즈 %d 진입 (HP %d/%d)\n", p, m_hp, m_maxHp);
    }
};
```

> ### ⚠️ 페이즈 전환 시 잠시 쉬는 시간을 준다
>
> 전환 직후 바로 새 탄막이 나오면 피할 수 없다.
> **0.5~1초의 여유**가 있어야 플레이어가 대응할 수 있다.

```cpp
class Item : public PooledObject
{
public:
    void Setup(double x, double y, int type)
    {
        m_x = x; m_y = y;
        m_type = type;
        m_fallSpeed = 90.0;
        m_bobTime = 0.0;
        m_active = true;
    }

    void Update(double dt) override
    {
        m_bobTime += dt;

        m_y += m_fallSpeed * dt;
        m_x += sin(m_bobTime * 3.0) * 30.0 * dt;      // 좌우로 살랑

        ApplyMagnet(dt);

        if (m_y > m_core->GetHeight() + 30) Destroy();
    }

    void Render(Renderer& r) const override
    {
        const ItemInfo& info = g_itemInfo[m_type];

        // 반짝임
        double pulse = 0.7 + 0.3 * sin(m_bobTime * 6.0);
        int size = (int)(12 * pulse);

        r.DrawCircle((int)m_x, (int)m_y, size + 3, info.color, false);
        r.DrawCircle((int)m_x, (int)m_y, size,     info.color, true);

        char s[2] = { info.symbol, 0 };
        r.DrawText((int)m_x - 4, (int)m_y - 7, s, RGB(30, 30, 40));
    }

    int GetItemType() const { return m_type; }

private:
    void ApplyMagnet(double dt);

    int    m_type = 0;
    double m_fallSpeed = 90.0;
    double m_bobTime = 0.0;
};
```

### Step 3 — 아이템 획득

```cpp
void Player::OnCollide(GameObject* other)
{
    if (other->GetTag() == TAG_ITEM)
    {
        Item* item = (Item*)other;
        AcquireItem(item->GetItemType());
        other->Destroy();
        return;
    }

    if (other->GetTag() == TAG_ENEMY_BULLET || other->GetTag() == TAG_ENEMY)
    {
        other->Destroy();
        OnHit();
    }
}

void Player::AcquireItem(int type)
{
    switch (type)
    {
    case ITEM_POWER:
        if (m_powerLevel < MAX_WEAPON_LEVEL) m_powerLevel++;
        else m_score += 3000;                   // 만렙이면 점수로
        break;

    case ITEM_LIFE:   m_lives = min(m_lives + 1, MAX_LIVES); break;
    case ITEM_BOMB:   m_bombs = min(m_bombs + 1, MAX_BOMBS); break;
    case ITEM_SCORE:  m_score += 5000; break;
    case ITEM_SHIELD: m_invincibleTimer = 5.0; break;
    }

    SoundManager::Get().PlaySfx("item");
    ShowPickupText(g_itemInfo[type].name);
}
```

```cpp
    // 좌우 이동
    m_x = m_baseX + sin(m_phaseTimer * 0.8) * 180.0;

    m_fireTimer -= dt;
    if (m_fireTimer <= 0)
    {
        FireSpread(m_x, m_y + 40, 7, 1.2, 180);
        m_fireTimer = 1.4;
    }
}

void Boss::UpdatePhase2(double dt)
{
    m_x = m_baseX + sin(m_phaseTimer * 1.4) * 240.0;
    m_y = m_baseY + sin(m_phaseTimer * 0.9) * 40.0;

    m_fireTimer -= dt;
    if (m_fireTimer <= 0)
    {
        FireCircle(m_x, m_y, 14, 150, m_spiralAngle);
        m_spiralAngle += 0.24;                  // 나선
        m_fireTimer = 0.45;
    }
}

void Boss::UpdatePhase3(double dt)
{
    m_x = m_baseX + sin(m_phaseTimer * 2.0) * 260.0;

    m_fireTimer -= dt;
    if (m_fireTimer <= 0)
    {
        FireCircle(m_x, m_y, 18, 170, m_spiralAngle);
        m_spiralAngle -= 0.31;

        FireAimed(m_x, m_y + 40, m_playerX, m_playerY, 300);

        m_fireTimer = 0.35;
    }
}
```

## 6. 자주 막히는 곳

| 증상 | 원인 | 해결 |
|---|---|---|
| 리스폰 직후 즉사 | 적 총알 미제거 | `ClearEnemyBullets()` |
| 무적인데 죽음 | 무적 검사 누락 | `OnHit` 맨 위에서 확인 |
| 페이즈 전환 직후 즉사 | 여유 시간 없음 | `m_fireTimer = 0.5` |
| 보스가 안 죽음 | 체력 판정 순서 | `ratio <= 0.0` 먼저 |
| 아이템이 안 떨어짐 | `OnDespawn`에서 처리 안 함 | 격추 여부 확인 |
| 파워업이 너무 강함 | 밸런스 | 레벨별 데미지 조정 |
| 탄막을 피할 수 없음 | 판정이 큼 | 플레이어 판정을 6×6으로 |
| 클리어가 안 됨 | 보스 활성 판정 | `!m_boss->IsActive()` |
| 보스 폭발이 짧음 | 연출 시간 부족 | 3초 이상 |
