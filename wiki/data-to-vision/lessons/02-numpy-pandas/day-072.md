### Step 1. 수집 실행기

```python
"""src/collect.py — 오늘의 수집 실행기."""

import json
import logging
import os
import time
from datetime import datetime
from pathlib import Path

import pandas as pd
import requests
from dotenv import load_dotenv

load_dotenv()

TODAY = datetime.now().strftime("%Y-%m-%d")
RAW_DIR = Path("data/raw") / TODAY
RAW_DIR.mkdir(parents=True, exist_ok=True)

LOG_PATH = Path("data/raw/collection_log.csv")
FAIL_PATH = RAW_DIR / "failures.csv"

DATA_KEY = os.getenv("DATA_GO_KR_KEY")
KAKAO_KEY = os.getenv("KAKAO_REST_KEY")

BASE = "http://apis.data.go.kr/B551011/KorService1"
COMMON = {"serviceKey": DATA_KEY, "MobileOS": "ETC",
          "MobileApp": "TravelProject", "_type": "json"}

HEADERS = {"User-Agent": "TravelProject/1.0 (educational-purpose)"}

DELAY = 0.3
TIMEOUT = 15
RETRIES = 2
DAILY_LIMIT = 1000

logging.basicConfig(level=logging.INFO,
                    format="%(asctime)s %(message)s", datefmt="%H:%M:%S")
log = logging.getLogger("collect")

_calls = {"count": 0}
_failures = []


class QuotaExceeded(Exception):
    """일일 한도 초과."""


class Blocked(Exception):
    """접근 거부."""


# ══════════ 호출 ══════════

def call_api(operation, **params):
    """TourAPI 호출. 실패 시 None."""
    if _calls["count"] >= DAILY_LIMIT:
        raise QuotaExceeded(f"일일 한도 {DAILY_LIMIT} 도달")

    url = f"{BASE}/{operation}"
    query = {**COMMON, **params}

    for attempt in range(RETRIES + 1):
        time.sleep(DELAY)
        _calls["count"] += 1

        try:
            res = requests.get(url, params=query, headers=HEADERS,
                               timeout=TIMEOUT)
        except requests.RequestException as e:
            wait = 2 ** attempt
            log.warning(f"  {type(e).__name__} — {wait}초 후 재시도")
            time.sleep(wait)
            continue

        if res.status_code in (401, 403):
            raise Blocked(f"HTTP {res.status_code}")

        if res.text.lstrip().startswith("<"):
            log.warning("  XML 응답 — 재시도")
            continue

        try:
            data = res.json()
        except json.JSONDecodeError:
            continue

        header = data.get("response", {}).get("header", {})
        code = str(header.get("resultCode", "")).strip()

        if code.zfill(2) == "22":
            raise QuotaExceeded(f"[22] {header.get('resultMsg')}")

        if code not in ("00", "0000"):
            log.warning(f"  API [{code}] {header.get('resultMsg')}")
            return None

        return data

    return None


def extract_items(data):
    """응답에서 항목 리스트를 안전하게 꺼냅니다."""
    body = (data or {}).get("response", {}).get("body", {})
    items = body.get("items")

    if not items:
        return []

    item = items.get("item") if isinstance(items, dict) else items

    if item is None:
        return []
    return [item] if isinstance(item, dict) else item


# ══════════ ① 관광지 목록 ══════════

def collect_spots(area_code=6, content_types=(12, 14, 28, 38, 39),
                  rows=100, max_pages_per_type=6):
    """지역의 관광지 목록을 유형별로 수집합니다."""
    import math

    all_items = []

    log.info(f"[관광지 목록] areaCode={area_code} · "
             f"유형 {len(content_types)}종")

    for ctype in content_types:
        cache = RAW_DIR / f"spots_area{area_code}_type{ctype}.json"

        if cache.exists():
            items = json.loads(cache.read_text(encoding="utf-8"))
            all_items.extend(items)
            log.info(f"  type={ctype} 캐시 사용 ({len(items)}건)")
            continue

        # ── 전체 건수 확인 ──
        first = call_api("areaBasedList1", numOfRows=rows, pageNo=1,
                         areaCode=area_code, contentTypeId=ctype,
                         arrange="A")

        if first is None:
            _failures.append({"소스": "spots", "대상": f"type{ctype}",
                              "사유": "첫 요청 실패"})
            continue

        total = int(first["response"]["body"].get("totalCount", 0))

        if total == 0:
            log.info(f"  type={ctype} 데이터 없음")
            continue

        pages = min(math.ceil(total / rows), max_pages_per_type)
        items = extract_items(first)

        log.info(f"  type={ctype} 전체 {total:,}건 → {pages}페이지 수집")

        for page in range(2, pages + 1):
            try:
                data = call_api("areaBasedList1", numOfRows=rows,
                                pageNo=page, areaCode=area_code,
                                contentTypeId=ctype, arrange="A")
            except QuotaExceeded:
                raise
            except Exception as e:
                _failures.append({"소스": "spots",
                                  "대상": f"type{ctype}_p{page}",
                                  "사유": str(e)[:60]})
                continue

            page_items = extract_items(data)
            if not page_items:
                break

            items.extend(page_items)

        for it in items:
            it["_content_type"] = ctype

        cache.write_text(json.dumps(items, ensure_ascii=False, indent=1),
                         encoding="utf-8")

        all_items.extend(items)
        log.info(f"  type={ctype} {len(items)}건 저장 "
                 f"(누적 {len(all_items):,}건)")

    return all_items


# ══════════ ② 상세 정보 ══════════

def collect_details(spots, limit=None, chunk_save=25):
    """관광지 상세(요금·시간)를 수집합니다. 개별 캐시."""
    detail_dir = RAW_DIR / "details"
    detail_dir.mkdir(exist_ok=True)

    targets = spots[:limit] if limit else spots
    results = []

    log.info(f"[상세 정보] {len(targets)}건")

    for i, spot in enumerate(targets, 1):
        cid = str(spot.get("contentid", ""))
        ctype = spot.get("contenttypeid", spot.get("_content_type", 12))

        if not cid:
            continue

        cache = detail_dir / f"{cid}.json"

        if cache.exists():
            results.append(json.loads(cache.read_text(encoding="utf-8")))
            continue

        try:
            data = call_api("detailIntro1", contentId=cid,
                            contentTypeId=ctype)
        except QuotaExceeded:
            log.error(f"⛔ 한도 초과 — {i}/{len(targets)} 에서 중단")
            break
        except Exception as e:
            _failures.append({"소스": "detail", "대상": cid,
                              "사유": str(e)[:60]})
            continue

        items = extract_items(data)

        if not items:
            _failures.append({"소스": "detail", "대상": cid,
                              "사유": "상세 없음"})
            continue

        record = items[0]
        record["contentid"] = cid

        cache.write_text(json.dumps(record, ensure_ascii=False),
                         encoding="utf-8")
        results.append(record)

        if i % chunk_save == 0:
            log.info(f"  {i}/{len(targets)} · 성공 {len(results)} · "
                     f"호출 {_calls['count']}")

    return results


# ══════════ ③ 지하철역 ══════════

def collect_stations(city="부산"):
    """
    지하철역 좌표를 확보합니다.
    CSV 가 있으면 사용, 없으면 역명 목록을 지오코딩합니다. (Plan B)
    """
    csv_path = Path("data/raw/busan_subway.csv")
    cache = RAW_DIR / "stations.json"

    if cache.exists():
        stations = json.loads(cache.read_text(encoding="utf-8"))
        log.info(f"[지하철역] 캐시 사용 ({len(stations)}개)")
        return stations

    # ── Plan A: CSV ──
    if csv_path.exists():
        df = pd.read_csv(csv_path, encoding="utf-8-sig")

        lat_col = next((c for c in df.columns
                        if "위도" in c or c.lower() in ("lat", "y")), None)
        lng_col = next((c for c in df.columns
                        if "경도" in c or c.lower() in ("lng", "lon", "x")), None)
        name_col = next((c for c in df.columns
                         if "역명" in c or "name" in c.lower()), df.columns[0])

        if lat_col and lng_col:
            stations = [{"name": r[name_col],
                         "lat": float(r[lat_col]), "lng": float(r[lng_col])}
                        for _, r in df.iterrows()
                        if pd.notna(r[lat_col]) and pd.notna(r[lng_col])]

            log.info(f"[지하철역] CSV 에서 {len(stations)}개 확보")
            cache.write_text(json.dumps(stations, ensure_ascii=False),
                             encoding="utf-8")
            return stations

        log.warning("[지하철역] CSV 에 좌표 열이 없음 — Plan B 로 전환")
        names = df[name_col].dropna().unique().tolist()
    else:
        log.warning("[지하철역] CSV 없음 — Plan B (역명 지오코딩)")
        names = ["부산역", "서면역", "해운대역", "남포역", "광안역",
                 "센텀시티역", "사상역", "동래역", "연산역", "덕천역"]

    # ── Plan B: 지오코딩 ──
    if not KAKAO_KEY:
        log.error("[지하철역] KAKAO_REST_KEY 없음 — 수집 불가")
        _failures.append({"소스": "station", "대상": "전체",
                          "사유": "카카오 키 없음"})
        return []

    stations = []
    for name in names:
        time.sleep(0.15)
        query = name if name.endswith("역") else f"{name}역"

        try:
            res = requests.get(
                "https://dapi.kakao.com/v2/local/search/keyword.json",
                params={"query": f"{city} {query}", "size": 1,
                        "category_group_code": "SW8"},
                headers={"Authorization": f"KakaoAK {KAKAO_KEY}"},
                timeout=10)
            docs = res.json().get("documents", [])
        except Exception as e:
            _failures.append({"소스": "station", "대상": name,
                              "사유": type(e).__name__})
            continue

        if docs:
            stations.append({"name": name,
                             "lat": float(docs[0]["y"]),
                             "lng": float(docs[0]["x"])})
        else:
            _failures.append({"소스": "station", "대상": name,
                              "사유": "검색 결과 없음"})

    log.info(f"[지하철역] 지오코딩으로 {len(stations)}개 확보")
    cache.write_text(json.dumps(stations, ensure_ascii=False),
                     encoding="utf-8")
    return stations


# ══════════ 실행 ══════════

def run(area_code=6, detail_limit=400):
    """오늘의 수집을 실행합니다."""
    started = datetime.now()
    log.info("=" * 62)
    log.info(f"수집 시작 — {started:%Y-%m-%d %H:%M:%S}")
    log.info("=" * 62)

    stats = {}

    try:
        spots = collect_spots(area_code=area_code)
        stats["spots"] = len(spots)

        details = collect_details(spots, limit=detail_limit)
        stats["details"] = len(details)

        stations = collect_stations()
        stats["stations"] = len(stations)

        status = "완료"

    except QuotaExceeded as e:
        log.error(f"⛔ {e}")
        status = f"한도 초과: {e}"
        spots = spots if "spots" in dir() else []
        details, stations = [], []

    except Blocked as e:
        log.error(f"⛔ 차단: {e}")
        status = f"차단: {e}"
        spots, details, stations = [], [], []

    ended = datetime.now()
```

```python
"""src/integrate.py — 소스 3종을 하나로."""

import json
import re
from datetime import datetime
from pathlib import Path

import numpy as np
import pandas as pd

TODAY = datetime.now().strftime("%Y-%m-%d")
RAW_DIR = Path("data/raw") / TODAY
INTERIM_DIR = Path("data/interim")
INTERIM_DIR.mkdir(parents=True, exist_ok=True)

CONTENT_TYPE_NAME = {
    12: "관광지", 14: "문화시설", 15: "축제공연",
    25: "여행코스", 28: "레포츠", 32: "숙박",
    38: "쇼핑", 39: "음식점",
}

WALK_SPEED_KMH = 4.0
EARTH_R = 6371.0088


def haversine_matrix(lat1, lng1, lat2, lng2):
    """(n,) × (m,) → (n, m) 거리 행렬. 벡터화."""
    lat1 = np.radians(np.asarray(lat1))[:, None]
    lng1 = np.radians(np.asarray(lng1))[:, None]
    lat2 = np.radians(np.asarray(lat2))[None, :]
    lng2 = np.radians(np.asarray(lng2))[None, :]

    dlat, dlng = lat2 - lat1, lng2 - lng1
    a = np.sin(dlat/2)**2 + np.cos(lat1)*np.cos(lat2)*np.sin(dlng/2)**2

    return 2 * EARTH_R * np.arcsin(np.sqrt(np.clip(a, 0, 1)))


# ══════════ ① 소스별 로드·표준화 ══════════

def load_spots():
    """관광지 목록 JSON 들을 하나로."""
    records = []
    for path in sorted(RAW_DIR.glob("spots_*.json")):
        records += json.loads(path.read_text(encoding="utf-8"))

    df = pd.json_normalize(records)

    keep = {
        "contentid": "id", "title": "name", "addr1": "address",
        "mapx": "lng", "mapy": "lat", "contenttypeid": "type_id",
        "firstimage": "image", "tel": "tel", "sigungucode": "sigungu_code",
        "modifiedtime": "modified",
    }

    out = df[[c for c in keep if c in df.columns]].rename(columns=keep)

    out["id"] = out["id"].astype(str).str.strip()
    out["lat"] = pd.to_numeric(out["lat"], errors="coerce")
    out["lng"] = pd.to_numeric(out["lng"], errors="coerce")
    out["type_id"] = pd.to_numeric(out["type_id"], errors="coerce")
    out["category"] = out["type_id"].map(CONTENT_TYPE_NAME)
    out["has_image"] = out.get("image", "").fillna("").str.len() > 0

    return out


def load_details():
    """상세 JSON 들을 하나로."""
    detail_dir = RAW_DIR / "details"

    if not detail_dir.exists():
        return pd.DataFrame(columns=["id"])

    records = [json.loads(p.read_text(encoding="utf-8"))
               for p in detail_dir.glob("*.json")]

    if not records:
        return pd.DataFrame(columns=["id"])

    df = pd.json_normalize(records)
    df["id"] = df["contentid"].astype(str).str.strip()

    # ── 요금 필드는 유형마다 이름이 다름 ──
    fee_cols = [c for c in df.columns
                if any(k in c.lower() for k in ("usefee", "usetime",
                                                "parking", "restdate"))]

    out = df[["id"] + fee_cols].copy()

    # 요금 텍스트를 하나로 통합
    fee_text_cols = [c for c in fee_cols if "usefee" in c.lower()]
    if fee_text_cols:
        out["fee_text"] = (out[fee_text_cols]
                           .fillna("").astype(str)
                           .apply(lambda r: " ".join(x for x in r if x), axis=1)
                           .str.strip())
    else:
        out["fee_text"] = ""

    # 운영시간
    time_cols = [c for c in fee_cols if "usetime" in c.lower()]
    if time_cols:
        out["open_text"] = (out[time_cols].fillna("").astype(str)
                            .apply(lambda r: " ".join(x for x in r if x), axis=1)
                            .str.strip())
    else:
        out["open_text"] = ""

    return out[["id", "fee_text", "open_text"]]


def load_stations():
    """지하철역."""
    path = RAW_DIR / "stations.json"

    if not path.exists():
        return pd.DataFrame(columns=["name", "lat", "lng"])

    return pd.DataFrame(json.loads(path.read_text(encoding="utf-8")))


# ══════════ ② 통합 ══════════

def integrate(verbose=True):
    """소스 3종을 통합합니다."""
    report = []

    spots = load_spots()
    details = load_details()
    stations = load_stations()

    if verbose:
        print("=" * 78)
        print("                       데이터 통합")
        print("=" * 78)
        print(f"\n[① 소스별 로드]")
        print(f"  {'소스':<14}{'행':>8}{'열':>6}  주요 열")
        print("  " + "-" * 62)
        for name, df in [("관광지", spots), ("상세", details),
                         ("지하철역", stations)]:
            cols = ", ".join(df.columns[:4])
            print(f"  {name:<14}{len(df):>8,}{df.shape[1]:>6}  {cols}")

    # ══════════ 중복 제거 ══════════
    before = len(spots)
    spots = spots.drop_duplicates(subset=["id"], keep="first").reset_index(drop=True)
    dup_removed = before - len(spots)

    report.append({"단계": "중복 제거", "대상": "관광지",
                   "전": before, "후": len(spots), "비고": f"{dup_removed}건 제거"})

    # ══════════ 좌표 검증 ══════════
    valid = spots["lat"].between(33, 39) & spots["lng"].between(124, 132)
    invalid_n = (~valid).sum()

    spots["coord_valid"] = valid

    report.append({"단계": "좌표 검증", "대상": "관광지",
                   "전": len(spots), "후": int(valid.sum()),
                   "비고": f"무효 {invalid_n}건 (제거 안 함, 플래그만)"})

    # ══════════ 상세 병합 ══════════
    before_cols = spots.shape[1]
    before_rows = len(spots)

    merged = spots.merge(details, on="id", how="left",
                         validate="one_to_one", indicator="_m_detail")

    detail_matched = (merged["_m_detail"] == "both").sum()

    report.append({"단계": "상세 병합", "대상": "1:1",
                   "전": before_rows, "후": len(merged),
                   "비고": f"매칭 {detail_matched}건 "
                           f"({detail_matched/len(merged)*100:.0f}%)"})

    if len(merged) != before_rows:
        raise ValueError(f"병합으로 행이 변했습니다: "
                         f"{before_rows} → {len(merged)}")

    # ══════════ 지하철역 매칭 (좌표 기반) ══════════
    if len(stations) and merged["coord_valid"].any():
        ok = merged["coord_valid"]

        dist = haversine_matrix(
            merged.loc[ok, "lat"], merged.loc[ok, "lng"],
            stations["lat"], stations["lng"])

        nearest_idx = dist.argmin(axis=1)
        nearest_km = dist.min(axis=1)

        merged.loc[ok, "station"] = stations.iloc[nearest_idx]["name"].values
        merged.loc[ok, "station_km"] = nearest_km.round(3)
        merged.loc[ok, "walk_min"] = (nearest_km / WALK_SPEED_KMH * 60).round(1)

        within_1km = (merged["station_km"] <= 1.0).sum()

        report.append({"단계": "역 매칭", "대상": "좌표 기반",
                       "전": int(ok.sum()), "후": int(ok.sum()),
                       "비고": f"도보 15분(1km) 이내 {within_1km}건"})
    else:
        merged["station"] = pd.NA
        merged["station_km"] = np.nan
        merged["walk_min"] = np.nan

        report.append({"단계": "역 매칭", "대상": "-", "전": 0, "후": 0,
                       "비고": "⚠ 지하철역 데이터 없음 — 접근성 계산 불가"})

    merged = merged.drop(columns=[c for c in merged.columns
                                  if c.startswith("_m_")])

    # ══════════ 저장 ══════════
    out_path = INTERIM_DIR / "raw_merged.parquet"
    merged.to_parquet(out_path, index=False)

    csv_path = INTERIM_DIR / "raw_merged.csv"
    merged.to_csv(csv_path, index=False, encoding="utf-8-sig")

    report_df = pd.DataFrame(report)
    report_df.to_csv(INTERIM_DIR / "integration_report.csv",
                     index=False, encoding="utf-8-sig")

    if verbose:
        print(f"\n[② 통합 과정]")
        print(report_df.to_string(index=False))

        print(f"\n[③ 결과]  {len(merged):,}행 × {merged.shape[1]}열")

        print(f"\n  [카테고리 분포]")
        cat = merged["category"].value_counts()
        top = cat.max()
        for c, n in cat.items():
            bar = "■" * int(n / top * 22)
            print(f"    {str(c):<10}{n:>6,}  {bar}")

        print(f"\n  [주요 열 채움률]")
        key_cols = ["name", "address", "lat", "lng", "category",
                    "fee_text", "station", "walk_min", "image"]
        for c in key_cols:
            if c not in merged.columns:
                continue
            filled = merged[c].notna() & (merged[c].astype(str).str.strip() != "")
            rate = filled.mean() * 100
            mark = "⚠" if rate < 50 else " "
            bar = "■" * int(rate / 5)
            print(f"   {mark}{c:<14}{rate:>6.1f}%  {bar}")

        print(f"\n  [저장]")
        for p in (out_path, csv_path):
            print(f"    {p}  ({p.stat().st_size/1024:,.0f} KB)")

        # ── 내일을 위한 확인 ──
        print(f"\n[④ 내일(Day 073) 전처리에서 확인할 것]")

        todos = []
        fee_rate = (merged["fee_text"].fillna("").str.strip() != "").mean()
        if fee_rate < 0.5:
            todos.append(f"요금 텍스트 채움률 {fee_rate*100:.0f}% — "
                         f"무료/유료 이진값으로 축소 검토")

        if merged["station"].isna().all():
            todos.append("⚠ 지하철역 데이터 확보 필요 — 접근성이 핵심 지표")
        elif merged["walk_min"].notna().any():
            far = (merged["walk_min"] > 15).mean()
            todos.append(f"도보 15분 초과 {far*100:.0f}% — "
                         f"기준을 20분으로 완화할지 검토")

        if invalid_n:
            todos.append(f"좌표 무효 {invalid_n}건 — 재지오코딩 대상")

        dup_name = merged["name"].duplicated().sum()
        if dup_name:
            todos.append(f"동명 관광지 {dup_name}건 — 중복인지 확인")

        for t in todos or ["특이사항 없음"]:
            print(f"  · {t}")

        print("\n" + "=" * 78)

    return merged


if __name__ == "__main__":
    merged = integrate()
```

```python
        ok = actual >= target

                       "단위": unit, "달성": ok, "필수": critical})

    add("좌표 유효", valid_coord, int(len(merged) * 0.95))
```

```python
        missing_cols = [c for c in src_cols if c not in merged.columns]

        if arrived:
            fill = merged[arrived].notna().mean().mean() * 100
        else:
            fill = 0.0

        if verbose:
            note = ", ".join(missing_cols[:3]) if missing_cols else "-"
            print(f"  {name:<12}{len(src_cols):>9}{len(arrived):>7}"
                  f"{fill:>8.1f}%  {note}")

        if not arrived and src_cols:
            issues.append(("치명", f"'{name}' 의 열이 하나도 유입되지 않음"))
        elif fill < 10 and arrived:
            issues.append(("경고", f"'{name}' 유입 열의 채움률 {fill:.0f}% "
                                   f"— 키 매칭 실패 의심"))

    # ══════════ ④ 값 보존 확인 (샘플) ══════════
    if key in merged.columns and key in base.columns:
        sample_keys = merged[key].dropna().sample(
            min(20, len(merged)), random_state=1)

        mismatches = 0
        check_cols = [c for c in base.columns
                      if c != key and c in merged.columns][:5]

        for k in sample_keys:
            src_row = base[base[key] == k]
            mrg_row = merged[merged[key] == k]

            if src_row.empty or mrg_row.empty:
                continue

            for c in check_cols:
                a, b = src_row[c].iloc[0], mrg_row[c].iloc[0]
                if pd.isna(a) and pd.isna(b):
                    continue
                if str(a) != str(b):
                    mismatches += 1

        if verbose:
            print(f"\n[④ 값 보존]  샘플 {len(sample_keys)}건 × "
                  f"{len(check_cols)}열")
            print(f"  불일치: {mismatches}건")

        if mismatches:
            issues.append(("치명", f"원본과 값이 다른 셀 {mismatches}건"))

    # ══════════ ⑤ 좌표 검증 ══════════
    if {"lat", "lng"} <= set(merged.columns):
        in_korea = (merged["lat"].between(33, 39)
                    & merged["lng"].between(124, 132))
        swapped = (merged["lat"].between(124, 132)
                   & merged["lng"].between(33, 39))

        if verbose:
            print(f"\n[⑤ 좌표]")
            print(f"  한국 범위  : {in_korea.sum():,}/{len(merged):,} "
                  f"({in_korea.mean()*100:.1f}%)")
            print(f"  뒤바뀜 의심: {swapped.sum()}건")
            print(f"  결측       : {merged['lat'].isna().sum()}건")

        if swapped.any():
            issues.append(("치명", f"위경도 뒤바뀜 의심 {swapped.sum()}건"))
        if in_korea.mean() < 0.9:
            issues.append(("경고", f"한국 범위 좌표가 "
                                   f"{in_korea.mean()*100:.0f}% 뿐"))

    # ══════════ ⑥ 계산 열 검증 ══════════
    if "walk_min" in merged.columns and "station_km" in merged.columns:
        expected = merged["station_km"] / 4.0 * 60
        diff = (merged["walk_min"] - expected).abs()
        bad = (diff > 0.5).sum()

        if verbose:
            print(f"\n[⑥ 계산 열]")
            print(f"  walk_min = station_km / 4km/h × 60")
            print(f"  불일치: {bad}건")

        if bad:
            issues.append(("경고", f"도보 시간 계산 불일치 {bad}건"))

    # ══════════ 결과 ══════════
    fatal = sum(1 for l, _ in issues if l == "치명")
    warn = sum(1 for l, _ in issues if l == "경고")
    passed = fatal == 0

    if verbose:
        print(f"\n{line}")
        print(f"  판정: {'✅ 통합 정상' if passed else '❌ 재통합 필요'}"
              f"   (치명 {fatal} · 경고 {warn})")
        print(line)

        for level, msg in issues:
            print(f"  {'🔴' if level == '치명' else '🟡'} {msg}")

        if not issues:
            print("  문제 없음")

        print(line)

    return {"passed": passed, "issues": issues}


# ══════════ 사용 ══════════
if __name__ == "__main__":
    verify_integration(
        merged,
        sources={"관광지": load_spots(), "상세": load_details()},
        key="id",
    )
```
