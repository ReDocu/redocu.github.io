```cpp
class TilePalette
{
public:
    void Init(TileSet* tileSet, const UIRect& area, int cellSize);

    void Render(Renderer& r) const;

    // 클릭 좌표 → 타일 인덱스 (-1 = 빈 곳)
    int HitTest(int mx, int my) const
    {
        if (!m_area.Contains(mx, my)) return -1;

        int lx = mx - m_area.x - m_padding;
        int ly = my - m_area.y - m_padding - m_headerH;

        if (lx < 0 || ly < 0) return -1;

        int col = lx / m_cellSize;
        int row = ly / m_cellSize;

        if (col >= m_colsPerRow) return -1;

        int index = row * m_colsPerRow + col;

        return (index < m_tileSet->GetCount()) ? index : -1;
    }

    int  GetSelected() const { return m_selected; }
    void SetSelected(int i)  { m_selected = i; }

private:
    TileSet* m_tileSet = nullptr;
    UIRect   m_area;
    int      m_cellSize = 40;
    int      m_colsPerRow = 4;
    int      m_padding = 8;
    int      m_headerH = 28;
    int      m_selected = 0;
};
```

### 3-4. ⚠️ 빠른 드래그 — 보간

**왜 필요한가** — 프레임 사이에 마우스가 여러 칸을 건너뛴다.

```
   60fps 에서 마우스를 빠르게 움직이면

   프레임 N   : 타일 (5, 3)
   프레임 N+1 : 타일 (12, 3)      ← 7칸을 건너뛰었다

   결과:  ▓ ▓                     ← 중간이 비어 있다
          5     12
```

**해결 — 브레젠험 직선 알고리즘**

```cpp
void MapToolScene::PaintLine(int x0, int y0, int x1, int y1)
{
    int dx = abs(x1 - x0);
    int dy = abs(y1 - y0);
    int sx = (x0 < x1) ? 1 : -1;
    int sy = (y0 < y1) ? 1 : -1;
    int err = dx - dy;

    while (true)
    {
        PaintAt(x0, y0);

        if (x0 == x1 && y0 == y1) break;

        int e2 = err * 2;

        if (e2 > -dy) { err -= dy; x0 += sx; }
        if (e2 <  dx) { err += dx; y0 += sy; }
    }
}
```

```
   브레젠험 직선 알고리즘

   (5,3) 에서 (12,6) 으로

   ▓ ▓ ▓
       ▓ ▓ ▓
             ▓ ▓          ← 정수 연산만으로 자연스러운 직선
   5 6 7 8 9 10 11 12

   ★ 곱셈·나눗셈 없이 덧셈과 비교만으로 그린다
     1962년에 나온 알고리즘이지만 지금도 표준이다
```

**채우기 도구 (Flood Fill)**

```cpp
void MapToolScene::FloodFill(int startCol, int startRow, int newTile)
{
    Tile* start = m_map.At(startCol, startRow);
    if (!start) return;

    int oldTile = start->tileIndex;
    if (oldTile == newTile) return;            // ★ 같으면 무한 루프

    // BFS (재귀 대신 큐 — 스택 오버플로 방지)
    struct P { int c, r; };

    const int MAX_Q = 512 * 512;
    P* queue = new P[MAX_Q];
    int head = 0, tail = 0;

    queue[tail++] = { startCol, startRow };

    while (head < tail)
    {
        P p = queue[head++];

        Tile* t = m_map.At(p.c, p.r);
        if (!t || t->tileIndex != oldTile) continue;

        t->tileIndex = newTile;

        const int dc[] = { 0, 0, -1, 1 };
        const int dr[] = { -1, 1, 0, 0 };

        for (int i = 0; i < 4; i++)
        {
            int nc = p.c + dc[i], nr = p.r + dr[i];

            Tile* nt = m_map.At(nc, nr);
            if (nt && nt->tileIndex == oldTile && tail < MAX_Q)
                queue[tail++] = { nc, nr };
        }
    }

    delete[] queue;
}
```

```
   ★ 재귀 대신 큐를 쓰는 이유

   재귀 방식:  맵이 크면 스택 오버플로 (512×512 = 26만 번 재귀)
   큐 방식:    힙을 쓰므로 안전

   ★ 이 BFS 가 Day 61의 길찾기와 같은 구조다
```

### 3-7. Undo / Redo

**왜 필요한가** — 실수를 되돌릴 수 없으면 편집이 불안하다.

```cpp
struct TileChange
{
    int col, row;
    int oldTile, newTile;
};

struct Stroke                                  // 한 번의 편집 동작
{
    TileChange* changes = nullptr;
    int         count = 0;
};

class UndoStack
{
public:
    void BeginStroke()
    {
        m_recording = true;
        m_tempCount = 0;
    }

    void Record(int col, int row, int oldT, int newT)
    {
        if (!m_recording) return;
        if (m_tempCount >= MAX_TEMP) return;

        m_temp[m_tempCount++] = { col, row, oldT, newT };
    }

    void EndStroke()
    {
        m_recording = false;

        if (m_tempCount == 0) return;          // 변화 없음

        // Redo 스택 비우기 (새 편집이 일어났으므로)
        ClearRedo();

        // Undo 스택에 추가
        if (m_undoCount >= MAX_STROKES)
            RemoveOldestUndo();

        Stroke& s = m_undo[m_undoCount++];
        s.changes = new TileChange[m_tempCount];
        s.count   = m_tempCount;

        memcpy(s.changes, m_temp, sizeof(TileChange) * m_tempCount);

        m_tempCount = 0;
    }

    bool Undo(TileMap& map)
    {
        if (m_undoCount <= 0) return false;

        Stroke& s = m_undo[--m_undoCount];

        // 역순으로 되돌린다
        for (int i = s.count - 1; i >= 0; i--)
        {
            Tile* t = map.At(s.changes[i].col, s.changes[i].row);
            if (t) t->tileIndex = s.changes[i].oldTile;
        }

        m_redo[m_redoCount++] = s;             // Redo로 이동
        return true;
    }

    bool Redo(TileMap& map)
    {
        if (m_redoCount <= 0) return false;

        Stroke& s = m_redo[--m_redoCount];

        for (int i = 0; i < s.count; i++)
        {
            Tile* t = map.At(s.changes[i].col, s.changes[i].row);
            if (t) t->tileIndex = s.changes[i].newTile;
        }

        m_undo[m_undoCount++] = s;
        return true;
    }

    int GetUndoCount() const { return m_undoCount; }
    int GetRedoCount() const { return m_redoCount; }

private:
    static const int MAX_STROKES = 64;
    static const int MAX_TEMP    = 512 * 512;

    Stroke m_undo[MAX_STROKES];  int m_undoCount = 0;
    Stroke m_redo[MAX_STROKES];  int m_redoCount = 0;

    TileChange* m_temp = nullptr;  int m_tempCount = 0;
    bool m_recording = false;
};
```

```
   ★ Undo 의 단위는 "스트로크"다

   드래그로 100칸을 칠했다면
   → Undo 한 번에 100칸이 전부 되돌아가야 한다
   → 한 칸씩 되돌아가면 100번 눌러야 한다

   BeginStroke ~ EndStroke 사이의 변경을 하나로 묶는다
```

```
   ★ 새 편집이 일어나면 Redo 스택을 비운다

   A → B → C 편집 후
   Undo Undo (A 상태)
   D 편집       ← 여기서 B, C 는 사라진다

   (모든 편집기가 이렇게 동작한다)
```

### 3-8. 메모리 관리

**왜 필요한가** — Undo 스택이 메모리를 계속 먹는다.

```
   맵 512×512 를 채우기로 한 번 칠하면
   → 262,144 칸 변경
   → TileChange 하나 16바이트 = 4.2 MB

   Undo 64단계면 최대 268 MB
```

**대책**

| 방법 | 설명 |
|---|---|
| 스택 크기 제한 | 오래된 것부터 삭제 ← **기본** |
| 총 메모리 제한 | 합계가 넘으면 삭제 |
| 압축 | 같은 값 반복은 축약 |
| 스냅샷 방식 | 맵 전체를 저장 (작은 맵에 유리) |

```cpp
    void RemoveOldestUndo()
    {
        delete[] m_undo[0].changes;

        for (int i = 0; i < m_undoCount - 1; i++)
            m_undo[i] = m_undo[i + 1];

        m_undoCount--;
    }
```

```cpp
bool TileSet::Init(const char* imageKey, int tileW, int tileH)
{
    m_image = ImageManager::Get().Find(imageKey);
    if (!m_image || !m_image->IsValid()) return false;

    m_tileW = tileW; m_tileH = tileH;
    m_cols = m_image->GetWidth()  / tileW;
    m_rows = m_image->GetHeight() / tileH;

    DebugLog("[TileSet] %d개 타일 (%dx%d)\n", GetCount(), m_cols, m_rows);
    return true;
}

void TileSet::Draw(Renderer& r, int index, int x, int y, int size) const
{
    if (!m_image || index < 0 || index >= GetCount()) return;

    int sc = index % m_cols;
    int sr = index / m_cols;

    m_image->DrawPartScaled(r.GetDC(), x, y, size, size,
                            sc * m_tileW, sr * m_tileH, m_tileW, m_tileH);
}
```

## 6. 자주 막히는 곳

| 증상 | 원인 | 해결 |
|---|---|---|
| 팔레트 클릭이 맵에도 적용 | 영역 판정 없음 | `Contains` 분기 |
| 드래그 시 점선 | 보간 없음 | 브레젠험 `PaintLine` |
| Undo가 한 칸씩 | 스트로크 단위 오류 | `BeginStroke`/`EndStroke` |
| Undo 후 Redo 안 됨 | Redo 스택 미구현 | 3-7절 |
| 새 편집 후 Redo가 살아있음 | Redo 미삭제 | `EndStroke`에서 `ClearRedo` |
| 채우기가 무한 루프 | 같은 타일로 채움 | `oldTile == newTile` 검사 |
| 채우기 시 스택 오버플로 | 재귀 방식 | **큐(BFS) 방식** |
| 메모리 급증 | Undo 스택 무제한 | 개수·용량 제한 |
| 브러시 커서가 안 맞음 | 오프셋 계산 | `col - half` |
| 변화 없는데 Undo 기록 | 같은 값 재기록 | `if (t->tileIndex == newIndex) continue` |
