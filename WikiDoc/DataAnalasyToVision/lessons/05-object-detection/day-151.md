### ★ 모델 버전 규칙

**파일명만 보고 조건을 알 수 있어야 합니다.**

```
v1.0-yolo11n-960-ep120-cctv_v2-s42.pt
 │    │        │    │      │        └─ seed
 │    │        │    │      └─ 데이터셋 버전
 │    │        │    └─ 에폭
 │    │        └─ imgsz
 │    └─ 모델 아키텍처
 └─ 릴리스 버전
```

**릴리스 버전 규칙 (SemVer 변형)**

```
v{major}.{minor}

major  데이터셋이 바뀌거나 클래스가 바뀜   (v1 → v2)
minor  같은 데이터에서 설정만 바뀜        (v1.0 → v1.1)

예
  v1.0  cctv_v1 · n/640
  v1.1  cctv_v1 · n/960          ← 설정만 변경
  v2.0  cctv_v2 (하드네거티브)    ← ★ 데이터 변경
  v2.1  cctv_v2 · s/960
```

> **★ 데이터가 바뀌면 major 를 올리세요.**
> 같은 v1 안에서 데이터가 다르면 **비교가 무의미**해집니다.

```python
# 데이터셋 지문(fingerprint) 기록
import hashlib
from pathlib import Path

def dataset_fingerprint(root):
    """파일 목록 + 크기로 지문 생성"""
    h = hashlib.sha256()
    for p in sorted(Path(root).rglob("*")):
        if p.is_file() and p.suffix.lower() in (".jpg", ".png", ".txt"):
            rel = p.relative_to(root).as_posix()
            h.update(f"{rel}:{p.stat().st_size}".encode())
    return h.hexdigest()[:16]
```

    "conf_op", "iou_op", "fps_cpu",

def dataset_fingerprint(root):
    h = hashlib.sha256()
    n = 0
    for p in sorted(Path(root).rglob("*")):
        if p.is_file() and p.suffix.lower() in (".jpg", ".jpeg", ".png", ".txt"):
            h.update(f"{p.relative_to(root).as_posix()}:{p.stat().st_size}".encode())
            n += 1
    return h.hexdigest()[:12], n


def register(exp_id, version, run_dir, dataset_root, metrics,
             conf_op=None, iou_op=None, fps_cpu=None,
             status="완료", note="", copy_weights=True):
    run_dir = Path(run_dir)
    args = yaml.safe_load((run_dir / "args.yaml").read_text(encoding="utf-8"))
    fp, n_files = dataset_fingerprint(dataset_root)

    n_train = len(list((Path(dataset_root) / "images/train").glob("*.*")))
    n_val = len(list((Path(dataset_root) / "images/val").glob("*.*")))

    arch = Path(args["model"]).stem
    weights_name = (f"{version}-{arch}-{args['imgsz']}-ep{args['epochs']}"
                    f"-{Path(dataset_root).name}-s{args.get('seed', 0)}.pt")

    if copy_weights:
        dst = ROOT / "models" / weights_name
        dst.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(run_dir / "weights/best.pt", dst)

    env_path = ROOT / "experiments/environment.json"
    env_hash = hashlib.sha256(
        env_path.read_bytes()).hexdigest()[:8] if env_path.exists() else ""

    row = {
        "exp_id": exp_id, "version": version,
        "date": datetime.now().strftime("%Y-%m-%d"), "status": status,
        "model_arch": arch, "imgsz": args["imgsz"], "epochs": args["epochs"],
        "batch": args["batch"], "seed": args.get("seed"),
        "dataset": Path(dataset_root).name, "dataset_fp": fp,
        "n_train": n_train, "n_val": n_val,
        "mAP50": round(metrics["mAP50"], 4),
        "mAP50_95": round(metrics["mAP50_95"], 4),
        "P": round(metrics["P"], 4), "R": round(metrics["R"], 4),
        "conf_op": conf_op, "iou_op": iou_op, "fps_cpu": fps_cpu,
        "weights": weights_name, "run_path": str(run_dir),
        "env_hash": env_hash, "note": note,
    }

    REGISTRY.parent.mkdir(parents=True, exist_ok=True)
    new = not REGISTRY.exists()
    with REGISTRY.open("a", newline="", encoding="utf-8-sig") as f:
        w = csv.DictWriter(f, fieldnames=FIELDS)
        if new:
            w.writeheader()
        w.writerow(row)
    return row

from datetime import datetime
from pathlib import Path

import pandas as pd

def archive_experiment(exp_id, run_dir, dataset_root, include_dataset=False):
    """실험 하나를 재현 가능한 패키지로 묶기"""
    run_dir = Path(run_dir)

        src = run_dir / "weights" / w
        if src.exists():

    # ② 설정

    if (ds / "data.yaml").exists():
        shutil.copy2(ds / "data.yaml", out / "data.yaml")
    if (ds / "DATASET_CARD.md").exists():

```python
"""src/sync_experiment.py — runs/ 에서 Git 에 넣을 것만 골라 복사"""
import shutil
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
KEEP = ["args.yaml", "results.csv", "results.png",
        "confusion_matrix.png", "labels.jpg"]
MAX_KB = 500


def sync(exp_id, run_dir):
    dst = ROOT / "reports" / exp_id
    dst.mkdir(parents=True, exist_ok=True)
    copied = []
    for name in KEEP:
        src = Path(run_dir) / name
        if src.exists() and src.stat().st_size <= MAX_KB * 1024:
            shutil.copy2(src, dst / name)
            copied.append((name, src.stat().st_size / 1024))
    return dst, copied


dst, copied = sync("E05", "C:/vision/runs/w30/E05_n_960_hardneg")
print(f"{dst} 에 {len(copied)}개 복사")
for n, kb in copied:
    print(f"  {kb:>7.0f} KB  {n}")
print(f"  합계 {sum(kb for _, kb in copied):.0f} KB")
```
