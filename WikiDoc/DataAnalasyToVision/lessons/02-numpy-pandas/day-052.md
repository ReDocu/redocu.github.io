어제 HTML 구조와 선택자를 배웠습니다. 오늘은 **실제로 가져와서 뽑아냅니다.**

```

```


### 숫자 정제

수집한 텍스트는 **거의 항상 문자열**입니다.

```python
"1,234원"    →  1234
"평점 4.5"    →  4.5
"조회 1.2만"  →  12000
```

```python
import re


def to_int(text, default=0):
    """문자열에서 정수를 추출합니다."""
    if not text:
        return default
    digits = re.sub(r"[^\d-]", "", str(text))
    return int(digits) if digits and digits != "-" else default


def to_float(text, default=0.0):
    """문자열에서 실수를 추출합니다."""
    if not text:
        return default
    match = re.search(r"-?\d+\.?\d*", str(text).replace(",", ""))
    return float(match.group()) if match else default
```

```python
to_int("1,234원")        # 1234
to_int("조회수 5,678")    # 5678
to_float("평점 4.5점")    # 4.5
to_float("₩12,500")      # 12500.0
```

### Step 2. 안전한 추출 함수 만들기

`safe_extract.py`

```python
"""안전한 HTML 추출 도구 — 요소가 없어도 죽지 않습니다."""

import re
from urllib.parse import urljoin

from bs4 import BeautifulSoup


# ══════════ 안전한 추출 함수 ══════════

def get_text(parent, selector=None, default="", separator=" "):
    """
    선택자로 텍스트를 안전하게 추출합니다.

    parent    : BeautifulSoup 요소
    selector  : CSS 선택자 (None 이면 parent 자신)
    default   : 없을 때 반환값
    """
    if parent is None:
        return default

    el = parent if selector is None else parent.select_one(selector)
    if el is None:
        return default

    return el.get_text(separator, strip=True)


def get_attr(parent, selector, attr, default=""):
    """선택자로 속성을 안전하게 추출합니다."""
    if parent is None:
        return default

    el = parent.select_one(selector)
    if el is None:
        return default

    value = el.get(attr, default)
    # class 등은 리스트로 반환되므로 문자열로 합침
    return " ".join(value) if isinstance(value, list) else value


def get_link(parent, selector, base_url="", default=""):
    """링크를 절대 URL 로 변환해 반환합니다."""
    href = get_attr(parent, selector, "href", default)
    return urljoin(base_url, href) if href and base_url else href


def get_all_text(parent, selector, default=None):
    """선택자에 해당하는 모든 요소의 텍스트를 리스트로 반환합니다."""
    if parent is None:
        return default if default is not None else []
    return [el.get_text(" ", strip=True) for el in parent.select(selector)]


def has_class(el, class_name):
    """요소가 특정 클래스를 가지고 있는지 확인합니다."""
    if el is None:
        return False
    return class_name in el.get("class", [])


# ══════════ 숫자 정제 ══════════

def to_int(text, default=0):
    """문자열에서 정수를 추출합니다. '1,234원' → 1234"""
    if text is None or text == "":
        return default

    digits = re.sub(r"[^\d-]", "", str(text))
    if not digits or digits in ("-", ""):
        return default

    try:
        return int(digits)
    except ValueError:
        return default


def to_float(text, default=0.0):
    """문자열에서 실수를 추출합니다. '평점 4.5점' → 4.5"""
    if text is None or text == "":
        return default

    cleaned = str(text).replace(",", "")
    match = re.search(r"-?\d+\.?\d*", cleaned)
    if not match:
        return default

    try:
        return float(match.group())
    except ValueError:
        return default


def to_int_korean(text, default=0):
    """한국식 축약 표기를 정수로. '1.2만' → 12000, '3천' → 3000"""
    if not text:
        return default

    s = str(text).replace(",", "").strip()

    multipliers = {"만": 10_000, "천": 1_000, "억": 100_000_000}
    for unit, mul in multipliers.items():
        if unit in s:
            num = to_float(s.split(unit)[0])
            return int(num * mul)

    return to_int(s, default)


# ══════════ 테스트 ══════════
if __name__ == "__main__":
    HTML = """
    <ul class="list">
        <li class="item featured">
            <h3 class="name">완전한 항목</h3>
            <span class="price">12,500원</span>
            <span class="rating">평점 4.7</span>
            <span class="views">1.2만</span>
            <a class="link" href="/detail/1">상세</a>
            <span class="tag">신상</span>
            <span class="tag">인기</span>
        </li>
        <li class="item">
            <h3 class="name">가격이 없는 항목</h3>
            <span class="rating">평점 3.9</span>
            <a class="link" href="/detail/2">상세</a>
        </li>
        <li class="item">
            <span class="price">무료</span>
        </li>
    </ul>
    """

    soup = BeautifulSoup(HTML, "html.parser")
    BASE = "https://example.com/list"

    print("=" * 70)
    print("             안전한 추출 — 요소가 없어도 동작")
    print("=" * 70)

    # ── ❌ 위험한 방식 ──
    print(f"\n[❌ 위험한 방식]")
    try:
        for item in soup.select(".item"):
            name = item.select_one(".name").get_text(strip=True)
            price = item.select_one(".price").get_text(strip=True)
            print(f"  {name} — {price}")
    except AttributeError as e:
        print(f"  💥 AttributeError: {e}")
        print(f"  → 두 번째 항목에 .price 가 없어서 전체가 멈췄습니다")

    # ── ✅ 안전한 방식 ──
    print(f"\n[✅ 안전한 방식]")
    results = []

    for i, item in enumerate(soup.select(".item"), 1):
        row = {
            "no": i,
            "name": get_text(item, ".name", default="(이름 없음)"),
            "price_text": get_text(item, ".price", default="0"),
            "price": to_int(get_text(item, ".price")),
            "rating": to_float(get_text(item, ".rating")),
            "views": to_int_korean(get_text(item, ".views")),
            "link": get_link(item, ".link", BASE),
            "tags": get_all_text(item, ".tag"),
            "featured": has_class(item, "featured"),
        }
        results.append(row)

    print(f"\n  {'no':<4}{'이름':<18}{'가격':>8}{'평점':>6}{'조회':>8}  추천  태그")
    print("  " + "-" * 66)
    for r in results:
        star = "★" if r["featured"] else " "
        tags = ",".join(r["tags"]) if r["tags"] else "-"
        print(f"  {r['no']:<4}{r['name']:<18}{r['price']:>8,}"
              f"{r['rating']:>6.1f}{r['views']:>8,}   {star}   {tags}")

    print(f"\n  링크 (절대 URL 로 변환):")
    for r in results:
        print(f"    {r['name']:<18}{r['link'] or '(없음)'}")

    # ── 숫자 정제 테스트 ──
    print(f"\n{'=' * 70}")
    print("[숫자 정제]")
    print("=" * 70)

    tests = [
        ("to_int", to_int, ["1,234원", "조회 5,678", "₩12,500", "무료", "", None]),
        ("to_float", to_float, ["평점 4.5점", "4.7 / 5.0", "-3.2", "없음"]),

    for name, func, cases in tests:
```

```python
"""quotes.toscrape.com 에서 명언을 수집해 DataFrame 으로 만듭니다."""

import logging
import time
from urllib.parse import urljoin

import pandas as pd
import requests
from bs4 import BeautifulSoup

# ── safe_extract.py 의 함수를 여기 포함 (실제로는 import) ──
from safe_extract import get_text, get_attr, get_link, get_all_text

logging.basicConfig(level=logging.INFO,
                    format="%(asctime)s [%(levelname)s] %(message)s",
                    datefmt="%H:%M:%S")
log = logging.getLogger("crawler")

BASE_URL = "https://quotes.toscrape.com"
HEADERS = {
    "User-Agent": (
        "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
        "AppleWebKit/537.36 (KHTML, like Gecko) "
        "Chrome/120.0.0.0 Safari/537.36 (educational-purpose)"
    )
}
DELAY = 1.0
STOP_CODES = {401, 403, 429}


def fetch(url, session=None, retries=2):
    """
    페이지를 가져옵니다. 실패하면 None.
    403/429 를 받으면 즉시 중단(예외 발생)합니다.
    """
    getter = session.get if session else requests.get

    for attempt in range(retries + 1):
        time.sleep(DELAY)

        try:
            res = getter(url, headers=HEADERS, timeout=10)
        except requests.RequestException as e:
            log.warning(f"요청 실패 ({type(e).__name__}) — 시도 {attempt + 1}")
            continue

        if res.status_code in STOP_CODES:
            raise PermissionError(
                f"HTTP {res.status_code} — 수집이 거부되었습니다. 중단합니다.")

        if res.ok:
            return res

        log.warning(f"HTTP {res.status_code} — 시도 {attempt + 1}")

    log.error(f"최종 실패: {url}")
    return None


def parse_quotes(html, page_url):
    """
    한 페이지의 HTML 에서 명언을 추출합니다.
    반환: (성공 리스트, 실패 리스트)
    """
    soup = BeautifulSoup(html, "html.parser")

    results, failures = [], []

    for i, block in enumerate(soup.select("div.quote")):
        try:
            text = get_text(block, "span.text")
            author = get_text(block, "small.author")

            if not text or not author:              # 필수 항목 검증
                failures.append({"index": i, "reason": "필수 항목 누락"})
                continue

            results.append({
                "text": text.strip("“”\""),      # 따옴표 제거
                "author": author,
                "author_url": get_link(block, "span a", BASE_URL),
                "tags": ", ".join(get_all_text(block, "a.tag")),
                "tag_count": len(get_all_text(block, "a.tag")),
                "length": len(text),
                "source_page": page_url,
            })

        except Exception as e:
            failures.append({"index": i, "reason": f"{type(e).__name__}: {e}"})

    return results, failures


def find_next_page(html, current_url):
    """다음 페이지 URL 을 찾습니다. 없으면 None."""
    soup = BeautifulSoup(html, "html.parser")
    next_link = soup.select_one("li.next a")

    if next_link and next_link.get("href"):
        return urljoin(current_url, next_link["href"])
    return None


def crawl(start_url, max_pages=3):
    """
    페이지를 따라가며 수집합니다.

    max_pages : 수집 상한 (안전장치)
    """
    session = requests.Session()

    all_quotes, all_failures = [], []
    url = start_url
    page = 0

    log.info(f"수집 시작 — 최대 {max_pages}페이지")

    while url and page < max_pages:
        page += 1
        log.info(f"[{page}/{max_pages}] {url}")

        try:
            res = fetch(url, session=session)
        except PermissionError as e:
            log.error(f"⛔ {e}")
            break

        if res is None:
            log.warning(f"페이지 {page} 실패 — 다음으로")
            break

        quotes, failures = parse_quotes(res.text, url)
        all_quotes.extend(quotes)
        all_failures.extend(failures)

        log.info(f"  → {len(quotes)}건 수집 (누적 {len(all_quotes)}건)")

        url = find_next_page(res.text, url)

        if url is None:
            log.info("  → 마지막 페이지")

    session.close()
    return all_quotes, all_failures


# ══════════ 실행 ══════════
if __name__ == "__main__":
    quotes, failures = crawl(BASE_URL, max_pages=3)

    print("\n" + "=" * 76)
    print(f"  수집 완료: 성공 {len(quotes)}건 · 실패 {len(failures)}건")
    print("=" * 76)

    if not quotes:
        print("  수집된 데이터가 없습니다.")

    df = pd.DataFrame(quotes)
```

```python
import re
import time
from urllib.parse import urljoin

import pandas as pd
import requests
from bs4 import BeautifulSoup

BASE_URL = "https://example.com/list"
HEADERS = {"User-Agent": "Mozilla/5.0 ... (educational-purpose)"}


# ── 안전한 추출 도구 ──
def get_text(parent, selector, default=""):
    el = parent.select_one(selector)
    return el.get_text(" ", strip=True) if el else default


def get_attr(parent, selector, attr, default=""):
    el = parent.select_one(selector)
    return el.get(attr, default) if el else default


def to_int(text, default=0):
    digits = re.sub(r"[^\d-]", "", str(text or ""))
    return int(digits) if digits and digits != "-" else default


# ── ① 요청 (헤더·타임아웃·지연) ──
time.sleep(1.0)
try:
    res = requests.get(BASE_URL, headers=HEADERS, timeout=10)
except requests.RequestException as e:
    print(f"요청 실패: {e}")
    raise SystemExit

# ── ② 상태 코드 확인 ──
if res.status_code in (401, 403, 429):
    print(f"⛔ HTTP {res.status_code} — 수집이 거부되었습니다")
    raise SystemExit

if not res.ok:
    print(f"HTTP {res.status_code} — 중단")
    raise SystemExit

res.encoding = res.apparent_encoding or "utf-8"
soup = BeautifulSoup(res.text, "html.parser")

# ── ③~⑥ 안전한 추출 ──
items, failures = [], []

for i, item in enumerate(soup.select(".product")):
    try:
        name = get_text(item, ".name")

        if not name:                              # 필수 항목 검증
            failures.append({"index": i, "reason": "이름 없음"})
            continue

        href = get_attr(item, "a", "href")

        items.append({
            "name": name,
            "price": to_int(get_text(item, ".price")),      # ④ 안전한 변환
            "price_text": get_text(item, ".price", "-"),     # 원문도 보관
            "link": urljoin(BASE_URL, href) if href else "",  # ⑤ 절대 URL
            "tags": ", ".join(                                # ⑥ 리스트 처리
                t.get_text(" ", strip=True) for t in item.select(".tag")
            ),
        })

    except Exception as e:
        failures.append({"index": i, "reason": f"{type(e).__name__}: {e}"})

print(f"수집 {len(items)}건 · 실패 {len(failures)}건")

if items:
    df = pd.DataFrame(items)
    df.to_csv("data/products.csv", index=False, encoding="utf-8-sig")
```

| 항목 | 개선 |
|---|---|
| 요청 | `headers` + `timeout` + `sleep` |
| 상태 | 403/429 즉시 중단, `ok` 확인 |
| 추출 | 없어도 기본값 반환 |
| 숫자 | 정규식으로 정제 |
| 링크 | `urljoin` 으로 절대화 |
| 리스트 | 반복문으로 텍스트 수집 |
| 예외 | 개별 항목 실패를 기록하고 계속 |
| 원문 보관 | `price_text` — 나중에 검증 가능 |

```python
import re
from urllib.parse import urljoin

import pandas as pd
from bs4 import BeautifulSoup

HTML = """..."""          # 위 HTML
BASE_URL = "https://example.com/board"


# ══════════ 추출 도구 ══════════
def get_text(parent, selector=None, default=""):
    if parent is None:
        return default
    el = parent if selector is None else parent.select_one(selector)
    return el.get_text(" ", strip=True) if el else default


def get_attr(parent, selector, attr, default=""):
    if parent is None:
        return default
    el = parent.select_one(selector)
    return el.get(attr, default) if el else default


def get_all_text(parent, selector):
    return [el.get_text(" ", strip=True) for el in parent.select(selector)] \
        if parent else []


def to_int_korean(text, default=0):
    """'조회 5.6만' → 56000"""
    if not text:
        return default

    s = str(text).replace(",", "")
    match = re.search(r"(\d+\.?\d*)\s*(만|천|억)?", s)
    if not match:
        return default

    num = float(match.group(1))
    unit = match.group(2)
    mul = {"만": 10_000, "천": 1_000, "억": 100_000_000}.get(unit, 1)

    return int(num * mul)


# ══════════ 파싱 ══════════
def parse_posts(html, base_url):
    """게시글을 추출합니다. 반환: (결과, 실패)"""
    soup = BeautifulSoup(html, "html.parser")
    results, failures = [], []

    for i, post in enumerate(soup.select("article.post")):
        try:
            # 제목 — 링크가 있을 수도, 없을 수도
            title_link = post.select_one("h2 a")
            if title_link:
                title = title_link.get_text(" ", strip=True)
                href = title_link.get("href", "")
                link = urljoin(base_url, href) if href else ""
            else:
                title = get_text(post, "h2")
                link = ""

            if not title:
                failures.append({"index": i, "reason": "제목 없음"})
                continue

            tags = get_all_text(post, ".tags a")

            results.append({
                "id": post.get("data-id", ""),
                "title": title,
                "link": link,
                "has_link": bool(link),
                "author": get_text(post, ".author", default="(미상)"),
                "date": get_text(post, ".date", default=""),
                "views_text": get_text(post, ".views", default="0"),
                "views": to_int_korean(get_text(post, ".views")),
                "tags": ", ".join(tags),
                "tag_count": len(tags),
            })

        except Exception as e:
            failures.append({"index": i, "reason": f"{type(e).__name__}: {e}"})

    return results, failures


# ══════════ 실행 ══════════
posts, failures = parse_posts(HTML, BASE_URL)

print("=" * 78)
print(f"  추출 {len(posts)}건 · 실패 {len(failures)}건")
print("=" * 78)

print(f"\n[DataFrame]  shape={df.shape}")

      .to_string(index=False))

print(f"\n[결측 상황]")
na = pd.DataFrame({
    "결측": df.isna().sum(),
    "빈 문자열": (df == "").sum(),
})
print(na[(na["결측"] > 0) | (na["빈 문자열"] > 0)].to_string())

print(f"\n[구조가 달랐던 부분]")
print(f"  링크 없는 글    : {(~df['has_link']).sum()}건")
print(f"  작성자 없는 글  : {(df['author'] == '(미상)').sum()}건")
print(f"  날짜 없는 글    : {df['date'].isna().sum()}건")
print(f"  태그 없는 글    : {(df['tag_count'] == 0).sum()}건")

print(f"\n[조회수 정제 결과]")
for _, r in df.iterrows():
    print(f"  {r['views_text']:<14} → {r['views']:>8,}")

print(f"\n[통계]")
print(f"  총 조회수: {df['views'].sum():,}")
print(f"  평균     : {df['views'].mean():,.0f}")
print(f"  최다     : {df.loc[df['views'].idxmax(), 'title']} "
      f"({df['views'].max():,})")

print("\n" + "=" * 78)
```

```python
"""수집 결과 검증 도구."""

import pandas as pd


def validate_crawl_result(df, required_cols=None, unique_col=None,
                          numeric_cols=None, min_rows=1, verbose=True):
    """
    수집 결과를 검증합니다.

    df            : 수집 결과 DataFrame
    required_cols : 반드시 값이 있어야 하는 열
    unique_col    : 중복이 없어야 하는 열 (예: id, url)
    numeric_cols  : 숫자여야 하는 열
    min_rows      : 최소 수집 건수

    반환: dict (passed, issues, stats)
    """
    issues = []
    stats = {}

    line = "=" * 72

    if verbose:
        print(line)
        print("                    수집 결과 검증")
        print(line)

    # ══════════ ① 건수 ══════════
    stats["rows"] = len(df)
    stats["cols"] = len(df.columns)

    if len(df) == 0:
        issues.append(("치명", "수집된 데이터가 없습니다"))
    elif len(df) < min_rows:
        issues.append(("경고", f"수집 건수 부족: {len(df)}건 < 기대 {min_rows}건"))

    if verbose:
        print(f"\n[① 규모]")
        print(f"  행 {len(df):,} × 열 {len(df.columns)}")

    if len(df) == 0:
        return {"passed": False, "issues": issues, "stats": stats}

    # ══════════ ② 필수 열 존재 ══════════
    if required_cols:
        missing = [c for c in required_cols if c not in df.columns]
        if missing:
            issues.append(("치명", f"필수 열 누락: {missing}"))

        if verbose:
            print(f"\n[② 필수 열]")
            for c in required_cols:
                mark = "✓" if c in df.columns else "✗"
                print(f"  {mark} {c}")

    # ══════════ ③ 결측·빈 문자열 ══════════
    if verbose:
        print(f"\n[③ 결측 / 빈 값]")
        print(f"  {'열':<18}{'결측':>8}{'빈문자열':>10}{'비율':>8}")
        print("  " + "-" * 44)

    for col in df.columns:
        na = df[col].isna().sum()
        empty = (df[col].astype(str).str.strip() == "").sum()
        total = na + empty
        pct = total / len(df) * 100

        if verbose and total > 0:
            print(f"  {col:<18}{na:>8}{empty:>10}{pct:>7.1f}%")

        # 필수 열에 빈 값이 있으면 문제
        if required_cols and col in required_cols and total > 0:
            level = "치명" if pct > 50 else "경고"
            issues.append((level, f"필수 열 '{col}' 에 빈 값 {total}건 ({pct:.1f}%)"))

        stats[f"empty_{col}"] = int(total)

    # ══════════ ④ 중복 ══════════
    dup_all = df.duplicated().sum()
    stats["duplicated"] = int(dup_all)

    if verbose:
        print(f"\n[④ 중복]")
        print(f"  완전 중복 행: {dup_all}건")

    if dup_all > 0:
        issues.append(("경고", f"완전 중복 행 {dup_all}건 — drop_duplicates() 검토"))

    if unique_col and unique_col in df.columns:
        dup_key = df[unique_col].duplicated().sum()
        stats["duplicated_key"] = int(dup_key)

        if verbose:
            print(f"  '{unique_col}' 중복: {dup_key}건")

        if dup_key > 0:
            issues.append(("치명", f"고유해야 할 '{unique_col}' 에 중복 {dup_key}건"))

    # ══════════ ⑤ 자료형 ══════════
    if numeric_cols:
        if verbose:
            print(f"\n[⑤ 숫자 열 검증]")

        for col in numeric_cols:
            if col not in df.columns:
                continue

            converted = pd.to_numeric(df[col], errors="coerce")
            failed = converted.isna().sum() - df[col].isna().sum()

            if verbose:
                mark = "✓" if failed == 0 else "✗"
                print(f"  {mark} {col:<16}변환 실패 {failed}건  "
                      f"범위 {converted.min():,.0f} ~ {converted.max():,.0f}")

            if failed > 0:
                issues.append(("경고", f"'{col}' 숫자 변환 실패 {failed}건"))

            # 모든 값이 0이면 정제 실패 의심
            if (converted.fillna(0) == 0).all():
                issues.append(("경고", f"'{col}' 이 전부 0 — 정제 규칙 확인 필요"))

    # ══════════ ⑥ 고유값 ══════════
    if verbose:
        print(f"\n[⑥ 고유값]")
        for col in df.columns:
            n = df[col].nunique()
            flag = ""
            if n == 1:
                flag = "  ⚠ 모든 값이 동일"
                issues.append(("경고", f"'{col}' 의 값이 전부 같음 — 추출 실패 의심"))
            elif n == len(df):
                flag = "  (식별자)"
            print(f"  {col:<18}{n:>6}개{flag}")

    # ══════════ 결과 ══════════
    fatal = [i for i in issues if i[0] == "치명"]
    warn = [i for i in issues if i[0] == "경고"]
    passed = len(fatal) == 0

    if verbose:
        print(f"\n{line}")
        print(f"  검증 결과: {'✅ 통과' if passed else '❌ 실패'}"
              f"   (치명 {len(fatal)}건, 경고 {len(warn)}건)")
        print(line)

        for level, msg in issues:
            icon = "🔴" if level == "치명" else "🟡"
            print(f"  {icon} [{level}] {msg}")

        if not issues:
            print("  발견된 문제 없음")

        print(line)

    return {"passed": passed, "issues": issues, "stats": stats}


# ══════════ 사용 ══════════
if __name__ == "__main__":
    # 문제가 있는 수집 결과
    bad = pd.DataFrame({
        "id": [1, 2, 2, 4],                          # 중복
        "title": ["글1", "", "글3", "글4"],           # 빈 값
        "author": ["홍길동", "김철수", "이영희", "박민수"],

        "source": ["web", "web", "web", "web"],       # 전부 같음
    })

    validate_crawl_result(
        bad,
        required_cols=["id", "title", "views"],
        unique_col="id",
        numeric_cols=["views"],
        min_rows=10,
    )
```

## 오늘의 정리

| 개념 | 핵심 |
|---|---|
| `requests.get` | `headers` `timeout` **필수** · `res.ok` 확인 |
| 인코딩 | `res.encoding = res.apparent_encoding` |
| `Session` | 연결 재사용 — 빠르고 서버 부담 적음 |
| `BeautifulSoup` | `"html.parser"`(내장) · `"lxml"`(빠름) |
| 찾기 | **`select`**(리스트) · **`select_one`**(하나 또는 `None`) |
| 텍스트 | **`get_text(" ", strip=True)`** — 붙는 것 방지 |
| 속성 | `el.get("href", "")` · **`class` 는 리스트** |
| 링크 | **`urljoin(base, href)`** 로 절대 URL |
| **안전한 추출** | `el if el else default` — 없어도 죽지 않게 |
| 개별 실패 | `try/except` 로 감싸고 **실패 기록** |
| 숫자 정제 | 정규식으로 `"1,234원"` → `1234` |
| 검증 | 수집 직후 **결측·중복·동일값** 확인 |

**오늘 반드시 기억할 세 가지**
> ① `select_one` 은 없으면 `None` 입니다. **`.get_text()` 를 바로 붙이지 마세요.**
> ② 반복문 안에서는 **`soup` 이 아니라 `item`** 에서 찾으세요. 안 그러면 전부 같은 값이 나옵니다.
> ③ 수집 직후 **검증 함수를 돌리세요.** 100건 수집 후 전부 같은 값이면 시간을 날립니다.
