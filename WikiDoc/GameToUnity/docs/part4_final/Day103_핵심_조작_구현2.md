### 3-1. 핵심 동사 식별

**왜 필요한가** — 무엇을 먼저 만들지.

### 3-2. 판정의 3가지 방식

**왜 필요한가** — 상황에 맞는 것을 고른다.

| 방식 | 코드 | 용도 |
|---|---|---|
| **겹침 검사** | `OverlapSphere/Box` | 근접 공격, 폭발 |
| **광선** | `Raycast/SphereCast` | 총, 클릭 선택 |
| **트리거** | `OnTriggerEnter` | 아이템 획득, 지속 영역 |

```csharp
    // ① 겹침 (Day 83)
    int count = Physics.OverlapBoxNonAlloc(
        center, halfExtents, buffer, rotation, targetLayer);

    // ② 광선 (Day 87)
    if (Physics.Raycast(ray, out RaycastHit hit, maxDist, targetLayer)) { }

    // ③ 트리거 (Day 73)
    private void OnTriggerEnter(Collider other) { }
```

```
   ★ NonAlloc을 쓴다
```

```csharp
namespace OneFloor.FX
{
    public class HitFlash : MonoBehaviour
    {
        [SerializeField] private Renderer[] renderers;
        [SerializeField] private Color flashColor = Color.white;
        [SerializeField] private float duration = 0.08f;

        private static readonly int ColorId = Shader.PropertyToID("_BaseColor");

        private MaterialPropertyBlock block;
        private Color[] originalColors;
        private Coroutine running;

        private void Awake()
        {
            block = new MaterialPropertyBlock();

            if (renderers == null || renderers.Length == 0)
                renderers = GetComponentsInChildren<Renderer>();

            originalColors = new Color[renderers.Length];
            for (int i = 0; i < renderers.Length; i++)
                originalColors[i] = renderers[i].sharedMaterial.color;
        }

        public void Flash()
        {
            if (running != null) StopCoroutine(running);
            running = StartCoroutine(FlashRoutine());
        }

        private IEnumerator FlashRoutine()
        {
            SetColor(flashColor);
            yield return new WaitForSecondsRealtime(duration);
            RestoreColors();
            running = null;
        }

        private void SetColor(Color c)
        {
            foreach (Renderer r in renderers)
            {
                r.GetPropertyBlock(block);
                block.SetColor(ColorId, c);
                r.SetPropertyBlock(block);     // ★ Day 85. 배칭 유지
            }
        }

        private void RestoreColors()
        {
            for (int i = 0; i < renderers.Length; i++)
            {
                renderers[i].GetPropertyBlock(block);
                block.SetColor(ColorId, originalColors[i]);
                renderers[i].SetPropertyBlock(block);
            }
        }
    }
}
```

### 3-7. 상태 관리

**왜 필요한가** — 핵심 수치 1~2개.

```csharp
namespace OneFloor.Core
{
    public class Health : MonoBehaviour, IDamageable
    {
        [SerializeField] private int maxHp = 100;
        [SerializeField] private HitFlash hitFlash;

        private int currentHp;

        public int CurrentHp => currentHp;
        public int MaxHp => maxHp;
        public bool IsAlive => currentHp > 0;
        public Transform Transform => transform;

        public event System.Action<int, int> OnHpChanged;
        public event System.Action<DamageInfo> OnDamaged;
        public event System.Action OnDied;

        private void Awake() { currentHp = maxHp; }

        public void TakeDamage(DamageInfo info)
        {
            if (!IsAlive) return;

            currentHp = Mathf.Max(0, currentHp - info.amount);

            hitFlash?.Flash();

            OnHpChanged?.Invoke(currentHp, maxHp);
            OnDamaged?.Invoke(info);

            DebugUtil.Log($"{name} 피해 {info.amount} → {currentHp}/{maxHp}");

            if (currentHp <= 0) Die();
        }

        public void Heal(int amount)
        {
            if (!IsAlive) return;

            currentHp = Mathf.Min(maxHp, currentHp + amount);
            OnHpChanged?.Invoke(currentHp, maxHp);
        }

        private void Die()
        {
            OnDied?.Invoke();
        }
    }
}
```

```
   ★ 이벤트로 분리하는 이유

   Health는 "체력"만 안다
   ① UI가 구독 → HP 바 갱신
   ② FX가 구독 → 이펙트
   ③ AI가 구독 → 사망 처리

   → 서로를 몰라도 된다
```

```
   ⚠️ 이벤트 구독은 반드시 해제 (Day 75)

   private void OnEnable()  { health.OnDied += HandleDied; }
   private void OnDisable() { health.OnDied -= HandleDied; }
```

```csharp
namespace OneFloor.AI
{
    [RequireComponent(typeof(Health))]
    public class SimpleEnemy : MonoBehaviour
    {

        private Health health;

        private void Awake() { health = GetComponent<Health>(); }

        private void OnEnable()  { health.OnDied += HandleDied; }
        private void OnDisable() { health.OnDied -= HandleDied; }

            GameManager.Instance?.OnEnemyKilled(this);

            Destroy(gameObject, deathDelay);
```

### Step 5 — 히트박스

```csharp
namespace OneFloor.Combat
{
    public class Hitbox : MonoBehaviour
    {
        [Header("판정")]
        [SerializeField] private Vector3 offset = new Vector3(0, 1f, 1f);
        [SerializeField] private Vector3 size = new Vector3(1.5f, 1.5f, 2f);
        [SerializeField] private LayerMask targetLayer;

        [Header("데미지")]
        [SerializeField] private int damage = 10;
        [SerializeField] private float knockback = 5f;

        private readonly HashSet<Collider> alreadyHit = new HashSet<Collider>();
        private readonly Collider[] buffer = new Collider[16];

        public bool IsActive { get; private set; }

        public event System.Action<IDamageable, DamageInfo> OnHit;

        public void Activate()
        {
            alreadyHit.Clear();                // ★ Day 83
            IsActive = true;
        }

        public void Deactivate() => IsActive = false;

        private void FixedUpdate()
        {
            if (!IsActive) return;

            Vector3 center = transform.TransformPoint(offset);

            int count = Physics.OverlapBoxNonAlloc(
                center, size * 0.5f, buffer, transform.rotation, targetLayer);

            for (int i = 0; i < count; i++)
            {
                Collider col = buffer[i];

                if (alreadyHit.Contains(col)) continue;
                alreadyHit.Add(col);

                IDamageable target = col.GetComponentInParent<IDamageable>();
                if (target == null || !target.IsAlive) continue;

                Vector3 dir = target.Transform.position - transform.position;
                dir.y = 0;

                DamageInfo info = new DamageInfo
                {
                    amount = damage,
                    knockback = knockback,
                    direction = dir.normalized,
                    hitPoint = col.ClosestPoint(center),
                    source = transform,
                };

                target.TakeDamage(info);
                OnHit?.Invoke(target, info);
            }
        }

        private void OnDrawGizmos()
        {
            Gizmos.color = IsActive
                ? new Color(1f, 0.2f, 0.2f, 0.5f)
                : new Color(0.5f, 0.5f, 0.5f, 0.15f);

            Matrix4x4 old = Gizmos.matrix;
            Gizmos.matrix = Matrix4x4.TRS(
                transform.TransformPoint(offset), transform.rotation, Vector3.one);

            Gizmos.DrawCube(Vector3.zero, size);
            Gizmos.DrawWireCube(Vector3.zero, size);
            Gizmos.matrix = old;
        }
    }
}
```

### Step 7 — 공격 액션

```csharp
namespace OneFloor.Combat
{
    public class PlayerCombat : MonoBehaviour
    {
        [Header("공격")]
        [SerializeField] private Hitbox hitbox;
        [SerializeField] private float startupTime = 0.15f;
        [SerializeField] private float activeTime = 0.12f;
        [SerializeField] private float recoveryTime = 0.25f;
        [SerializeField] private float inputBufferTime = 0.15f;

        private float bufferTimer;
        private Coroutine attackRoutine;

        public bool IsAttacking { get; private set; }

        private void Update()
        {
            if (Input.GetMouseButtonDown(0)) bufferTimer = inputBufferTime;
            else                             bufferTimer -= Time.deltaTime;

            if (bufferTimer > 0 && !IsAttacking)
            {
                bufferTimer = 0;
                attackRoutine = StartCoroutine(AttackRoutine());
            }
        }

        private IEnumerator AttackRoutine()
        {
            IsAttacking = true;

            yield return new WaitForSeconds(startupTime);

            hitbox.Activate();
            yield return new WaitForSeconds(activeTime);
            hitbox.Deactivate();

            yield return new WaitForSeconds(recoveryTime);

            IsAttacking = false;
            attackRoutine = null;
        }

        private void OnDisable()
        {
            // ★ 안전장치 (Day 83)
            hitbox.Deactivate();
            IsAttacking = false;
        }
    }
}
```

```csharp
    private void HandleDied()
    {

    private IEnumerator SinkRoutine()
    {
        Vector3 start = transform.position;
        float t = 0;

        while (t < 0.5f)
        {
            t += Time.deltaTime;
            transform.position = start + Vector3.down * (t / 0.5f * 2f);
            yield return null;
        }

        Destroy(gameObject);
    }
```
