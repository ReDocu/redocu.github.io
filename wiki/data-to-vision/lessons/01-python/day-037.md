```python
"""모든 게임 객체의 공통 부모."""

import pygame

from ..settings import SCREEN_WIDTH, SCREEN_HEIGHT


class Entity(pygame.sprite.Sprite):
    """float 위치를 관리하는 스프라이트."""

    def __init__(self, x, y, image):
        super().__init__()
        self.image = image
        self.rect = self.image.get_rect(center=(x, y))
        self.pos_x = float(x)
        self.pos_y = float(y)

    def sync_rect(self):
        self.rect.center = (int(self.pos_x), int(self.pos_y))

    def sync_pos(self):
        self.pos_x = float(self.rect.centerx)
        self.pos_y = float(self.rect.centery)

    def clamp_to_screen(self):
        self.rect.clamp_ip(pygame.Rect(0, 0, SCREEN_WIDTH, SCREEN_HEIGHT))
        self.sync_pos()

    def is_out_of_screen(self, margin=0):
        return (self.rect.bottom < -margin
                or self.rect.top > SCREEN_HEIGHT + margin
                or self.rect.right < -margin
                or self.rect.left > SCREEN_WIDTH + margin)

    def kill_if_out(self, margin=0):
        if self.is_out_of_screen(margin):
            self.kill()
            return True
        return False
```

```python
"""플레이어."""

import pygame

from .base import Entity
from .bullet import Bullet
from ..surfaces import make_ship
from ..utils import normalize
from ..settings import (PLAYER_SIZE, PLAYER_SPEED, PLAYER_MAX_HP,
                        PLAYER_INVINCIBLE_FRAMES,
                        PLAYER_REVIVE_INVINCIBLE_FRAMES, PLAYER_RADIUS,
                        BULLET_COOLDOWN, BLINK_INTERVAL,
                        COLOR_PLAYER, COLOR_PLAYER_HIT,
                        SCREEN_WIDTH, SCREEN_HEIGHT)


class Player(Entity):
    """조종 가능한 우주선."""

    def __init__(self, x, y):
        self.image_normal = make_ship(PLAYER_SIZE, COLOR_PLAYER)
        self.image_hit = make_ship(PLAYER_SIZE, COLOR_PLAYER_HIT)
        super().__init__(x, y, self.image_normal)

        self.speed = PLAYER_SPEED
        self.max_hp = PLAYER_MAX_HP
        self._hp = PLAYER_MAX_HP
        self.invincible_timer = 0
        self.shoot_timer = 0
        self.radius = PLAYER_RADIUS

    @property
    def hp(self):
        return self._hp

    @hp.setter
    def hp(self, value):
        self._hp = max(0, min(value, self.max_hp))

    @property
    def is_alive(self):
        return self._hp > 0

    @property
    def is_invincible(self):
        return self.invincible_timer > 0

    @property
    def can_shoot(self):
        return self.shoot_timer <= 0

    def take_damage(self, amount):
        """피해를 입습니다. 무적이면 False."""
        if self.is_invincible:
            return False
        self.hp -= amount
        self.invincible_timer = PLAYER_INVINCIBLE_FRAMES
        return True

    def heal(self, amount):
        before = self._hp
        self.hp += amount
        return self._hp - before

    def revive(self):
        self.hp = self.max_hp
        self.invincible_timer = PLAYER_REVIVE_INVINCIBLE_FRAMES
        self.pos_x = float(SCREEN_WIDTH // 2)
        self.pos_y = float(SCREEN_HEIGHT - 70)

    def shoot(self):
        """총알을 반환합니다. 쿨타임 중이면 None."""
        if not self.can_shoot:
            return None
        self.shoot_timer = BULLET_COOLDOWN
        return Bullet(self.rect.centerx, self.rect.top)

    def handle_input(self, keys):
        dx = dy = 0
        if keys[pygame.K_LEFT] or keys[pygame.K_a]:
            dx -= 1
        if keys[pygame.K_RIGHT] or keys[pygame.K_d]:
            dx += 1
        if keys[pygame.K_UP] or keys[pygame.K_w]:
            dy -= 1
        if keys[pygame.K_DOWN] or keys[pygame.K_s]:
            dy += 1

        dx, dy = normalize(dx, dy)
        self.pos_x += dx * self.speed
        self.pos_y += dy * self.speed

    def update(self):
        if self.shoot_timer > 0:
            self.shoot_timer -= 1
        if self.invincible_timer > 0:
            self.invincible_timer -= 1

        self.image = (self.image_hit
                      if self.is_invincible
                      and (self.invincible_timer // BLINK_INTERVAL) % 2 == 0
                      else self.image_normal)

        self.sync_rect()
        self.clamp_to_screen()
```

```python
import random

from .base import Entity
from ..surfaces import make_enemy
from ..settings import (ENEMY_SIZE, ENEMY_SPEED, ENEMY_HP, ENEMY_SCORE,
                        COLOR_ENEMY, COLOR_HP, COLOR_GRAY, SCREEN_WIDTH)

import pygame


class Enemy(Entity):
    """아래로 내려오는 기본 적."""

    HP = ENEMY_HP
    SPEED = ENEMY_SPEED
    SCORE = ENEMY_SCORE
    SIZE = ENEMY_SIZE
    COLOR = COLOR_ENEMY
    NAME = "기본"

    def __init__(self, x, y):
        super().__init__(x, y, make_enemy(self.SIZE, self.COLOR))
        self.hp = self.HP
        self.max_hp = self.HP
        self.radius = self.SIZE // 2

    def move(self):
        """이동 로직. 자식이 오버라이딩합니다."""
        self.pos_y += self.SPEED

    def update(self):
        self.move()
        self.sync_rect()
        self.kill_if_out()

    def take_damage(self, amount):
        """피해를 입습니다. 죽으면 True."""
        self.hp -= amount
        if self.hp > 0:
            return False
        self.kill()
        return True

    def draw_hp_bar(self, surface):
        """체력이 2 이상인 적만 표시합니다."""
        if self.max_hp <= 1:
            return
        w = self.rect.width
        x, y = self.rect.left, self.rect.top - 8
        pygame.draw.rect(surface, COLOR_GRAY, (x, y, w, 5))
        pygame.draw.rect(surface, COLOR_HP,
                         (x, y, int(w * self.hp / self.max_hp), 5))

    @classmethod
    def spawn(cls):
        x = random.randint(cls.SIZE, SCREEN_WIDTH - cls.SIZE)
        return cls(x, -cls.SIZE)
```

```python
"""게임 진행 상태 — 점수·목숨·통계."""

from .settings import PLAYER_LIVES, COMBO_MULTIPLIER_STEP


class GameStats:
    """한 판의 점수와 통계를 관리합니다."""

    def __init__(self, lives=PLAYER_LIVES):
        self.score = 0
        self.lives = lives
        self.max_lives = lives
        self.kills = 0
        self.shots = 0
        self.hits = 0
        self.combo = 0
        self.max_combo = 0
        self.stage = 1
        self.frames = 0

    def add_kill(self, enemy):
        """적 처치 — 콤보 배율을 적용해 점수를 더합니다."""
        self.kills += 1
        self.combo += 1
        self.max_combo = max(self.max_combo, self.combo)

        multiplier = 1 + self.combo // COMBO_MULTIPLIER_STEP
        self.score += enemy.SCORE * multiplier

    def reset_combo(self):
        self.combo = 0

    def lose_life(self):
        """목숨을 잃습니다. 게임 오버면 True."""
        self.lives -= 1
        self.reset_combo()
        return self.lives <= 0

    def tick(self):
        self.frames += 1

    @property
    def accuracy(self):
        return (self.hits / self.shots * 100) if self.shots else 0.0

    @property
    def multiplier(self):
        return 1 + self.combo // COMBO_MULTIPLIER_STEP

    @property
    def play_seconds(self):
        return self.frames // 60

    @property
    def is_game_over(self):
        return self.lives <= 0

    def summary(self):
        """결과 요약을 딕셔너리로 반환합니다."""
        return {
            "score": self.score,
            "kills": self.kills,
            "max_combo": self.max_combo,
            "accuracy": self.accuracy,
            "stage": self.stage,
            "play_seconds": self.play_seconds,
        }
```

```python
def draw_hud(surface, stats, player, fonts):
```

```python
"""게임 상태 관리 (Day 037 — Must 완성)."""

import random

import pygame

from .hud import draw_hud
from ..entities import Player, Enemy
from ..stats import GameStats
from ..settings import (SCREEN_WIDTH, SCREEN_HEIGHT, COLOR_BG, COLOR_TEXT,
                        COLOR_TEXT_DIM, COLOR_BULLET, COLOR_ENEMY,
                        ENEMY_SPAWN_INTERVAL, MAX_BULLETS, PLAYER_HIT_DAMAGE)
from ..utils import make_font, draw_text, draw_overlay

TITLE = "TITLE"
PLAYING = "PLAYING"
PAUSED = "PAUSED"
GAME_OVER = "GAME_OVER"


class GameManager:
    """상태 전환·갱신·그리기를 담당합니다."""

    def __init__(self, screen):
        self.screen = screen
        self.state = TITLE
        self.fonts = {
            "big": make_font(44),
            "normal": make_font(22),
            "small": make_font(15),
        }

        self.player = None
        self.stats = None
        self.all_sprites = pygame.sprite.Group()
        self.bullets = pygame.sprite.Group()
        self.enemies = pygame.sprite.Group()
        self.spawn_timer = 0

    # ══════════ 상태 ══════════

    def _change_state(self, new_state):
        self.state = new_state

    def start_game(self):
        """새 게임을 시작합니다."""
        self.player = Player(SCREEN_WIDTH // 2, SCREEN_HEIGHT - 70)
        self.stats = GameStats()

        self.all_sprites = pygame.sprite.Group(self.player)
        self.bullets = pygame.sprite.Group()
        self.enemies = pygame.sprite.Group()
        self.spawn_timer = 0

        self._change_state(PLAYING)

    # ══════════ 이벤트 ══════════

    def handle_event(self, event):
        if event.type != pygame.KEYDOWN:
            return True

        if event.key == pygame.K_ESCAPE:
            return False

        if self.state == TITLE:
            if event.key == pygame.K_RETURN:
                self.start_game()

        elif self.state == PLAYING:
            if event.key == pygame.K_p:
                self._change_state(PAUSED)

        elif self.state == PAUSED:
            if event.key == pygame.K_p:
                self._change_state(PLAYING)
            elif event.key == pygame.K_q:
                self._change_state(TITLE)

        elif self.state == GAME_OVER:
            if event.key == pygame.K_RETURN:
                self._change_state(TITLE)

        return True

    # ══════════ 갱신 ══════════

    def update(self):
        if self.state != PLAYING:
            return

        self.stats.tick()
        self._spawn_enemies()
        self._handle_input()
        self.all_sprites.update()
        self._handle_collisions()

    def _spawn_enemies(self):
        self.spawn_timer += 1
        if self.spawn_timer < ENEMY_SPAWN_INTERVAL:
            return

        self.spawn_timer = 0
        enemy = Enemy.spawn()
        self.all_sprites.add(enemy)
        self.enemies.add(enemy)

    def _handle_input(self):
        keys = pygame.key.get_pressed()
        self.player.handle_input(keys)

        if keys[pygame.K_SPACE] and len(self.bullets) < MAX_BULLETS:
            bullet = self.player.shoot()
            if bullet:
                self.all_sprites.add(bullet)
                self.bullets.add(bullet)
                self.stats.shots += 1

    def _handle_collisions(self):
        # ① 총알 ↔ 적 (공격이 먼저)
        hits = pygame.sprite.groupcollide(self.bullets, self.enemies, True, False)
        for bullet, enemy_list in hits.items():
            self.stats.hits += 1
            for enemy in enemy_list:
                if enemy.take_damage(bullet.damage):
                    self.stats.add_kill(enemy)

        # ② 플레이어 ↔ 적
        touched = pygame.sprite.spritecollide(
            self.player, self.enemies, False, pygame.sprite.collide_circle)

        if touched and self.player.take_damage(PLAYER_HIT_DAMAGE):
            if self.player.hp <= 0:
                if self.stats.lose_life():
                    self._change_state(GAME_OVER)
                else:
                    self.player.revive()
            else:
                self.stats.reset_combo()

    # ══════════ 그리기 ══════════

    def draw(self):
        screen = self.screen
        big, font, small = (self.fonts["big"], self.fonts["normal"],
                            self.fonts["small"])
        cx = SCREEN_WIDTH // 2

        screen.fill(COLOR_BG)

        if self.state == TITLE:
            draw_text(screen, "SPACE SHOOTER", big, COLOR_BULLET, (cx, 160), True)
            draw_text(screen, "ENTER = 시작    ESC = 종료", font, COLOR_TEXT,
                      (cx, 300), True)
            draw_text(screen, "이동 = 방향키/WASD   발사 = SPACE   일시정지 = P",
                      small, COLOR_TEXT_DIM, (cx, 360), True)
            return

        # 게임 화면
        self.all_sprites.draw(screen)
        for e in self.enemies:
            e.draw_hp_bar(screen)

        draw_hud(screen, self.stats, self.player, self.fonts)

        if self.state == PAUSED:
            draw_overlay(screen)
            draw_text(screen, "PAUSED", big, COLOR_BULLET, (cx, 250), True)
            draw_text(screen, "P = 계속   Q = 타이틀", small, COLOR_TEXT,
                      (cx, 320), True)

        elif self.state == GAME_OVER:
            draw_overlay(screen, 190)
            draw_text(screen, "GAME OVER", big, COLOR_ENEMY, (cx, 170), True)

            s = self.stats.summary()
            lines = [
                f"점수        {s['score']:,}",
                f"처치        {s['kills']}마리",
                f"최고 콤보    {s['max_combo']}",
                f"명중률      {s['accuracy']:.1f}%",
                f"플레이 시간  {s['play_seconds']}초",
            ]
            for i, line in enumerate(lines):
                draw_text(screen, line, font, COLOR_TEXT, (cx, 250 + i * 34), True)

            draw_text(screen, "ENTER = 타이틀로", small, COLOR_TEXT_DIM,
                      (cx, 470), True)

    def quit(self):
        pass
```

**2.** 아래 코드는 적을 죽여도 점수가 오르지 않습니다. 원인을 찾고 고치세요.

```python
def _handle_collisions(self):
    hits = pygame.sprite.groupcollide(self.bullets, self.enemies, True, True)

    for enemy in hits:
        self.stats.add_kill(enemy)
```

<details>
<summary>답 보기</summary>

**원인 두 가지**

**① `groupcollide` 는 딕셔너리를 반환합니다.**

`for enemy in hits:` 는 **키**를 꺼내는데, 키는 첫 번째 그룹(`bullets`)의 스프라이트입니다.
따라서 `enemy` 는 사실 **총알**이고, `enemy.SCORE` 에서 `AttributeError` 가 납니다.

```
AttributeError: 'Bullet' object has no attribute 'SCORE'
```

**② `dokillB=True` 로 적이 즉사합니다.**

체력이 5인 탱커도 총알 한 발에 죽습니다. `take_damage()` 를 부를 기회조차 없습니다.

**수정**

```python
def _handle_collisions(self):
    # 총알은 사라지고(True), 적은 체력 처리를 위해 남긴다(False)
    hits = pygame.sprite.groupcollide(self.bullets, self.enemies, True, False)

    for bullet, enemy_list in hits.items():
        self.stats.hits += 1
        for enemy in enemy_list:
            if enemy.take_damage(bullet.damage):     # 죽으면 True
                self.stats.add_kill(enemy)
```

**검증 방법**

```python
# 임시로 로그를 찍어 확인
for bullet, enemy_list in hits.items():
    print(f"[디버그] 총알 {type(bullet).__name__} → "
          f"적 {[type(e).__name__ for e in enemy_list]}")
```

> **교훈** — `groupcollide` 의 반환 형태(`{A: [B, B, ...]}`)를 정확히 기억하세요.
> 헷갈리면 `print(hits)` 로 한 번 찍어 보는 것이 가장 빠릅니다. ([Day 024](day-024.md))
</details>
