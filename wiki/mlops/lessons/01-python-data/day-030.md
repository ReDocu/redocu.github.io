**"동작한다"와 "제출 가능하다"는 다릅니다.**

| 영역 | 항목 |
|---|---|
| **설치** | 깨끗한 환경에서 clone → 설치 → 실행이 되는가 |
| **실행** | 명령 한 줄로 끝나는가 · 도움말이 있는가 · 종료 코드가 맞는가 |

| **테스트** | 핵심 로직에 테스트가 있는가 · 경계값을 다루는가 |
| **문서** | README가 5분 안에 실행하게 하는가 · 출처가 표기됐는가 |
| **안전** | 키가 커밋되지 않았는가 · `.gitignore` 가 있는가 |
| **재현성** | 두 번 실행해도 같은 결과인가 (멱등) |

### Step 2. 수집 단계를 실제 소스와 연결

`src/datapipe/stages/collect.py` 를 수정해 `sources` 를 쓰게 합니다.

```python
"""수집 단계 — sources 모듈을 사용합니다."""

from __future__ import annotations

import logging

import pandas as pd

from datapipe.config import FetchConfig, SourceConfig
from datapipe.context import Context
from datapipe.models import SourceError
from datapipe.sources import create_source
from datapipe.stages.base import Stage, StageError

logger = logging.getLogger(__name__)


class CollectStage(Stage):
    """설정에 정의된 소스들에서 데이터를 수집합니다.

    params:
        sources: [{name, type, url, params, ...}]  — SourceConfig 목록
        fetch: {timeout, max_attempts, min_interval, ...}
        mode: "sequential" | "thread"     여러 소스를 병렬 수집
        fail_on_empty: 결과가 0건이면 실패로 볼지
    """

    def execute(self, context: Context, data: pd.DataFrame | None) -> pd.DataFrame:
        source_configs = self.params.get("sources", [])
        if not source_configs:
            raise StageError("params.sources 가 비어 있습니다")

        fetch = FetchConfig(**self.params.get("fetch", {}))
        mode = self.params.get("mode", "sequential")

        frames: list[pd.DataFrame] = []
        failures: list[dict] = []

        if mode == "thread" and len(source_configs) > 1:
            from datapipe.utils.parallel import run_parallel

            def fetch_one(raw: dict) -> pd.DataFrame:
                return self._collect_one(raw, fetch)

            result = run_parallel(fetch_one, source_configs,
                                  mode="thread", progress=False)
            frames = [df for df in result.results if df is not None and not df.empty]
            failures = result.errors
        else:
            for raw in source_configs:
                try:
                    df = self._collect_one(raw, fetch)
                    if not df.empty:
                        frames.append(df)
                except SourceError as e:
                    logger.error("[%s] 소스 '%s' 실패: %s",
                                 self.name, raw.get("name", "?"), e)
                    failures.append({"source": raw.get("name"), "error": str(e)})

        if failures:
            context.set("collect_failures", failures)

        if not frames:
            if self.params.get("fail_on_empty", True):
                raise StageError(
                    f"수집 결과가 없습니다 (실패 {len(failures)}건). "
                    f"설정과 네트워크를 확인하세요.")
            return pd.DataFrame()

        combined = pd.concat(frames, ignore_index=True)
        combined["collected_at"] = pd.Timestamp.now()

        logger.info("[%s] 소스 %d개 중 %d개 성공 — 총 %d행",
                    self.name, len(source_configs), len(frames), len(combined))
        return combined

    def _collect_one(self, raw: dict, fetch: FetchConfig) -> pd.DataFrame:
        source_config = SourceConfig(**raw)
        source = create_source(source_config, fetch)
        records = source.collect()

        df = pd.DataFrame(records)
        if not df.empty:
            df["source"] = source_config.name
        return df
```

```python
"""명령행 인터페이스."""

from __future__ import annotations

import argparse
import json
import logging
import sys
from datetime import date, datetime
from pathlib import Path

from dotenv import load_dotenv

from datapipe import __version__
from datapipe.config import ConfigError, load_config
from datapipe.models import DataPipeError
from datapipe.runner import print_report, run_pipeline
from datapipe.state import RunState
from datapipe.utils.logging_config import setup_logging

logger = logging.getLogger(__name__)


def parse_date(value: str) -> date:
    try:
        return datetime.strptime(value, "%Y-%m-%d").date()
    except ValueError:
        raise argparse.ArgumentTypeError(f"날짜 형식은 YYYY-MM-DD 입니다: {value}")


# ==================== 명령 ====================

def cmd_run(args) -> int:
    config = load_config(args.config)
    setup_logging(args.log_level or config.log_level,
                  log_file=args.log_file)

    state = run_pipeline(
        config, run_date=args.date, force=args.force,
        force_stages=args.force_stage, only=args.only, resume=args.resume,
    )

    if args.json:
        print(json.dumps(state.to_dict(), ensure_ascii=False, indent=2))
    else:
        print_report(state)

    return 0 if state.success else 1


def cmd_validate(args) -> int:
    setup_logging(args.log_level or "INFO")
    config = load_config(args.config, strict_env=False)

    print(f"✓ 설정이 유효합니다: {args.config}\n")
    print(f"  파이프라인 : {config.name}")
    print(f"  데이터 폴더 : {config.data_dir}")
    print(f"  설정 해시  : {config.hash()}")
    print(f"\n  단계 {len(config.stages)}개 (활성 {len(config.enabled_stages)}개)")
    print(f"  {'순서':<6}{'이름':<14}{'타입':<14}{'활성':<6}")
    print("  " + "-" * 42)
    for i, s in enumerate(config.stages, start=1):
        print(f"  {i:<6}{s.name:<14}{s.type:<14}{'✓' if s.enabled else '-':<6}")

    # 수집 소스도 확인
    collect = config.stage("collect")
    if collect:
        sources = collect.params.get("sources", [])
        print(f"\n  수집 소스 {len(sources)}개")
        for s in sources:
            url = s.get("url", "")
            print(f"    {s.get('name', '?'):<14}{s.get('type', '?'):<14}"
                  f"{url[:44]}")
    return 0


def cmd_status(args) -> int:
    setup_logging(args.log_level or "WARNING")
    config = load_config(args.config, strict_env=False)

    state = RunState.load_latest(config.state_dir)
    if state is None:
        print("실행 기록이 없습니다.")
        return 0

    if args.json:
        print(json.dumps(state.to_dict(), ensure_ascii=False, indent=2))
    else:
        print_report(state)
    return 0 if state.success else 1


def cmd_history(args) -> int:
    setup_logging(args.log_level or "WARNING")
    config = load_config(args.config, strict_env=False)

    files = sorted(Path(config.state_dir).glob("run_*.json"), reverse=True)
    if not files:
        print("실행 기록이 없습니다.")
        return 0

    print(f"{'run_id':<18}{'날짜':<12}{'결과':<8}{'단계':<6}{'행 수':>10}{'시간(초)':>10}")
    print("-" * 66)
    for path in files[:args.limit]:
        try:
            d = json.loads(path.read_text(encoding="utf-8"))
        except (json.JSONDecodeError, OSError):
            continue
        stages = d.get("stages", {})
        total_sec = sum(s.get("seconds", 0) for s in stages.values())
        last_rows = max((s.get("rows_out", 0) for s in stages.values()), default=0)
        result = "✓ 성공" if d.get("success") else "✗ 실패"
        print(f"{d.get('run_id', '?'):<18}{d.get('run_date', '?'):<12}"
              f"{result:<8}{len(stages):<6}{last_rows:>10,}{total_sec:>10.1f}")
    return 0


def cmd_info(args) -> int:
    """생성된 데이터의 요약."""
    setup_logging(args.log_level or "WARNING")
    config = load_config(args.config, strict_env=False)

    import pandas as pd

    print(f"데이터 폴더: {config.data_dir}\n")
    if not config.data_dir.exists():
        print("  아직 실행되지 않았습니다. `datapipe run` 을 먼저 실행하세요.")
        return 0

    print(f"  {'단계':<14}{'파일':>7}{'행 수':>12}{'크기(MB)':>12}")
    print("  " + "-" * 46)

    for stage_dir in sorted(config.data_dir.iterdir()):
        if not stage_dir.is_dir():
            continue
        files = list(stage_dir.rglob("*.parquet"))
        if not files:
            continue
        rows = 0
        for f in files:
            try:
                rows += len(pd.read_parquet(f, columns=[]))
            except Exception:
                pass
        size = sum(f.stat().st_size for f in files) / 1024**2
        print(f"  {stage_dir.name:<14}{len(files):>7}{rows:>12,}{size:>12.2f}")
    return 0


def cmd_clean(args) -> int:
    setup_logging(args.log_level or "INFO")
    config = load_config(args.config, strict_env=False)

    import shutil

    targets = []
    if args.data:
        targets.append(config.data_dir)
    if args.state:
        targets.append(config.state_dir)
    if not targets:
        targets = [config.data_dir, config.state_dir]

    for path in targets:
        if not path.exists():
            continue
        if not args.yes:
            if input(f"{path} 를 삭제할까요? (y/N): ").strip().lower() != "y":
                print("  건너뜁니다.")
                continue
        shutil.rmtree(path)
        print(f"  삭제: {path}")
    return 0


# ==================== 파서 ====================

def build_parser() -> argparse.ArgumentParser:

    parser.add_argument("-c", "--config", type=Path,

    parser.add_argument("-l", "--log-level",
                        choices=["DEBUG", "INFO", "WARNING", "ERROR"])
    parser.add_argument("--log-file", type=Path, help="로그를 파일로도 기록")

    sub = parser.add_subparsers(dest="command")

    p_run = sub.add_parser("run", help="파이프라인 실행")

    p_run.add_argument("--json", action="store_true")
    p_run.set_defaults(func=cmd_run)

    p_val = sub.add_parser("validate", help="설정 검증")
    p_val.set_defaults(func=cmd_validate)

    p_st = sub.add_parser("status", help="최근 실행 상태")

    return parser


def main(argv: list[str] | None = None) -> int:
    load_dotenv()
    parser = build_parser()
    args = parser.parse_args(argv)

    if args.command is None:
        parser.print_help()
        return 2

    try:
        return args.func(args)
    except ConfigError as e:
```

        mark = "✓" if passed else ("✗" if critical else "△")
        line = f"  {mark} {name}"
        if detail and not passed:
            line += f"\n      → {detail}"
        print(line)

    @property
    def failed(self) -> list[Check]:
        return [c for c in self.checks if not c.passed and c.critical]

    @property
    def warnings(self) -> list[Check]:
        return [c for c in self.checks if not c.passed and not c.critical]

    offenders = []
    for path in (ROOT / "src").rglob("*.py"):
        content = path.read_text(encoding="utf-8")
        for pattern in patterns:

    report.add("보안", "소스에 하드코딩된 키 없음", not offenders,
               f"의심 파일: {offenders}")

    for entry in [".env", "data/", "config.yaml"]:
        report.add("보안", f".gitignore 에 {entry}", entry in gitignore)

    code, out = run(["git", "ls-files", ".env"])
    report.add("보안", ".env 가 커밋되지 않음", not out.strip(),
               "⚠ 키를 즉시 재발급하세요!")

    example = (ROOT / "configs/config.example.yaml").read_text(encoding="utf-8")
    report.add("보안", "예시 설정이 환경변수를 참조", "${" in example)

    # ==================== 5. 조건 4: 문서 ====================
    print("\n" + "=" * 72)
    print("5. 조건 4 — README 와 출처 표기")
    print("=" * 72)

    readme = (ROOT / "README.md").read_text(encoding="utf-8")

    for keyword in ["설치", "pip install", "사용", "datapipe run", "구조", "pytest"]:
        report.add("문서", f"README 에 '{keyword}'", keyword in readme)

    has_source = any(k in readme for k in ["출처", "데이터 출처"])
    report.add("문서", "데이터 출처 표기", has_source)

    has_license = any(k in readme for k in ["저작권", "라이선스", "License"])
    report.add("문서", "저작권·라이선스 표기", has_license)

    has_ethics = any(k in readme for k in ["robots.txt", "요청 간격", "수집 윤리"])
    report.add("문서", "수집 윤리 언급", has_ethics)

    report.add("문서", f"실행 예시 블록 4개 이상 (현재 {readme.count('```') // 2}개)",
               readme.count("```") >= 8)

    print("6. 재현성")
    print("=" * 72)

    code, out = run(["git", "status", "--porcelain"])
    report.add("재현성", "커밋되지 않은 변경 없음", not out.strip(),
               "변경사항을 커밋하세요", critical=False)

    report.add("재현성", "README 에 재현 방법", has_readme_repro, critical=False)

    print("결과")
    print("=" * 72)

    total = len(report.checks)
    passed_count = sum(1 for c in report.checks if c.passed)
    print(f"  통과 {passed_count} / {total}")

    if report.warnings:
        print(f"\n  △ 권장 사항 {len(report.warnings)}건:")

    print("\n  ✓ 모든 필수 항목을 충족합니다. 제출 준비 완료!")

````markdown
## 파이프라인 단계

```
collect  →  clean  →  transform  →  store
```

| 단계 | 하는 일 | 출력 |
|---|---|---|
| **collect** | API·크롤링으로 수집. 재시도·간격 제어·robots.txt 준수 | `data/collect/YYYYMMDD.parquet` |
| **clean** | 타입 변환 · 공백 정규화 · 중복 제거 · 도메인 규칙 · 이상치 · 결측 대치 | `data/clean/YYYYMMDD.parquet` |
| **transform** | 시간·그룹 파생 컬럼 · 등급 부여 · dtype 최적화 | `data/transform/YYYYMMDD.parquet` |
| **store** | 최종 Parquet(파티션) · 요약 CSV · 사람이 볼 CSV | `data/final/` |

각 단계는 결과를 저장하므로 **실패한 지점부터 재시작**할 수 있습니다.

## 수집 윤리

이 도구는 기본적으로 다음을 지킵니다.

- **robots.txt 준수** — 금지된 경로는 수집하지 않습니다 (`respect_robots: true`)
- **요청 간격** — 기본 1초 이상. `Crawl-delay` 가 있으면 그 값을 따릅니다
- **User-Agent 명시** — 정체와 연락처를 밝힙니다
- **지수 백오프** — 실패 시 간격을 늘려 재시도하고, `429` 는 `Retry-After` 를 존중합니다
- **메타데이터만 수집** — 저작물(기사 본문·이미지)은 저장하지 않습니다

수집 전 반드시 [적법성 점검표](docs/legal_check.md)를 작성하세요.
자세한 내용은 과정 자료 [Day 018 크롤링 실전과 윤리](https://github.com/사용자명/…)를 참고합니다.
````

````markdown
    │        │
    └───┬────┘

| 하고 싶은 것 | 고칠 곳 |

5. **실패는 시끄럽게** — 조용한 실패를 만들지 않음
6. **관측 가능** — 각 단계의 입출력·시간·품질을 기록
````

| datapipe (오늘) | Airflow (W27) |
|---|---|
| Stage | Task |
| `--date` | `execution_date` |
| `--resume` | 실패 Task 재시도 |
| `state/latest.json` | 메타DB |
| 단계 순서 | DAG |
| 캐싱 | XCom · 센서 |

**직접 만들어 봤기 때문에 Airflow가 왜 필요한지 압니다.**

로드맵 8절 5항 — "도구를 배우기 전에 그 도구가 없던 시절의 문제를 먼저 겪게 합니다."

**Airflow가 대신 해 주는 것**

| 지금 직접 만든 것 | Airflow |
|---|---|
| 순차 실행 | 의존 그래프 (병렬 가능) |
| 상태를 JSON에 | 메타DB + 웹 UI |
| cron으로 스케줄 | 내장 스케줄러 |
| 로그를 파일에 | Task별 로그 뷰 |
| 실패 시 로그 확인 | 알림 · 자동 재시도 |
| 재시작 옵션 | Clear & Rerun |

**앞으로 고칠 부분**

| 시점 | 개선 |
