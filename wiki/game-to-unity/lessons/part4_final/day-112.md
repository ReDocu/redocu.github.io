### 3-5. 화면 효과

**왜 필요한가** — 강한 순간을 강조.

```csharp
namespace OneFloor.FX
{
    public class ScreenDamageEffect : MonoBehaviour
    {
        [SerializeField] private Image overlay;
        [SerializeField] private float peakAlpha = 0.35f;
        [SerializeField] private float fadeDuration = 0.4f;

        private Coroutine routine;

        public void Flash(float intensity = 1f)
        {
            if (routine != null) StopCoroutine(routine);
            routine = StartCoroutine(FlashRoutine(intensity));
        }

        private IEnumerator FlashRoutine(float intensity)
        {
            Color c = overlay.color;
            c.a = peakAlpha * intensity;
            overlay.color = c;

            float t = 0;

            while (t < fadeDuration)
            {
                t += Time.unscaledDeltaTime;   // ★ 히트스톱 중에도

                c.a = Mathf.Lerp(peakAlpha * intensity, 0f, t / fadeDuration);
                overlay.color = c;

                yield return null;
            }

            c.a = 0;
            overlay.color = c;
            routine = null;
        }
    }
}
```

```

   HP 25% 이하면 지속적으로 맥동
```

### 3-6. 사운드 구조

**왜 필요한가** — 관리 가능하게.

```csharp
namespace OneFloor.Audio
{
    public enum SfxType
    {
        None,
        Footstep, Swing, Hit, HitCritical,
        PlayerHurt, PlayerDie,
        EnemyHurt, EnemyDie,
        Dodge, WeaponPickup,
        FloorClear, Victory, Defeat,
        UIClick, UIHover, UIDenied,
    }

    [System.Serializable]
    public class SfxEntry
    {
        public SfxType type;
        public AudioClip[] clips;              // ★ 여러 개 중 랜덤
        [Range(0f, 1f)] public float volume = 1f;
        [Range(0f, 0.3f)] public float pitchVariance = 0.08f;
        public float minInterval = 0.02f;      // ★ 중복 재생 방지
    }

    public class AudioManager : MonoBehaviour
    {
        public static AudioManager Instance { get; private set; }

        [SerializeField] private AudioSource bgmSource;
        [SerializeField] private AudioSource[] sfxSources;   // ★ 풀
        [SerializeField] private AudioSource uiSource;
        [SerializeField] private List<SfxEntry> sfxEntries = new();

        private readonly Dictionary<SfxType, SfxEntry> lookup = new();
        private readonly Dictionary<SfxType, float> lastPlayTime = new();
        private int sourceIndex;

        public void Initialize()
        {
            lookup.Clear();

            foreach (SfxEntry e in sfxEntries)
            {
                if (e.type == SfxType.None) continue;
                lookup[e.type] = e;
            }

            LoadVolumes();
        }

        public void PlaySfx(SfxType type, Vector3? position = null)
        {
            if (!lookup.TryGetValue(type, out SfxEntry entry)) return;
            if (entry.clips == null || entry.clips.Length == 0) return;

            // ★ 중복 재생 방지
            if (lastPlayTime.TryGetValue(type, out float last))
            {
                if (Time.unscaledTime - last < entry.minInterval) return;
            }

            lastPlayTime[type] = Time.unscaledTime;

            AudioClip clip = entry.clips[Random.Range(0, entry.clips.Length)];

            // ★ 소스 순환
            AudioSource src = sfxSources[sourceIndex];
            sourceIndex = (sourceIndex + 1) % sfxSources.Length;

            src.pitch = 1f + Random.Range(-entry.pitchVariance, entry.pitchVariance);

            if (position.HasValue)
            {
                src.transform.position = position.Value;
                src.spatialBlend = 1f;         // 3D
            }
            else
            {
                src.spatialBlend = 0f;         // 2D
            }

            src.PlayOneShot(clip, entry.volume * sfxVolume);
        }
    }
}
```

```
   ★ 클립 여러 개 중 랜덤

   같은 소리가 반복되면 지루하다
   → 3~4개를 돌아가며
```

```
   ★ 피치 변주

   ±8% 정도면 자연스럽다
   → 같은 클립도 다르게 들린다
```

```
   ★ 소스 풀

   PlayOneShot은 겹쳐 재생되지만
   피치를 바꾸면 이전 재생도 영향받는다
   → 소스를 여러 개 두고 순환
```

```csharp
    // 크로스페이드
    public void PlayBgm(AudioClip clip, float fadeDuration = 1f)
    {
        if (bgmSource.clip == clip) return;

        StartCoroutine(CrossFadeRoutine(clip, fadeDuration));
    }

    private IEnumerator CrossFadeRoutine(AudioClip next, float duration)
    {
        float start = bgmSource.volume;
        float t = 0;

        while (t < duration * 0.5f)
        {
            t += Time.unscaledDeltaTime;
            bgmSource.volume = Mathf.Lerp(start, 0f, t / (duration * 0.5f));
            yield return null;
        }

        bgmSource.clip = next;
        bgmSource.Play();

        t = 0;
        while (t < duration * 0.5f)
        {
            t += Time.unscaledDeltaTime;
            bgmSource.volume = Mathf.Lerp(0f, bgmVolume, t / (duration * 0.5f));
            yield return null;
        }

        bgmSource.volume = bgmVolume;
    }
```
