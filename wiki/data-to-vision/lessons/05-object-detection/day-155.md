### docstring 작성 규칙

```python
def predict(self, source, conf=None, iou=None, classes=None) -> dict:
    """단일 이미지에서 객체를 검출한다.

    Args:
        source: 입력. 다음 중 하나
            - str | Path: 이미지 파일 경로 (한글 경로 지원)
            - np.ndarray: BGR uint8 배열 (H, W, 3)
            - bytes: 인코딩된 이미지 데이터
            - PIL.Image: PIL 이미지 객체
        conf: 신뢰도 임계값. None 이면 인스턴스 기본값(0.45)
        iou: NMS IoU 임계값. None 이면 기본값(0.7)
        classes: 검출할 클래스 id 목록. None 이면 전체

    Returns:
        표준 응답 dict. 주요 키:
            count (int): 검출 개수 (0 가능)
            counts_by_class (dict): {"person": 2}
            detections (list[dict]): 신뢰도 내림차순
            image (dict): {"width": int, "height": int}
            timing_ms (dict): 단계별 소요 시간

    Raises:
        FileNotFoundError: 파일 경로가 존재하지 않을 때
        ValueError: 이미지로 디코딩할 수 없을 때
        TypeError: 지원하지 않는 입력 타입일 때

    Note:
        검출이 0개인 것은 정상이며 예외가 아니다.
        반환값은 JSON 직렬화 가능하다.

    Example:
        >>> det = Detector("models/best.pt", conf=0.45)
        >>> res = det.predict("photo.jpg")
        >>> print(res["count"], res["counts_by_class"])
        3 {'person': 2, 'box': 1}
    """
```

```
★ 4가지가 반드시 있어야 합니다
  ① Args    — 타입과 의미
  ② Returns — 구조
  ③ Raises  — 어떤 예외가 언제
  ④ Example — ★ 복붙 가능한 코드
```

### 코드 리뷰 체크리스트

**★ 리뷰는 "틀린 곳 찾기"가 아니라 "다른 사람이 쓸 수 있는가"입니다**

```
[인터페이스]
  □ 함수 이름이 하는 일을 설명하는가
  □ 인자 순서가 자연스러운가
  □ 기본값이 합리적인가
  □ ★ 반환 타입이 일관적인가

[안전성]
  □ 입력을 검증하는가
  □ 예외를 삼키지 않는가
  □ ★ 원본을 수정하지 않는가
  □ 리소스를 해제하는가 (파일 · 카메라)

[가독성]
  □ 함수가 한 가지 일만 하는가
  □ 중첩이 3단계 이내인가
  □ 매직 넘버 대신 상수를 쓰는가
  □ ★ 주석이 "무엇"이 아니라 "왜"를 설명하는가

[재사용]
  □ 절대 경로가 없는가
  □ 설정이 분리되어 있는가
  □ ★ 전역 상태에 의존하지 않는가

[문서]
  □ docstring 4요소가 있는가
  □ 사용 예시가 동작하는가
```

### Step 1. 동영상 처리 추가

```python
    # ────────────────────── detector.py 에 추가 ──────────────────────

    def predict_video(self, source, out_video=None, out_csv=None,
                      skip=1, max_frames=None, conf=None, iou=None,
                      classes=None, progress=True, draw=True,
                      on_frame=None) -> dict:
        """동영상에서 객체를 검출한다.

        Args:
            source: 동영상 경로 · 카메라 인덱스(int) · RTSP URL
            out_video: 결과 영상 저장 경로 (None 이면 저장 안 함)
            out_csv: 프레임별 검출 CSV 경로
            skip: N프레임마다 1번 추론 (1=매 프레임)
            max_frames: 처리할 최대 프레임 (None=끝까지)
            conf, iou, classes: predict() 와 동일
            progress: 진행률 출력
            draw: 결과 영상에 박스를 그릴지
            on_frame: 콜백 fn(frame_idx, frame, response) -> bool
                      False 를 반환하면 중단

        Returns:
            요약 dict:
                frames_read, frames_inferred, total_detections,
                counts_by_class, fps_processing, duration_s, ...

        Raises:
            RuntimeError: 동영상을 열 수 없을 때

        Note:
            메모리를 위해 프레임별 결과를 리스트에 쌓지 않는다.
            상세 결과는 out_csv 로 스트리밍 기록된다.
            Ctrl+C 로 중단해도 지금까지의 결과가 저장된다.

        Example:
            >>> det = Detector("models/best.pt")
            >>> stat = det.predict_video("in.mp4", out_video="out.mp4",
            ...                          out_csv="det.csv", skip=2)
            >>> print(stat["fps_processing"], stat["total_detections"])
        """
        import csv as _csv

        cap = cv2.VideoCapture(source)
        if not cap.isOpened():
            raise RuntimeError(f"동영상을 열 수 없습니다: {source}")

        w = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
        h = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
        src_fps = cap.get(cv2.CAP_PROP_FPS) or 30.0
        total = int(cap.get(cv2.CAP_PROP_FRAME_COUNT)) or 0

        writer = None
        if out_video:
            Path(out_video).parent.mkdir(parents=True, exist_ok=True)
            writer = cv2.VideoWriter(
                str(out_video), cv2.VideoWriter_fourcc(*"mp4v"),
                src_fps / max(skip, 1), (w, h))

        csv_file = csv_writer = None
        if out_csv:
            Path(out_csv).parent.mkdir(parents=True, exist_ok=True)
            csv_file = open(out_csv, "w", newline="", encoding="utf-8-sig")
            csv_writer = _csv.writer(csv_file)
            csv_writer.writerow([
                "frame", "time_s", "class_id", "class", "conf",
                "x1", "y1", "x2", "y2", "w", "h", "area_ratio"])

        counts: dict[str, int] = {}
        n_read = n_inf = n_det = 0
        t0 = time.perf_counter()
        last_vis = None
        interrupted = False

        try:
            while True:
                ok, frame = cap.read()
                if not ok:
                    break
                n_read += 1

                if (n_read - 1) % max(skip, 1) == 0:
                    res = self.predict(frame, conf=conf, iou=iou,
                                       classes=classes)
                    n_inf += 1
                    n_det += res["count"]
                    for k, v in res["counts_by_class"].items():
                        counts[k] = counts.get(k, 0) + v

                    if csv_writer:
                        t_s = round((n_read - 1) / src_fps, 3)
                        for d in res["detections"]:
                            b, s = d["box"], d["size"]
                            csv_writer.writerow([
                                n_read - 1, t_s, d["class_id"], d["class"],
                                d["conf"], b["x1"], b["y1"], b["x2"], b["y2"],
                                s["w"], s["h"], s["area_ratio"]])

                    if writer and draw:
                        last_vis = self.draw(frame, res)
                    elif writer:
                        last_vis = frame

                    if on_frame is not None:
                        if on_frame(n_read - 1, frame, res) is False:
                            break

                    if progress and n_inf % 20 == 0:
                        el = time.perf_counter() - t0
                        pct = (f" {n_read/total*100:5.1f}%" if total else "")
                        print(f"\r  {n_read:>6}/{total or '?'}{pct}  "
                              f"추론 {n_inf}  검출 {n_det}  "
                              f"{n_inf/max(el, 1e-6):5.2f} FPS", end="")

                if writer and last_vis is not None:
                    writer.write(last_vis)

                if max_frames and n_read >= max_frames:
                    break
        except KeyboardInterrupt:                      # ★ 중단해도 결과 보존
            interrupted = True
            logger.warning("사용자 중단 — 지금까지의 결과를 저장합니다")
        finally:
            cap.release()
            if writer:
                writer.release()
            if csv_file:
                csv_file.close()
            if progress:
                print()

        dt = time.perf_counter() - t0
        return {
            "source": str(source),
            "video": {"width": w, "height": h, "fps": round(src_fps, 2),
                      "total_frames": total},
            "frames_read": n_read,
            "frames_inferred": n_inf,
            "skip": skip,
            "total_detections": n_det,
            "counts_by_class": counts,
            "avg_detections": round(n_det / max(n_inf, 1), 2),
            "duration_s": round(dt, 2),
            "fps_processing": round(n_inf / dt, 2) if dt else 0,
            "realtime_factor": round((n_read / src_fps) / dt, 2) if dt else 0,
            "interrupted": interrupted,
            "outputs": {"video": str(out_video) if out_video else None,
                        "csv": str(out_csv) if out_csv else None},
        }
```

### Step 2. 결과 내보내기 통합

```python
    def export_results(self, responses, out_dir, sources=None,
                       save_json=True, save_csv=True, save_images=False,
                       images=None) -> dict:
        """여러 결과를 JSON · CSV · 이미지로 내보낸다.

        Args:
            responses: predict() 결과 리스트
            out_dir: 출력 폴더
            sources: 각 결과의 원본 경로 (파일명에 사용)
            save_json / save_csv / save_images: 내보낼 형식
            images: save_images=True 일 때 원본 이미지 리스트

        Returns:
            {"json": path, "csv": path, "images": [paths], "summary": {...}}
        """
        import csv as _csv
        import json as _json

        out = Path(out_dir)
        out.mkdir(parents=True, exist_ok=True)
        result = {"json": None, "csv": None, "images": [], "summary": {}}

        # ── 요약 ──
        total, counts, errors = 0, {}, 0
        for r in responses:
            if "error" in r:
                errors += 1
                continue
            total += r.get("count", 0)
            for k, v in r.get("counts_by_class", {}).items():
                counts[k] = counts.get(k, 0) + v
        summary = {
            "files": len(responses), "ok": len(responses) - errors,
            "errors": errors, "total_detections": total,
            "counts_by_class": counts,
            "avg_per_image": round(total / max(len(responses) - errors, 1), 2),
            "model": self.weights.name,
            "conf": self.conf, "iou": self.iou, "imgsz": self.imgsz,
        }
        result["summary"] = summary

        # ── JSON ──
        if save_json:
            p = out / "results.json"
            p.write_text(_json.dumps(
                {"summary": summary, "results": responses},
                indent=2, ensure_ascii=False), encoding="utf-8")
            result["json"] = str(p)

        # ── CSV ──
        if save_csv:
            p = out / "detections.csv"
            with p.open("w", newline="", encoding="utf-8-sig") as f:
                w = _csv.writer(f)
                w.writerow(["source", "class_id", "class", "conf",
                            "x1", "y1", "x2", "y2", "w", "h", "area_ratio"])
                for i, r in enumerate(responses):
                    if "error" in r:
                        continue
                    src = (str(sources[i]) if sources else
                           r.get("source", f"#{i}"))
                    for d in r["detections"]:
                        b, s = d["box"], d["size"]
                        w.writerow([Path(src).name, d["class_id"], d["class"],
                                    d["conf"], b["x1"], b["y1"], b["x2"],
                                    b["y2"], s["w"], s["h"], s["area_ratio"]])
            result["csv"] = str(p)

        # ── 이미지 ──
        if save_images and images is not None:
            img_dir = out / "images"
            img_dir.mkdir(exist_ok=True)
            for i, (img, r) in enumerate(zip(images, responses)):
                if "error" in r:
                    continue
                name = (Path(sources[i]).name if sources else f"{i:05d}.jpg")
                vis = self.draw(img, r)
                cv2.imwrite(str(img_dir / name), vis,
                            [cv2.IMWRITE_JPEG_QUALITY, 92])
                result["images"].append(str(img_dir / name))

        return result
```

```python
def main():
    """CLI 진입점.

    사용:
        python -m detector IMAGE_OR_DIR [옵션]
        python -m detector video.mp4 --video-out out.mp4
        python -m detector 0 --show                    # 웹캠
    """
    import argparse
    import json as _json

    ap = argparse.ArgumentParser(
        prog="detector",
        description="Object Detection 예측 모듈",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="""
예시:
  # 이미지 1장
  python -m detector photo.jpg --out results/

  # 폴더 전체 (결과 이미지 + JSON + CSV)
  python -m detector data/samples --out results/ --save-images

  # 동영상 (2프레임마다, 결과 영상 저장)
  python -m detector video.mp4 --out results/ --skip 2 --video-out det.mp4

  # 웹캠 실시간 보기
  python -m detector 0 --show

  # 임계값 변경 + 특정 클래스만
  python -m detector photo.jpg --conf 0.6 --classes 0
""")
    ap.add_argument("source", help="이미지·폴더·동영상 경로 또는 카메라 번호")
    ap.add_argument("--weights", default=None, help="가중치 경로")
    ap.add_argument("--profile", default=None, help="pc | cpu | pi | server")
    ap.add_argument("--conf", type=float, default=None)
    ap.add_argument("--iou", type=float, default=None)
    ap.add_argument("--imgsz", type=int, default=None)
    ap.add_argument("--classes", type=int, nargs="+", default=None)
    ap.add_argument("--out", default="results", help="출력 폴더")
    ap.add_argument("--save-images", action="store_true")
    ap.add_argument("--video-out", default=None)
    ap.add_argument("--skip", type=int, default=1)
    ap.add_argument("--max-frames", type=int, default=None)
    ap.add_argument("--show", action="store_true", help="화면 표시 (q 로 종료)")
    ap.add_argument("--no-json", action="store_true")
    ap.add_argument("--no-csv", action="store_true")
    ap.add_argument("-q", "--quiet", action="store_true")
    args = ap.parse_args()

    logging.basicConfig(
        level=logging.WARNING if args.quiet else logging.INFO,
        format="%(levelname)s %(message)s")

    kw = {k: v for k, v in (("conf", args.conf), ("iou", args.iou),
                            ("imgsz", args.imgsz)) if v is not None}
    det = get_detector(args.weights, profile=args.profile, **kw)

    out_dir = Path(args.out)
    out_dir.mkdir(parents=True, exist_ok=True)

    # ── 입력 종류 판별 ──
    src = args.source
    is_camera = src.isdigit()
    src_path = None if is_camera else Path(src)
    is_video = (not is_camera and src_path.is_file()
                and src_path.suffix.lower() in
                (".mp4", ".avi", ".mov", ".mkv", ".webm"))

    # ── 동영상 · 카메라 ──
    if is_camera or is_video:
        source = int(src) if is_camera else str(src_path)

        def on_frame(idx, frame, res):
            if not args.show:
                return True
            vis = det.draw(frame, res)
            cv2.putText(vis, det.summary(res), (12, 30),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
            cv2.imshow("detector (q=quit)", vis)
            return cv2.waitKey(1) & 0xFF != ord("q")

        stat = det.predict_video(
            source,
            out_video=args.video_out,
            out_csv=None if args.no_csv else str(out_dir / "detections.csv"),
            skip=args.skip, max_frames=args.max_frames,
            classes=args.classes, on_frame=on_frame if args.show else None)

        if args.show:
            cv2.destroyAllWindows()
        if not args.no_json:
            (out_dir / "summary.json").write_text(
                _json.dumps(stat, indent=2, ensure_ascii=False),
                encoding="utf-8")

        print("\n" + "=" * 64)
        print(f"  프레임      {stat['frames_read']} 읽음 / "
              f"{stat['frames_inferred']} 추론")
        print(f"  검출        {stat['total_detections']}건  "
              f"{stat['counts_by_class']}")
        print(f"  처리 속도    {stat['fps_processing']} FPS  "
              f"(실시간 대비 {stat['realtime_factor']}배)")
        print(f"  소요        {stat['duration_s']}초")
        print("=" * 64)
        return

    # ── 이미지 · 폴더 ──
    files = ([p for p in sorted(src_path.iterdir())
              if p.suffix.lower() in IMG_EXTS]
             if src_path.is_dir() else [src_path])
    if not files:
        print(f"처리할 이미지가 없습니다: {src_path}")
        return

    responses = det.predict_batch(files, classes=args.classes)

    images = None
    if args.save_images:
        images = [f for f in files]

    exported = det.export_results(
        responses, out_dir, sources=files,
        save_json=not args.no_json, save_csv=not args.no_csv,
        save_images=args.save_images, images=images)

    for f, r in zip(files, responses):
        if "error" in r:
            print(f"  ✗ {f.name:<30}{r['error']}")
        else:
            print(f"  {f.name:<30}{det.summary(r)}")

    s = exported["summary"]
    print("\n" + "=" * 64)
    print(f"  파일        {s['ok']}/{s['files']} 성공  (오류 {s['errors']})")
    print(f"  검출        {s['total_detections']}건  {s['counts_by_class']}")
    print(f"  평균        {s['avg_per_image']}건/장")
    for k in ("json", "csv"):
        if exported[k]:
            print(f"  {k.upper():<10}  {exported[k]}")
    if exported["images"]:
        print(f"  이미지      {len(exported['images'])}장 → {out_dir/'images'}")
    print("=" * 64)


if __name__ == "__main__":
    main()
```

| 증상 | 원인 · 해결 |
|---|---|
| `FileNotFoundError: 가중치 파일이 없습니다` | Releases 에서 모델을 받아 `models/` 에 |

| 웹캠이 안 열림 | 다른 프로그램 종료 · 카메라 번호를 1, 2 로 |
| 검출이 하나도 없음 | `--conf 0.1` 로 낮춰 확인. 그래도 없으면 모델·데이터 확인 |

```python
"""cross_test.py — 다른 팀 모듈 실전 검증

Day 152 의 인터페이스 검증에 ★ 실사용 시나리오를 추가
"""
import importlib.util
import json
import subprocess
import sys
import tempfile
import time
from pathlib import Path

import cv2
import numpy as np

SCENARIOS = []


def scenario(name, weight=1):
    def deco(fn):
        SCENARIOS.append((name, fn, weight))
        return fn
    return deco


@scenario("README 3줄 예제가 동작", weight=3)
def s_readme(mod, ctx):
    det = mod.Detector(ctx["weights"])
    res = det.predict(ctx["image"])
    assert "count" in res
    return f"검출 {res['count']}건"


@scenario("CLI 실행", weight=2)
def s_cli(mod, ctx):
    r = subprocess.run(
        [sys.executable, str(ctx["module_path"]), str(ctx["image"]),
         "--out", str(ctx["tmp"])],
        capture_output=True, text=True, timeout=120)
    assert r.returncode == 0, r.stderr[-300:]
    return "정상 종료"


@scenario("폴더 일괄 처리", weight=2)
def s_batch(mod, ctx):
    det = ctx["det"]
    files = sorted(ctx["sample_dir"].glob("*.jpg"))[:5]
    results = det.predict_batch(files)
    assert len(results) == len(files)
    return f"{len(files)}장 처리"


@scenario("깨진 파일이 섞여도 계속", weight=3)
def s_broken(mod, ctx):
    det = ctx["det"]
    bad = ctx["tmp"] / "bad.jpg"
    bad.write_text("not an image", encoding="utf-8")
    results = det.predict_batch([ctx["image"], bad, ctx["image"]])
    assert len(results) == 3
    assert "error" in results[1] and "error" not in results[2]
    return "격리 확인"


@scenario("★ 동영상 처리", weight=3)
def s_video(mod, ctx):
    det = ctx["det"]
    if not hasattr(det, "predict_video"):
        raise AttributeError("predict_video 미구현")
    stat = det.predict_video(str(ctx["video"]), max_frames=30, progress=False)
    assert stat["frames_read"] > 0
    return f"{stat['frames_inferred']}프레임 · {stat['fps_processing']} FPS"


@scenario("JSON 직렬화", weight=3)
def s_json(mod, ctx):
    res = ctx["det"].predict(ctx["image"])
    json.dumps(res)
    return "OK"


@scenario("★ draw 가 원본 미수정", weight=2)
def s_draw(mod, ctx):
    img = cv2.imread(str(ctx["image"]))
    before = img.copy()
    res = ctx["det"].predict(img)
    ctx["det"].draw(img, res)
    assert np.array_equal(img, before)
    return "OK"


@scenario("검출 0개 정상 처리", weight=2)
def s_empty(mod, ctx):
    blank = np.full((640, 640, 3), 128, np.uint8)
    res = ctx["det"].predict(blank, conf=0.99)
    assert res["count"] == 0 and "error" not in res
    return "OK"


@scenario("설정 파일 존재", weight=1)
def s_config(mod, ctx):
    root = ctx["module_path"].resolve().parents[1]
    cands = list(root.glob("configs/*.yaml"))
    assert cands, "configs/*.yaml 없음"
    return cands[0].name


@scenario("docstring 완비", weight=2)
def s_doc(mod, ctx):
    d = mod.Detector.predict.__doc__ or ""
    missing = [k for k in ("Args", "Returns") if k not in d]
    assert not missing, f"누락: {missing}"
    assert len(d) > 200, "너무 짧음"
    return f"{len(d)}자"


@scenario("테스트 통과", weight=3)
def s_pytest(mod, ctx):
    root = ctx["module_path"].resolve().parents[1]
    if not (root / "tests").exists():
        raise FileNotFoundError("tests/ 없음")
    r = subprocess.run([sys.executable, "-m", "pytest", str(root / "tests"),
                        "-q", "--no-header"],
                       capture_output=True, text=True, timeout=300)
    assert r.returncode == 0, r.stdout[-300:]
    return r.stdout.strip().splitlines()[-1][:50]


@scenario("★ 성능 (960 기준 300ms 이내)", weight=1)
def s_speed(mod, ctx):
    det = ctx["det"]
    img = cv2.imread(str(ctx["image"]))
    for _ in range(5):
        det.predict(img)
    t0 = time.perf_counter()
    for _ in range(10):
        det.predict(img)
    ms = (time.perf_counter() - t0) / 10 * 1000
    assert ms < 300, f"{ms:.0f}ms"
    return f"{ms:.0f}ms"


def run(module_path, weights, sample_dir, video):
    spec = importlib.util.spec_from_file_location("team_det", module_path)
    mod = importlib.util.module_from_spec(spec)
    sys.modules["team_det"] = mod
    spec.loader.exec_module(mod)

    tmp = Path(tempfile.mkdtemp())
    ctx = {
        "module_path": Path(module_path), "weights": weights,
        "sample_dir": Path(sample_dir),
        "image": sorted(Path(sample_dir).glob("*.jpg"))[0],
        "video": video, "tmp": tmp,
    }
    ctx["det"] = mod.Detector(weights)

    rows = []
    for name, fn, w in SCENARIOS:
        try:
            note = fn(mod, ctx)
            rows.append((name, w, True, note or ""))
        except Exception as e:
            rows.append((name, w, False, f"{type(e).__name__}: {e}"[:60]))
    return rows


TEAMS = {

    "5조": ("external/team5/src/detector.py", "external/team5/models/best.pt"),
}

print("=" * 100)
print("★ 교차 실행 테스트")
print("=" * 100)

for team, (path, w) in TEAMS.items():
    if not Path(path).exists():
        print(f"\n[{team}] 파일 없음")
        continue
    print(f"\n[{team}]")
    print("-" * 100)
    try:
        rows = run(path, w, "tests/data", "tests/data/short.mp4")
    except Exception as e:
        print(f"  ✗ 모듈 로드 실패: {e}")
        continue
    got = sum(w for _, w, ok, _ in rows if ok)
    tot = sum(w for _, w, _, _ in rows)
    for name, wgt, ok, note in rows:
        print(f"  {'✓' if ok else '✗'} [{wgt}] {name:<32}{note}")
    print(f"  → ★ {got}/{tot}점 ({got/tot*100:.0f}%)")

print("=" * 100)
```

```python
def predict_video(self, path, out_path):
    cap = cv2.VideoCapture(path)
    writer = cv2.VideoWriter(out_path, cv2.VideoWriter_fourcc(*"mp4v"),
                             30, (1280, 720))
    results = []
    while True:
        ret, frame = cap.read()
        if not ret:
            break
        res = self.predict(frame)
        results.append(res)
        writer.write(self.draw(frame, res))
    cap.release()
    writer.release()
    return results
```

```python
def predict_video(self, source, out_video=None, out_csv=None,
                  skip=1, max_frames=None, progress=True,
                  conf=None, iou=None) -> dict:
    """동영상 검출. 결과를 스트리밍 기록하고 요약만 반환한다."""
    import csv as _csv

    cap = cv2.VideoCapture(source)
    if not cap.isOpened():                                    # ★ 3
        raise RuntimeError(f"동영상을 열 수 없습니다: {source}")

    w = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))                # ★ 2
    h = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
    fps = cap.get(cv2.CAP_PROP_FPS) or 30.0
    total = int(cap.get(cv2.CAP_PROP_FRAME_COUNT)) or 0

    writer = csv_file = csv_writer = None
    try:
        if out_video:                                          # ★ 10
            Path(out_video).parent.mkdir(parents=True, exist_ok=True)  # ★ 8
            writer = cv2.VideoWriter(
                str(out_video), cv2.VideoWriter_fourcc(*"mp4v"),
                fps / max(skip, 1), (w, h))
            if not writer.isOpened():                          # ★ 12
                raise RuntimeError("VideoWriter 열기 실패 (코덱 확인)")

        if out_csv:
            Path(out_csv).parent.mkdir(parents=True, exist_ok=True)
            csv_file = open(out_csv, "w", newline="", encoding="utf-8-sig")
            csv_writer = _csv.writer(csv_file)
            csv_writer.writerow(["frame", "time_s", "class", "conf",
                                 "x1", "y1", "x2", "y2"])

        counts, n_read, n_inf, n_det, n_err = {}, 0, 0, 0, 0
        t0 = time.perf_counter()
        last_vis, interrupted = None, False

        while True:
            ok, frame = cap.read()
            if not ok:
                break
            n_read += 1

            if (n_read - 1) % max(skip, 1) == 0:               # ★ 5
                try:
                    res = self.predict(frame, conf=conf, iou=iou)
                except Exception as e:                         # ★ 9
                    n_err += 1
                    logger.warning("프레임 %d 실패: %s", n_read - 1, e)
                    continue
                n_inf += 1
                n_det += res["count"]
                for k, v in res["counts_by_class"].items():
                    counts[k] = counts.get(k, 0) + v

                if csv_writer:                                 # ★ 1 (스트리밍)
                    t_s = round((n_read - 1) / fps, 3)
                    for d in res["detections"]:
                        b = d["box"]
                        csv_writer.writerow([n_read - 1, t_s, d["class"],
                                             d["conf"], b["x1"], b["y1"],
                                             b["x2"], b["y2"]])
                if writer:
                    last_vis = self.draw(frame, res)

                if progress and n_inf % 20 == 0:               # ★ 6
                    el = time.perf_counter() - t0
                    pct = f" {n_read/total*100:5.1f}%" if total else ""
                    print(f"\r  {n_read}/{total or '?'}{pct}  "
                          f"검출 {n_det}  {n_inf/max(el,1e-6):.2f} FPS", end="")

            if writer and last_vis is not None:
                writer.write(last_vis)
            if max_frames and n_read >= max_frames:
                break

    except KeyboardInterrupt:                                  # ★ 7
        interrupted = True
        logger.warning("중단 — 지금까지의 결과를 저장합니다")
    finally:                                                   # ★ 4
        cap.release()
        if writer:
            writer.release()
        if csv_file:
            csv_file.close()
        if progress:
            print()

    dt = time.perf_counter() - t0
    return {                                                   # ★ 11
        "video": {"width": w, "height": h, "fps": round(fps, 2),
                  "total_frames": total},
        "frames_read": n_read, "frames_inferred": n_inf,
        "errors": n_err, "total_detections": n_det,
        "counts_by_class": counts,
        "duration_s": round(dt, 2),
        "fps_processing": round(n_inf / dt, 2) if dt else 0,
        "realtime_factor": round((n_read / fps) / dt, 2) if dt else 0,
        "interrupted": interrupted,
    }
```
