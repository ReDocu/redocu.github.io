### 3-1. 웨이브 데이터 설계

**왜 필요한가** — 밸런싱이 게임의 절반이다.

| 웨이브 | 적 종류 | 수 | 간격(초) | 체력 배율 |
|:--:|---|:--:|:--:|:--:|
| 1 | 기본 | 5 | 1.0 | 1.0 |
| 2 | 기본 | 8 | 0.8 | 1.2 |
| 3 | 기본 + 빠름 | 6 + 3 | 0.8 | 1.3 |
| 4 | 기본 + 탱커 | 8 + 2 | 0.7 | 1.5 |
| 5 | 보스 | 1 | — | 10.0 |

```csharp
using System.Collections.Generic;
using UnityEngine;

[System.Serializable]
public class SpawnGroup
{
    public EnemyData enemyData;
    public int count = 5;
    public float interval = 1f;                // 개체 간 간격
    public float delayBeforeGroup = 0f;        // 그룹 시작 전 대기
}

[CreateAssetMenu(fileName = "Wave_", menuName = "Defense/Wave Data")]
public class WaveData : ScriptableObject
{
    [Header("웨이브 정보")]
    public int waveNumber = 1;
    public string label = "";

    [Header("구성")]
    public List<SpawnGroup> groups = new List<SpawnGroup>();

    [Header("배율")]
    [Range(0.5f, 30f)] public float hpMultiplier = 1f;
    [Range(0.5f, 3f)]  public float speedMultiplier = 1f;

    [Header("보상")]
    public int clearGold = 50;

    [Header("다음 웨이브")]
    public float restTime = 8f;

    // ─── 계산 ───

    public int TotalCount
    {
        get
        {
            int n = 0;
            foreach (SpawnGroup g in groups) n += g.count;
            return n;
        }
    }

    public float TotalHp
    {
        get
        {
            float sum = 0;
            foreach (SpawnGroup g in groups)
            {
                if (g.enemyData == null) continue;
                sum += g.enemyData.maxHp * hpMultiplier * g.count;
            }
            return sum;
        }
    }

    public float EstimatedDuration
    {
        get
        {
            float t = 0;
            foreach (SpawnGroup g in groups)
                t += g.delayBeforeGroup + g.interval * Mathf.Max(0, g.count - 1);
            return t;
        }
    }

    void OnValidate()
    {
        if (string.IsNullOrEmpty(label)) label = $"Wave {waveNumber}";
    }
}
```

```
   → "이 웨이브를 막으려면 DPS가 얼마나 필요한가"
```

### 3-3. 경로 이동 — 3가지 방식

**왜 필요한가** — 디펜스 게임의 이동은 액션 게임과 다르다.

| 방식 | 설명 | 장단점 |
|---|---|---|
| **웨이포인트** | 정해진 점들을 순서대로 | 단순, 예측 가능 / 우회 불가 |
| **NavMesh** | 목표까지 경로 탐색 | 우회 가능 / 비용 |
| **스플라인** | 곡선 경로 | 부드러움 / 설정 필요 |

```
   웨이포인트                 NavMesh

   ●──▶●──▶●──▶●             ●
   1   2   3   목표            ╲___
                                   ╲──▶ 목표
   정확히 그 길로만            상황에 따라 다른 길
```

```
   ★ 어느 것을 쓰나

   고전 타워 디펜스 (경로 고정)  →  웨이포인트
   미로형 (길을 막을 수 있음)     →  NavMesh  ★ 오늘
   레이싱 AI                      →  스플라인
```

```
   ★ 이 커리큘럼은 NavMesh를 쓴다

   Day 90에서 "길을 막으면 우회"가 핵심 메커니즘이 되기 때문
```

### 3-4. 비행 유닛

**왜 필요한가** — NavMesh를 무시하고 직선으로.

```csharp
public class FlyingMovement : MonoBehaviour, IEnemyMovement
{
    [SerializeField] private float height = 4f;
    [SerializeField] private float bobAmount = 0.3f;
    [SerializeField] private float bobSpeed = 2f;

    private Transform goal;
    private float speed;
    private float bobOffset;

    public void Initialize(Transform goalTransform, float moveSpeed)
    {
        goal = goalTransform;
        speed = moveSpeed;
        bobOffset = Random.value * Mathf.PI * 2f;   // ★ 위상 랜덤
    }

    public void Tick(float dt)
    {
        if (goal == null) return;

        Vector3 target = goal.position + Vector3.up * height;
        Vector3 dir = target - transform.position;
        dir.y = 0;

        if (dir.sqrMagnitude > 0.01f)
        {
            Vector3 move = dir.normalized * speed * dt;
            transform.position += move;

            transform.rotation = Quaternion.Slerp(
                transform.rotation, Quaternion.LookRotation(dir), 6f * dt);
        }

        // ★ 상하 흔들림
        float y = height + Mathf.Sin(Time.time * bobSpeed + bobOffset) * bobAmount;
        transform.position = new Vector3(
            transform.position.x, goal.position.y + y, transform.position.z);
    }

    public bool HasReachedGoal =>
        goal != null &&
        new Vector2(transform.position.x - goal.position.x,
                    transform.position.z - goal.position.z).sqrMagnitude < 1f;
}
```

```
   ⚠️ 비행 유닛에 NavMeshAgent가 붙어 있으면

   → 공중에 못 뜬다

   ★ 프리팹에서 Agent를 아예 빼거나
```


### 3-5. 지상 이동 (NavMesh)

**왜 필요한가** — Day 84의 재사용.

```csharp
using UnityEngine;
using UnityEngine.AI;

[RequireComponent(typeof(NavMeshAgent))]
public class GroundMovement : MonoBehaviour, IEnemyMovement
{
    private NavMeshAgent agent;
    private Transform goal;
    private float baseSpeed;

    void Awake() { agent = GetComponent<NavMeshAgent>(); }

    public void Initialize(Transform goalTransform, float speed)
    {
        goal = goalTransform;
        baseSpeed = speed;

        // ★ NavMesh 위로 보정 (Day 84)
        if (NavMesh.SamplePosition(transform.position, out NavMeshHit hit,
                                   5f, NavMesh.AllAreas))
        {
            agent.Warp(hit.position);
        }
        else
        {
            Debug.LogError($"[적] NavMesh 밖에서 스폰: {transform.position}");
        }

        agent.enabled = true;
        agent.speed = speed;
        agent.isStopped = false;
        agent.SetDestination(goal.position);
    }

    public void Tick(float dt)
    {
        if (goal == null || !agent.enabled) return;

        // ★ 경로가 무효화됐으면 재계산 (Day 90의 Carve 대비)
        if (agent.pathStatus == NavMeshPathStatus.PathInvalid ||
            (!agent.hasPath && !agent.pathPending))
        {
            agent.SetDestination(goal.position);
        }
    }

    public bool HasReachedGoal
    {
        get
        {
            if (!agent.enabled || agent.pathPending) return false;
            return agent.remainingDistance <= 1.2f;
        }
    }

    public void Stop()
    {
        if (agent.enabled) { agent.isStopped = true; agent.ResetPath(); }
    }

    public void SetSpeedMultiplier(float mult)
    {
        if (agent.enabled) agent.speed = baseSpeed * mult;
    }
}
```

### 3-6. 매니저 패턴 — Update 줄이기

**왜 필요한가** — 적 200기가 각자 Update를 돌면 느리다.

```
   ★ 각 오브젝트가 Update를 가지면

   Unity가 오브젝트마다 Update를 호출한다
   → 호출 오버헤드가 개수에 비례
   → 200개면 프레임당 200번의 네이티브→C# 전환
```

```csharp
public class EnemyManager : MonoBehaviour
{
    public static EnemyManager Instance { get; private set; }

    private readonly List<Enemy> active = new List<Enemy>();

    void Awake() { Instance = this; }

    public void Register(Enemy e)   { active.Add(e); }
    public void Unregister(Enemy e) { active.Remove(e); }

    void Update()                              // ★ Update 하나
    {
        float dt = Time.deltaTime;

        // ★ 역순 (Day 53) — 순회 중 제거 대비
        for (int i = active.Count - 1; i >= 0; i--)
        {
            Enemy e = active[i];

            if (e == null) { active.RemoveAt(i); continue; }

            e.Tick(dt);
        }
    }

    public int ActiveCount => active.Count;
    public IReadOnlyList<Enemy> Active => active;
}
```

| 방식 | 적 200기 프레임 시간 |
|---|:--:|
| 각자 Update | 약 1.8 ms |
| **매니저 일괄** | **약 0.6 ms** |


```csharp
    private IEnumerator SpawnWaveRoutine(WaveData wave)
    {
        isSpawning = true;

        foreach (SpawnGroup group in wave.groups)
        {
            if (group.enemyData == null) continue;

            if (group.delayBeforeGroup > 0)
                yield return new WaitForSeconds(group.delayBeforeGroup);

            for (int i = 0; i < group.count; i++)
            {
                SpawnEnemy(group.enemyData, wave);

                if (i < group.count - 1)
                    yield return new WaitForSeconds(group.interval);
            }
        }

        isSpawning = false;
        Debug.Log($"[웨이브 {wave.waveNumber}] 스폰 완료 ({wave.TotalCount}기)");
    }
```

### 3-10. 스폰 위치 보정

**왜 필요한가** — Day 84에서 배운 그 문제.

```csharp
    private Vector3 GetSpawnPosition(EnemyData data)
    {
        Vector3 basePos = spawnPoint.position;

        // ★ 겹침 방지 — 약간의 랜덤 오프셋
        Vector2 rand = Random.insideUnitCircle * spawnRadius;
        Vector3 pos = basePos + new Vector3(rand.x, 0, rand.y);

        if (data.moveType == EnemyMoveType.Flying)
            return pos + Vector3.up * 4f;

        // ★ 지상 유닛은 NavMesh 위로
        if (NavMesh.SamplePosition(pos, out NavMeshHit hit, 3f, NavMesh.AllAreas))
            return hit.position;

        Debug.LogWarning("[스폰] NavMesh를 못 찾음. 기본 위치 사용");
        return basePos;
    }
```

```csharp
using UnityEngine;

public class Enemy : MonoBehaviour, IDamageable, IPoolable
{
    [SerializeField] private HpBarWorld hpBar;
    [SerializeField] private HitFlash hitFlash;

    private IEnemyMovement movement;
    private EnemyData data;
    private PooledObject pooled;

    private int hp;
    private float hpMultiplier = 1f;
    private float slowMultiplier = 1f;
    private float slowTimer;
    private float initialDistance;

    public EnemyData Data => data;
    public bool IsAlive => hp > 0;
    public Vector3 Position => transform.position;

    public float Progress
    {
        get
        {
            if (initialDistance <= 0.01f) return 1f;
            float remain = Vector3.Distance(transform.position, GoalPosition);
            return Mathf.Clamp01(1f - remain / initialDistance);
        }
    }

    private Vector3 GoalPosition => WaveManager.Instance.Goal.position;

    void Awake()
    {
        movement = GetComponent<IEnemyMovement>();
        pooled = GetComponent<PooledObject>();
    }

    public void Setup(EnemyData d, float hpMult, float speedMult)
    {
        data = d;
        hpMultiplier = hpMult;

        hp = Mathf.RoundToInt(d.maxHp * hpMult);

        Transform goal = WaveManager.Instance.Goal;
        initialDistance = Vector3.Distance(transform.position, goal.position);

        movement.Initialize(goal, d.moveSpeed * speedMult);

        hpBar?.Set(hp, hp);

        EnemyManager.Instance.Register(this);
    }

    // ★ 매니저가 호출 (3-6)
    public void Tick(float dt)
    {
        if (!IsAlive) return;

        // 감속 지속시간
        if (slowTimer > 0)
        {
            slowTimer -= dt;
            if (slowTimer <= 0)
            {
                slowMultiplier = 1f;
                movement.SetSpeedMultiplier(1f);
            }
        }

        movement.Tick(dt);

        if (movement.HasReachedGoal)
            WaveManager.Instance.OnEnemyReachedGoal(this);
    }

    // ─── 피격 ───

    public void TakeDamage(DamageInfo info)
    {
        if (!IsAlive) return;

        int actual = data.CalcDamage(info.amount);   // ★ 방어력 적용
        hp -= actual;

        hitFlash?.Flash();
        hpBar?.Set(Mathf.Max(0, hp), Mathf.RoundToInt(data.maxHp * hpMultiplier));

        if (hp <= 0) Die();
    }

    public void ApplySlow(float multiplier, float duration, float resist)
    {
        float effective = Mathf.Lerp(multiplier, 1f, resist);

        // ★ 더 강한 감속만 적용
        if (effective < slowMultiplier || slowTimer <= 0)
        {
            slowMultiplier = effective;
            movement.SetSpeedMultiplier(effective);
        }

        slowTimer = Mathf.Max(slowTimer, duration);
    }

    private void Die()
    {
        WaveManager.Instance.OnEnemyKilled(this);
        Despawn();
    }

    public void Despawn()
    {
        movement.Stop();
        EnemyManager.Instance.Unregister(this);

        if (pooled != null) pooled.ReturnToPool();
        else                Destroy(gameObject);
    }

    // ─── 풀 (Day 74) ───

    public void OnSpawn()
    {
        slowMultiplier = 1f;
        slowTimer = 0f;
        hp = 1;                                // Setup에서 덮어씀
    }

    public void OnDespawn()
    {
        // 정리
    }
}
```

### Step 9 — WaveManager

```csharp
using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.AI;

public enum WaveState { Ready, Spawning, Fighting, Resting, Clear, GameOver }

public class WaveManager : MonoBehaviour
{
    public static WaveManager Instance { get; private set; }

    [Header("참조")]
    [SerializeField] private Transform spawnPoint;
    [SerializeField] private Transform goal;
    [SerializeField] private float spawnRadius = 1.5f;

    [Header("웨이브")]
    [SerializeField] private List<WaveData> waves = new List<WaveData>();

    [Header("게임")]
    [SerializeField] private int maxLife = 20;
    [SerializeField] private int startGold = 200;

    public Transform Goal => goal;
    public WaveState State { get; private set; } = WaveState.Ready;
    public int CurrentWaveNumber => currentIndex + 1;
    public int TotalWaves => waves.Count;
    public int Life { get; private set; }
    public int Gold { get; private set; }
    public float RestRemaining { get; private set; }

    public event System.Action<int, int> OnLifeChanged;
    public event System.Action<int> OnGoldChanged;
    public event System.Action<WaveState> OnStateChanged;

    private int currentIndex = -1;
    private int aliveCount;
    private bool isSpawning;
    private Coroutine spawnRoutine;
    private Coroutine restRoutine;

    void Awake()
    {
        Instance = this;
        Life = maxLife;
        Gold = startGold;
    }

    void Start()
    {
        SetState(WaveState.Ready);
    }

    // ─── 웨이브 진행 ───

    public void StartNextWave()
    {
        if (isSpawning) { Debug.LogWarning("[웨이브] 이미 스폰 중"); return; }
        if (State == WaveState.Clear || State == WaveState.GameOver) return;

        if (restRoutine != null) { StopCoroutine(restRoutine); restRoutine = null; }

        currentIndex++;

        if (currentIndex >= waves.Count) { SetState(WaveState.Clear); return; }

        WaveData wave = waves[currentIndex];

        Debug.Log($"[웨이브 {wave.waveNumber}] 시작 — " +
                  $"{wave.TotalCount}기, 총 HP {wave.TotalHp:F0}, " +
                  $"예상 {wave.EstimatedDuration:F1}초");

        SetState(WaveState.Spawning);
        spawnRoutine = StartCoroutine(SpawnWaveRoutine(wave));
    }

    private IEnumerator SpawnWaveRoutine(WaveData wave)
    {
        isSpawning = true;

        foreach (SpawnGroup group in wave.groups)
        {
            if (group.enemyData == null) continue;

            if (group.delayBeforeGroup > 0)
                yield return new WaitForSeconds(group.delayBeforeGroup);

            for (int i = 0; i < group.count; i++)
            {
                SpawnEnemy(group.enemyData, wave);

                if (i < group.count - 1)
                    yield return new WaitForSeconds(group.interval);
            }
        }

        isSpawning = false;
        spawnRoutine = null;

        SetState(WaveState.Fighting);

        Debug.Log($"[웨이브 {wave.waveNumber}] 스폰 완료");
    }

    private void SpawnEnemy(EnemyData data, WaveData wave)
    {
        Vector3 pos = GetSpawnPosition(data);

        GameObject go = PoolManager.Instance.Get(data.poolKey, pos, Quaternion.identity);

        if (go == null)
        {
            Debug.LogWarning($"[스폰] 풀 고갈: {data.poolKey}");
            return;
        }

        Enemy e = go.GetComponent<Enemy>();
        e.Setup(data, wave.hpMultiplier, wave.speedMultiplier);

        aliveCount++;
    }

    private Vector3 GetSpawnPosition(EnemyData data)
    {
        Vector2 rand = Random.insideUnitCircle * spawnRadius;
        Vector3 pos = spawnPoint.position + new Vector3(rand.x, 0, rand.y);

        if (data.moveType == EnemyMoveType.Flying)
            return pos + Vector3.up * 4f;

        if (NavMesh.SamplePosition(pos, out NavMeshHit hit, 3f, NavMesh.AllAreas))
            return hit.position;

        Debug.LogWarning("[스폰] NavMesh 못 찾음");
        return spawnPoint.position;
    }

    // ─── 이벤트 ───

    public void OnEnemyKilled(Enemy e)
    {
        AddGold(e.Data.goldReward);
        aliveCount--;
        CheckWaveEnd();
    }

    public void OnEnemyReachedGoal(Enemy e)
    {
        Life -= e.Data.lifeDamage;
        OnLifeChanged?.Invoke(Life, maxLife);

        Debug.Log($"[라이프] -{e.Data.lifeDamage} → {Life}");
        CameraShaker.Instance?.Shake(0.5f);

        e.Despawn();
        aliveCount--;

        if (Life <= 0) { SetState(WaveState.GameOver); return; }

        CheckWaveEnd();
    }

    private void CheckWaveEnd()
    {
        if (isSpawning) return;
        if (aliveCount > 0) return;
        if (State == WaveState.GameOver || State == WaveState.Clear) return;

        // ★ 웨이브 클리어
        WaveData wave = waves[currentIndex];
        AddGold(wave.clearGold);

        Debug.Log($"[웨이브 {wave.waveNumber}] 클리어! 보너스 +{wave.clearGold}");

        if (currentIndex + 1 >= waves.Count) { SetState(WaveState.Clear); return; }

        SetState(WaveState.Resting);
        restRoutine = StartCoroutine(RestRoutine(wave.restTime));
    }

    private IEnumerator RestRoutine(float duration)
    {
        RestRemaining = duration;

        while (RestRemaining > 0)
        {
            RestRemaining -= Time.deltaTime;
            yield return null;
        }

        restRoutine = null;
        StartNextWave();
    }

    // ─── 자원 ───

    public void AddGold(int amount)
    {
        Gold += amount;
        OnGoldChanged?.Invoke(Gold);
    }

    public bool TrySpendGold(int amount)
    {
        if (Gold < amount) return false;

        Gold -= amount;
        OnGoldChanged?.Invoke(Gold);
        return true;
    }

    // ─── 상태 ───

    private void SetState(WaveState s)
    {
        if (State == s) return;

        State = s;
        OnStateChanged?.Invoke(s);

        if (s == WaveState.GameOver) Debug.Log("[게임] 패배");
        if (s == WaveState.Clear)    Debug.Log("[게임] 승리!");
    }

    public int AliveCount => aliveCount;
}
```

## 6. 자주 막히는 곳

| 증상 | 원인 | 해결 |
|---|---|---|
| **적이 안 움직임** | 스폰 위치가 NavMesh 밖 | `SamplePosition` + `Warp` |
| 적이 안 움직임 | 목표 미지정 | `Goal` 연결 |
| **웨이브가 겹침** | 코루틴 중복 | `isSpawning` 플래그 |
| 비행 유닛이 땅에 붙음 | NavMeshAgent 존재 | 컴포넌트 제거 |
| 적이 겹쳐 스폰 | 같은 위치 | `insideUnitCircle` 오프셋 |
| 풀 고갈 | `maxSize` 부족 | 늘리기 |
| 재사용 시 감속이 남음 | `OnSpawn` 초기화 누락 | 리셋 |
| `aliveCount` 꼬임 | 증감 누락 | 스폰/처치/도달 모두 처리 |
| 웨이브가 안 끝남 | `isSpawning` 중 검사 | `CheckWaveEnd`에서 확인 |
| 후반에 프레임 저하 | 각자 Update | 매니저 패턴 |
| HP 바가 항상 켜짐 | 숨김 처리 없음 | `hideDelay` |
| HP 바가 뒤집힘 | 빌보드 없음 | `forward = cam.forward` |
| 이벤트 예외 | 구독 해제 누락 | `OnDestroy`에서 `-=` |
