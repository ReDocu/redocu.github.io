```csharp
using UnityEngine;

namespace OneFloor.Data
{
    [System.Serializable]
    public class EnemyGroup
    {
        public EnemyData enemyData;
        public int count = 3;
        public float spawnDelay = 0f;
    }

    [CreateAssetMenu(fileName = "Floor_", menuName = "OneFloor/Floor Data")]
    public class FloorData : ScriptableObject
    {
        [Header("기본")]
        public int floorNumber = 1;
        public string label = "";

        [Header("적 구성")]
        public List<EnemyGroup> groups = new List<EnemyGroup>();

        [Header("배율")]
        [Range(0.5f, 20f)] public float hpMultiplier = 1f;
        [Range(0.5f, 3f)]  public float speedMultiplier = 1f;
        [Range(0.5f, 5f)]  public float damageMultiplier = 1f;

        [Header("맵")]
        public GameObject roomPrefab;
        public bool isBossFloor;

        [Header("보상")]
        public int goldReward = 30;
        public int weaponChoices = 3;

        // ─── 계산 ───

        public int TotalEnemyCount
        {
            get
            {
                int n = 0;
                foreach (EnemyGroup g in groups) n += g.count;
                return n;
            }
        }

        public float TotalHp
        {
            get
            {
                float sum = 0;
                foreach (EnemyGroup g in groups)
                {
                    if (g.enemyData == null) continue;
                    sum += g.enemyData.maxHp * hpMultiplier * g.count;
                }
                return sum;
            }
        }

        private void OnValidate()
        {
            if (string.IsNullOrEmpty(label)) label = $"{floorNumber}층";
        }
    }
}
```

```csharp
namespace OneFloor.Data
{
    [CreateAssetMenu(fileName = "GameDatabase", menuName = "OneFloor/Game Database")]
    public class GameDatabase : ScriptableObject
    {
        [Header("층")]
        public List<FloorData> floors = new List<FloorData>();

        [Header("적")]
        public List<EnemyData> enemies = new List<EnemyData>();

        [Header("무기")]
        public List<WeaponData> weapons = new List<WeaponData>();

        private Dictionary<string, EnemyData> enemyLookup;

        public int FloorCount => floors.Count;

        public FloorData GetFloor(int index)
        {
            if (index < 0 || index >= floors.Count) return null;
            return floors[index];
        }

        public EnemyData GetEnemy(string id)
        {
            if (enemyLookup == null) BuildLookup();
            return enemyLookup.TryGetValue(id, out EnemyData e) ? e : null;
        }

        public void BuildLookup()
        {
            enemyLookup = new Dictionary<string, EnemyData>();

            foreach (EnemyData e in enemies)
            {
                if (e == null || string.IsNullOrEmpty(e.id)) continue;

                if (enemyLookup.ContainsKey(e.id))
                {
                    Debug.LogError($"[DB] ID 중복: {e.id}", e);
                    continue;
                }

                enemyLookup[e.id] = e;
            }
        }

#if UNITY_EDITOR
        [ContextMenu("자동 수집")]
        private void CollectAll()
        {
            floors = CollectAssets<FloorData>("Assets/_Project/Data/Floors");
            enemies = CollectAssets<EnemyData>("Assets/_Project/Data/Enemies");
            weapons = CollectAssets<WeaponData>("Assets/_Project/Data/Weapons");

            floors.Sort((a, b) => a.floorNumber.CompareTo(b.floorNumber));

            UnityEditor.EditorUtility.SetDirty(this);
            Debug.Log($"[DB] 층 {floors.Count}, 적 {enemies.Count}, 무기 {weapons.Count}");
        }

        private static List<T> CollectAssets<T>(string folder) where T : Object
        {
            var result = new List<T>();
            string[] guids = UnityEditor.AssetDatabase.FindAssets(
                $"t:{typeof(T).Name}", new[] { folder });

            foreach (string guid in guids)
            {
                string path = UnityEditor.AssetDatabase.GUIDToAssetPath(guid);
                T asset = UnityEditor.AssetDatabase.LoadAssetAtPath<T>(path);
                if (asset != null) result.Add(asset);
            }

            return result;
        }
#endif
    }
}
```

### 3-5. 층 진행 시스템

**왜 필요한가** — 데이터를 실제 게임으로.

```csharp
namespace OneFloor.Game
{
    public class FloorManager : MonoBehaviour
    {
        [SerializeField] private GameDatabase database;
        [SerializeField] private Transform spawnCenter;
        [SerializeField] private float spawnRadius = 6f;

        private readonly List<Enemy> activeEnemies = new List<Enemy>();
        private FloorData currentFloor;

        public int CurrentFloorIndex { get; private set; }
        public FloorData CurrentFloorData => currentFloor;
        public int AliveCount { get; private set; }

        public event System.Action<FloorData> OnFloorStarted;
        public event System.Action<FloorData> OnFloorCleared;
        public event System.Action OnAllFloorsCleared;

        public void StartFloor(int index)
        {
            CurrentFloorIndex = index;
            currentFloor = database.GetFloor(index);

            if (currentFloor == null)
            {
                OnAllFloorsCleared?.Invoke();
                return;
            }

            ClearEnemies();
            StartCoroutine(SpawnRoutine(currentFloor));

            OnFloorStarted?.Invoke(currentFloor);

            DebugUtil.Log($"[층 {currentFloor.floorNumber}] 시작 — " +
                          $"적 {currentFloor.TotalEnemyCount}기, " +
                          $"총 HP {currentFloor.TotalHp:F0}");
        }

        private IEnumerator SpawnRoutine(FloorData floor)
        {
            int index = 0;
            int total = floor.TotalEnemyCount;

            AliveCount = total;

            foreach (EnemyGroup group in floor.groups)
            {
                if (group.enemyData == null) continue;

                if (group.spawnDelay > 0)
                    yield return new WaitForSeconds(group.spawnDelay);

                for (int i = 0; i < group.count; i++)
                {
                    SpawnEnemy(group.enemyData, floor, index, total);
                    index++;
                }
            }
        }

        private void SpawnEnemy(EnemyData data, FloorData floor, int index, int total)
        {
            Vector3 pos = GetSpawnPosition(index, total);

            GameObject go = PoolManager.Instance.Get(data.poolKey, pos, Quaternion.identity);

            if (go == null)
            {
                DebugUtil.LogWarning($"[스폰] 풀 고갈: {data.poolKey}");
                AliveCount--;
                return;
            }

            Enemy e = go.GetComponent<Enemy>();

            e.Setup(data, floor.hpMultiplier, floor.speedMultiplier, floor.damageMultiplier);
            e.OnDied += HandleEnemyDied;

            activeEnemies.Add(e);
        }

        private Vector3 GetSpawnPosition(int index, int total)
        {
            float angle = (index / (float)Mathf.Max(1, total)) * Mathf.PI * 2f;

            Vector3 offset = new Vector3(
                Mathf.Cos(angle) * spawnRadius, 0, Mathf.Sin(angle) * spawnRadius);

            Vector3 pos = spawnCenter.position + offset;

            // ★ NavMesh 위로 보정 (Day 84)
            if (NavMesh.SamplePosition(pos, out NavMeshHit hit, 4f, NavMesh.AllAreas))
                return hit.position;

            return spawnCenter.position;
        }

        private void HandleEnemyDied(Enemy e)
        {
            e.OnDied -= HandleEnemyDied;
            activeEnemies.Remove(e);

            AliveCount--;

            if (AliveCount <= 0) ClearFloor();
        }

        private void ClearFloor()
        {
            DebugUtil.Log($"[층 {currentFloor.floorNumber}] 클리어");

            OnFloorCleared?.Invoke(currentFloor);
        }

        public void ClearEnemies()
        {
            for (int i = activeEnemies.Count - 1; i >= 0; i--)
            {
                if (activeEnemies[i] == null) continue;

                activeEnemies[i].OnDied -= HandleEnemyDied;
                activeEnemies[i].Despawn();
            }

            activeEnemies.Clear();
            AliveCount = 0;
        }

        // ★ 치트
        public void SkipFloor()
        {
            ClearEnemies();
            ClearFloor();
        }
    }
}
```

### 3-8. 적 AI 최소 구현

**왜 필요한가** — Day 84의 NavMesh 재사용.

```csharp
namespace OneFloor.AI
{
    public enum EnemyState { Idle, Chase, Attack, Hit, Dead }

    [RequireComponent(typeof(NavMeshAgent))]
    public class Enemy : MonoBehaviour, IPoolable
    {
        [SerializeField] private Health health;
        [SerializeField] private HitFlash hitFlash;

        private NavMeshAgent agent;
        private EnemyData data;
        private Transform target;
        private EnemyState state;

        private float hpMult = 1f, speedMult = 1f, damageMult = 1f;
        private float attackCooldown;

        public event System.Action<Enemy> OnDied;

        public bool IsAlive => health.IsAlive;

        private void Awake()
        {
            agent = GetComponent<NavMeshAgent>();
        }

        public void Setup(EnemyData d, float hpM, float speedM, float dmgM)
        {
            data = d;
            hpMult = hpM; speedMult = speedM; damageMult = dmgM;

            health.SetMaxHp(Mathf.RoundToInt(d.maxHp * hpM), true);

            agent.speed = d.moveSpeed * speedM;
            agent.stoppingDistance = d.attackRange * 0.8f;

            // ★ NavMesh 위로 보정 (Day 84)
            if (NavMesh.SamplePosition(transform.position, out NavMeshHit hit,
                                       4f, NavMesh.AllAreas))
                agent.Warp(hit.position);

            target = GameManager.Instance.PlayerTransform;
            state = EnemyState.Chase;

            health.OnDied += HandleDied;
        }

        // ★ 매니저가 호출 (Day 86)
        public void Tick(float dt)
        {
            if (state == EnemyState.Dead) return;
            if (!GameManager.Instance.IsPlaying) return;

            attackCooldown -= dt;

            float dist = Vector3.Distance(transform.position, target.position);

            switch (state)
            {
            case EnemyState.Chase:
                agent.isStopped = false;
                agent.SetDestination(target.position);

                if (dist <= data.attackRange) state = EnemyState.Attack;
                break;

            case EnemyState.Attack:
                agent.isStopped = true;

                LookAtTarget(dt);

                if (dist > data.attackRange * 1.3f) { state = EnemyState.Chase; break; }

                if (attackCooldown <= 0)
                {
                    Attack();
                    attackCooldown = data.attackCooldown;
                }
                break;
            }
        }

        private void Attack()
        {
            IDamageable t = target.GetComponentInParent<IDamageable>();
            if (t == null || !t.IsAlive) return;

            Vector3 dir = (target.position - transform.position).normalized;

            t.TakeDamage(new DamageInfo
            {
                amount = Mathf.RoundToInt(data.attackDamage * damageMult),
                knockback = 3f,
                direction = dir,
                source = transform,
            });
        }

        private void HandleDied()
        {
            state = EnemyState.Dead;
            agent.enabled = false;

            health.OnDied -= HandleDied;

            OnDied?.Invoke(this);

            StartCoroutine(DespawnRoutine());
        }

        // IPoolable (Day 74)
        public void OnSpawn()
        {
            state = EnemyState.Idle;
            attackCooldown = 0;
            agent.enabled = true;
        }

        public void OnDespawn() { }
    }
}
```

```csharp
namespace OneFloor.AI
{
    public class EnemyManager : MonoBehaviour
    {
        public static EnemyManager Instance { get; private set; }

        private readonly List<Enemy> active = new List<Enemy>();

        private void Awake() { Instance = this; }

        public void Register(Enemy e)   { if (!active.Contains(e)) active.Add(e); }
        public void Unregister(Enemy e) { active.Remove(e); }

        private void Update()
        {
            float dt = Time.deltaTime;

            // ★ 역순 (Day 53)
            for (int i = active.Count - 1; i >= 0; i--)
            {
                if (active[i] == null) { active.RemoveAt(i); continue; }
                active[i].Tick(dt);
            }
        }

        public IReadOnlyList<Enemy> Active => active;
        public int Count => active.Count;
    }
}
```

```csharp
#if UNITY_EDITOR
    [ContextMenu("데이터 검증")]
    private void Validate()
    {
        int errors = 0;

        for (int i = 0; i < floors.Count; i++)
        {
            FloorData f = floors[i];

            if (f == null) { Debug.LogError($"[{i}] 층 데이터 null"); errors++; continue; }

            if (f.groups.Count == 0)
            {
                Debug.LogError($"[{f.name}] 적 구성이 비어 있다", f);
                errors++;
            }

            foreach (EnemyGroup g in f.groups)
            {
                if (g.enemyData == null)
                {
                    Debug.LogError($"[{f.name}] 적 데이터 null", f);
                    errors++;
                }
                else if (g.enemyData.prefab == null)
                {
                    Debug.LogError($"[{g.enemyData.name}] 프리팹 null", g.enemyData);
                    errors++;
                }
            }

            if (f.floorNumber != i + 1)
                Debug.LogWarning($"[{f.name}] 층 번호 불일치 ({f.floorNumber} vs {i+1})", f);
        }

        Debug.Log(errors == 0 ? "[검증] 통과" : $"[검증] 오류 {errors}건");
    }
#endif
```
