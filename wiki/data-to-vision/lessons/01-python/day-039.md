## ① 학습 목표

이 레슨을 마치면 이렇게 할 수 있습니다.

- 랭킹 DB 를 연동해 게임을 완성한다.
- 예외 상황에서도 프로그램이 죽지 않게 방어 코드를 넣는다.
- 다른 사람의 PC에서 실행되도록 배포 준비를 마친다.

---

## ② 왜 필요한가

**"제 컴퓨터에서는 됐는데요."**

발표 당일 가장 많이 나오는 말이고, 가장 변명이 안 되는 말입니다.

| 내 PC에서만 되는 이유 | 해결 |
|---|---|
| 절대경로 하드코딩 | `Path(__file__)` 기준 |
| 이미지·사운드 파일 의존 | 없으면 대체 Surface |
| 가상환경에만 설치된 패키지 | `requirements.txt` |
| `data/` 폴더가 이미 있음 | `mkdir(exist_ok=True)` |
| 한글 폰트가 있음 | 폰트 없을 때 대체 |

**오늘 오후에 반드시 다른 PC 또는 새 가상환경에서 테스트합니다.**
발표 전날인 오늘 발견하면 고칠 수 있지만, 내일 아침에 발견하면 늦습니다.

```python
NAME_INPUT = "NAME_INPUT"
RANKING = "RANKING"

ALLOWED_CHARS = ("abcdefghijklmnopqrstuvwxyz"
                 "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789_ ")


class TextInput:
    """이름 입력 위젯."""

    def __init__(self, max_length=12):
        self.text = ""
        self.max_length = max_length
        self.cursor_timer = 0

    def handle_event(self, event):
        if event.type != pygame.KEYDOWN:
            return False
        if event.key == pygame.K_RETURN:
            return bool(self.text.strip())
        if event.key == pygame.K_BACKSPACE:
            self.text = self.text[:-1]
            return False

        ch = event.unicode
        if ch and ch in ALLOWED_CHARS and len(self.text) < self.max_length:
            self.text += ch
        return False

    def update(self):
        self.cursor_timer = (self.cursor_timer + 1) % 60

    @property
    def value(self):
        return self.text.strip() or "PLAYER"


class GameManager:
    def __init__(self, screen):
        ...
        self.text_input = TextInput()
        self.cached_ranking = []
        self.cached_best = None
        self.new_record_id = None
        self.last_rank = None

        try:
            db.init()
            self._refresh_title_info()
        except Exception as e:
            print(f"[경고] DB 초기화 실패 — 랭킹 기능이 비활성화됩니다: {e}")
            self.db_available = False
        else:
            self.db_available = True

    # ── DB 접근 (상태 전환 시에만) ──

    def _refresh_title_info(self):
        if not self.db_available:
            return
        try:
            top = db.get_ranking(1)
            self.cached_best = top[0] if top else None
        except Exception:
            self.cached_best = None

    def _refresh_ranking(self):
        if not self.db_available:
            self.cached_ranking = []
            return
        try:
            self.cached_ranking = db.get_ranking(10)
        except Exception as e:
            print(f"[경고] 랭킹 조회 실패: {e}")
            self.cached_ranking = []

    def _save_record(self):
        """기록을 저장합니다. 실패해도 게임은 계속됩니다."""
        if not self.db_available:
            return None
        try:
            s = self.stats.summary()
            record_id, rank = db.save_score(
                self.text_input.value, s["score"],
                kills=s["kills"], max_combo=s["max_combo"],
                shots=self.stats.shots, hits=self.stats.hits,
                play_seconds=s["play_seconds"], stage=s["stage"],
            )
            self.new_record_id = record_id
            self.last_rank = rank
            return rank
        except Exception as e:
            print(f"[경고] 기록 저장 실패: {e}")
            return None

    def _change_state(self, new_state):
        self.state = new_state

        if new_state == TITLE:
            self._refresh_title_info()
            self.new_record_id = None
            self.last_rank = None
        elif new_state == RANKING:
            self._refresh_ranking()
        elif new_state == NAME_INPUT:
            self.text_input.text = ""

    # ── 이벤트 ──

    def handle_event(self, event):
        if event.type != pygame.KEYDOWN:
            return True

        if event.key == pygame.K_ESCAPE and self.state != NAME_INPUT:
            return False

        if self.state == TITLE:
            if event.key == pygame.K_RETURN:
                self.start_game()
            elif event.key == pygame.K_r:
                self._change_state(RANKING)

        elif self.state == PLAYING:
            if event.key == pygame.K_p:
                self._change_state(PAUSED)

        elif self.state == PAUSED:
            if event.key == pygame.K_p:
                self._change_state(PLAYING)
            elif event.key == pygame.K_q:
                self._change_state(TITLE)

        elif self.state == GAME_OVER:
            if event.key == pygame.K_RETURN:
                is_new = False
                if self.db_available:
                    try:
                        is_new = db.is_new_record(self.stats.score)
                    except Exception:
                        is_new = False

                self._change_state(NAME_INPUT if is_new else RANKING)

        elif self.state == NAME_INPUT:
            if self.text_input.handle_event(event):
                self._save_record()
                self._change_state(RANKING)

        elif self.state == RANKING:
            if event.key == pygame.K_RETURN:
                self._change_state(TITLE)

        return True
```

_font_cache = {}
_font_warned = False


def make_font(size=FONT_SIZE_NORMAL, name=FONT_NAME):
    """폰트를 만듭니다. 한글이 안 되면 기본 폰트로 대체합니다."""
    global _font_warned

    key = (name, size)
    if key in _font_cache:
        return _font_cache[key]

    font = None
    try:
        candidate = pygame.font.SysFont(name, size)
        test = candidate.render("가", True, (255, 255, 255))
        if test.get_width() > 2:
            font = candidate
    except Exception:
        font = None

    if font is None:
        if not _font_warned:
            print(f"[경고] '{name}' 폰트를 사용할 수 없어 기본 폰트를 씁니다. "
                  f"한글이 깨질 수 있습니다.")
            _font_warned = True
        font = pygame.font.Font(None, size)

    _font_cache[key] = font
    return font

**`game/audio.py` — 사운드 대체**

```python
"""사운드 관리 — 파일이 없어도 동작합니다."""

import pygame

from .paths import SOUND_DIR

_sounds = {}
_enabled = True


def init():
    """믹서를 초기화합니다. 실패하면 무음 모드로 전환합니다."""
    global _enabled
    try:
        pygame.mixer.init()
        _enabled = True
    except pygame.error as e:
        print(f"[경고] 사운드 장치를 열 수 없어 무음으로 실행합니다: {e}")
        _enabled = False


def load(name, filename, volume=0.4):
    """사운드를 불러옵니다. 실패하면 None 을 등록합니다."""
    if not _enabled:
        _sounds[name] = None
        return

    path = SOUND_DIR / filename
    try:
        s = pygame.mixer.Sound(str(path))
        s.set_volume(volume)
        _sounds[name] = s
    except (pygame.error, FileNotFoundError):
        _sounds[name] = None


def play(name):
    """사운드를 재생합니다. 없으면 조용히 넘어갑니다."""
    s = _sounds.get(name)
    if s is not None:
        try:
            s.play()
        except pygame.error:
            pass


def load_all():
    """게임에서 쓰는 사운드를 모두 불러옵니다."""
    load("shoot", "shoot.wav", 0.2)
    load("hit", "hit.wav", 0.35)
    load("explode", "explode.wav", 0.5)
    load("item", "item.wav", 0.4)
    load("gameover", "gameover.wav", 0.5)

    missing = [k for k, v in _sounds.items() if v is None]
    if missing and _enabled:
        print(f"[안내] 사운드 파일 없음: {', '.join(missing)} (무음으로 진행)")
```

**`main.py` — 전체 예외 처리**

```python
"""슈팅게임 진입점."""

import sys
import traceback

import pygame

from game import audio, paths
from game.settings import SCREEN_WIDTH, SCREEN_HEIGHT, FPS, TITLE
from game.scenes.manager import GameManager


def main():
    paths.ensure_dirs()

    pygame.init()
    audio.init()
    audio.load_all()

    screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT))
    pygame.display.set_caption(TITLE)
    clock = pygame.time.Clock()

    manager = GameManager(screen)

    running = True
    while running:
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                running = False
            else:
                running = manager.handle_event(event)

        manager.update()
        manager.draw()

        pygame.display.flip()
        clock.tick(FPS)

    manager.quit()
    pygame.quit()


if __name__ == "__main__":
    try:
        main()
    except Exception:
        # 예상치 못한 오류도 사용자에게 알린 뒤 종료
        print("\n" + "=" * 60)
        print("예상치 못한 오류가 발생했습니다.")
        print("=" * 60)
        traceback.print_exc()
        print("=" * 60)
        pygame.quit()
        input("\nEnter 를 눌러 종료합니다...")     # 창이 바로 닫히지 않게
        sys.exit(1)
```

> **`input()` 을 넣은 이유** — 더블클릭으로 실행했을 때 오류가 나면 창이 즉시 닫혀
> 무슨 에러인지 볼 수 없습니다. 이 한 줄이 디버깅을 가능하게 합니다.

## ⑤ 자주 하는 실수

### 다른 PC 테스트를 안 함

**증상** — 발표 당일 실행이 안 됩니다.
**해결** — **오늘 오후에 반드시** 새 가상환경에서 테스트합니다.

### `requirements.txt` 를 갱신 안 함

**증상** — 받는 사람이 `pygame` 을 설치하지 못합니다.
**해결** — 마지막에 `pip freeze > requirements.txt` 를 다시 실행합니다.

### DB 파일을 Git 에 올림

**증상** — 병합 충돌이 나고, 남의 랭킹 기록이 섞입니다.
**해결** — `.gitignore` 에 `data/*.db` 를 추가합니다.

### 오류 창이 바로 닫혀 원인을 모름

**해결** — `main.py` 의 `except` 에 `input()` 을 넣습니다.

### 예외를 전부 삼킴

```python
try:
    ...
except:
    pass                    # 무슨 일이 있었는지 알 수 없음
```

**해결** — 최소한 `print` 로 남깁니다.

```python
except Exception as e:
    print(f"[경고] 랭킹 저장 실패: {e}")
```

### 발표 전날 새 기능 추가

**증상** — 버그가 생겼는데 고칠 시간이 없습니다.
**해결** — **오늘은 기능 동결(feature freeze)** 입니다. 버그 수정과 배포 준비만 합니다.

### README 에 실행 방법이 없음

**증상** — 받는 사람이 실행하지 못합니다.
**해결** — **복사해서 붙여넣으면 되는 명령**을 그대로 적습니다.

### 태그를 안 만듦

**증상** — 발표 직전에 코드를 망가뜨렸는데 되돌릴 지점이 없습니다.
**해결** — 작동하는 상태에서 `git tag v1.0`.

```python
"""안전한 초기화."""

import sqlite3
from pathlib import Path

import pygame

# ── ① 경로: 파일 위치 기준 ──
BASE_DIR = Path(__file__).resolve().parent
IMAGE_DIR = BASE_DIR / "assets" / "images"
SOUND_DIR = BASE_DIR / "assets" / "sounds"
DATA_DIR = BASE_DIR / "data"

# ── ④ 폴더 자동 생성 ──
for d in (IMAGE_DIR, SOUND_DIR, DATA_DIR):
    d.mkdir(parents=True, exist_ok=True)

pygame.init()

# ── ⑤ 믹서 초기화 (실패해도 계속) ──
try:
    pygame.mixer.init()
    audio_enabled = True
except pygame.error:
    print("[경고] 사운드 장치를 열 수 없어 무음으로 실행합니다.")
    audio_enabled = False

screen = pygame.display.set_mode((800, 600))


# ── ② 폰트: 대체 처리 ──
def make_font(size, name="malgungothic"):
    try:
        f = pygame.font.SysFont(name, size)
        if f.render("가", True, (255, 255, 255)).get_width() > 2:
            return f
    except Exception:
        pass
    print(f"[경고] '{name}' 폰트를 쓸 수 없어 기본 폰트를 사용합니다.")
    return pygame.font.Font(None, size)


font = make_font(24)


# ── ③ 이미지: 없으면 코드로 생성 ──
def make_ship_surface(size=44):
    s = pygame.Surface((size, size), pygame.SRCALPHA)
    pygame.draw.polygon(s, (52, 152, 219),
                        [(size // 2, 0), (size, size),
                         (size // 2, size * 0.78), (0, size)])
    return s


def load_image(filename, fallback):
    path = IMAGE_DIR / filename
    try:
        return pygame.image.load(str(path)).convert_alpha()
    except (pygame.error, FileNotFoundError):
        return fallback


player_img = load_image("player.png", make_ship_surface())


# ── ③ 사운드: 없으면 None ──
def load_sound(filename, volume=0.4):
    if not audio_enabled:
        return None
    try:
        s = pygame.mixer.Sound(str(SOUND_DIR / filename))
        s.set_volume(volume)
        return s
    except (pygame.error, FileNotFoundError):
        print(f"[안내] 사운드 없음: {filename} (무음으로 진행)")
        return None


def play(sound):
    if sound is not None:
        sound.play()


shoot_sound = load_sound("shoot.wav", 0.2)


# ── ④ DB: 폴더 생성 후 연결 ──
conn = sqlite3.connect(DATA_DIR / "game.db")
conn.row_factory = sqlite3.Row
```

```python
"""배포 전 자동 점검."""

import re
import sys
from pathlib import Path

BASE_DIR = Path(__file__).resolve().parent
IGNORE = {"venv", ".venv", "__pycache__", ".git"}

issues = []


def check_absolute_paths():
    """절대경로 하드코딩을 찾습니다."""
    pattern = re.compile(r'["\'](?:[A-Za-z]:[\\/]|/Users/|/home/)')

    for py in BASE_DIR.rglob("*.py"):
        if any(p in IGNORE for p in py.parts):
            continue
        try:
            for no, line in enumerate(py.read_text(encoding="utf-8").splitlines(), 1):
                if pattern.search(line):
                    issues.append(f"절대경로: {py.relative_to(BASE_DIR)}:{no}")
        except UnicodeDecodeError:
            pass


def check_required_files():
    """필수 파일이 있는지 확인합니다."""
    for name in ("main.py", "requirements.txt", "README.md", ".gitignore"):
        if not (BASE_DIR / name).exists():
            issues.append(f"파일 없음: {name}")


def check_gitignore():
    """.gitignore 항목을 확인합니다."""
    path = BASE_DIR / ".gitignore"
    if not path.exists():
        return

    content = path.read_text(encoding="utf-8")
    for entry in ("venv", "__pycache__", ".db"):
        if entry not in content:
            issues.append(f".gitignore 에 '{entry}' 없음")


def check_long_files():
    """300줄이 넘는 파일을 찾습니다."""
    for py in BASE_DIR.rglob("*.py"):
        if any(p in IGNORE for p in py.parts):
            continue
        try:
            lines = len(py.read_text(encoding="utf-8").splitlines())
            if lines > 300:
                issues.append(f"긴 파일 ({lines}줄): {py.relative_to(BASE_DIR)}")
        except UnicodeDecodeError:
            pass


if __name__ == "__main__":
    check_absolute_paths()
    check_required_files()
    check_gitignore()
    check_long_files()

    print("=" * 60)
    if issues:
        print(f"점검 결과: 문제 {len(issues)}건")
        print("-" * 60)
        for i in issues:
            print(f"  ⚠ {i}")
        print("=" * 60)
        sys.exit(1)
    else:
        print("점검 결과: 문제 없음 ✅")
        print("=" * 60)
```
