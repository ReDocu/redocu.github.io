### 요금 텍스트 파싱 — 오늘의 핵심 과제

```
"성인 3,000원 / 청소년 2,000원"   →  3000  (성인 기준)
"무료"                            →  0
"입장료 없음"                     →  0
"성인 5,000원(단체 4,000원)"      →  5000  (괄호 제거 후 첫 값)
NaN                               →  NaN   ("정보 없음"이지 무료가 아님!)
```

```python
FREE_WORDS = ["무료", "없음", "무료입장", "0원", "면제"]

def parse_fee(text):
    """
    요금 텍스트 → (가격, 무료여부, 신뢰도)

    신뢰도: 'exact'(숫자 명시) / 'free'(무료 명시) / 'unknown'
    """
    if pd.isna(text) or not str(text).strip():
        return np.nan, np.nan, "unknown"

    s = str(text)

    # 괄호 안 제거 (단체·할인 요금)
    s_clean = re.sub(r"\([^)]*\)", " ", s)

    # 무료 판정
    if any(w in s_clean for w in FREE_WORDS) and not re.search(r"\d", s_clean):
        return 0.0, True, "free"

    # 성인 요금 우선 추출
    adult = re.search(r"(?:성인|어른|일반)\D{0,6}([\d,]+)\s*원", s_clean)
    if adult:
        return float(adult.group(1).replace(",", "")), False, "exact"

    # 첫 숫자
    nums = re.findall(r"([\d,]{3,})\s*원", s_clean)
    if nums:
        return float(nums[0].replace(",", "")), False, "exact"

    # 무료 단어는 있으나 숫자도 있는 애매한 경우
    if any(w in s_clean for w in FREE_WORDS):
        return 0.0, True, "free"

    return np.nan, np.nan, "unknown"
```

> ⚠️ **`NaN` 을 0으로 채우지 마세요.**
> "정보 없음"과 "무료"는 완전히 다릅니다.
> 채우면 **유료 시설이 무료로 둔갑**합니다.

```
발견: "도보 15분 이내는 14%뿐이고, 그중 음식점이 60%"
   ↓
기준: "도보 20분으로 완화하고, 카테고리 다양성을 강제한다"
```

```
발견: "요금 정보가 18%뿐이지만, 무료 여부는 62%에서 판정 가능"
   ↓
기준: "가격 대신 '무료 여부'를 가성비 지표로 쓴다"
```

> **EDA 의 결과가 설계 결정으로 이어져야 합니다.**
> "재미있는 그래프를 그렸다"로 끝나면 EDA 가 아닙니다.

### 추천 기준 확정 문서

| 항목 | 결정 | 근거(EDA) |
|---|---|---|
| 후보 필터 | 좌표 유효 + 카테고리 5종 | 좌표 무효 5건, 숙박 제외 |
| 접근성 기준 | 도보 **20분** | 15분은 14%뿐 |
| 가성비 지표 | **무료 여부** (가격 ❌) | 가격 채움률 18% |
| 평점 대체 | **이미지 보유 + 리뷰 수** | 평점 데이터 없음 |
| 다양성 | **필수** (MMR 0.4) | 음식점 39% 편중 |

## ④ 단계별 실습

### Step 1. 전처리 실행

```python
"""src/preprocess.py — 프로젝트 전처리 파이프라인."""

import re
import unicodedata
from datetime import datetime
from pathlib import Path

import numpy as np
import pandas as pd

INTERIM = Path("data/interim")
PROCESSED = Path("data/processed")
DOCS = Path("docs")

for d in (PROCESSED, DOCS):
    d.mkdir(parents=True, exist_ok=True)

INVISIBLE = r"[​-‏﻿\xa0　\t\r\n]"
NA_TEXTS = ["", " ", "-", "--", "N/A", "없음정보", "미상", "null", "?"]

FREE_WORDS = ["무료", "없음", "무료입장", "0원", "면제", "무료관람"]

DOMAIN_RULES = {
    "lat": (33.0, 39.0),
    "lng": (124.0, 132.0),
    "price": (0, 500_000),
    "station_km": (0, 100),
    "walk_min": (0, 1500),
}

USE_CATEGORIES = ["관광지", "문화시설", "레포츠", "쇼핑", "음식점"]

WALK_SPEED_KMH = 4.0


# ══════════ 단계별 함수 ══════════

def clean_strings(df):
    """① 문자열 정제."""
    out = df.copy()

    for col in out.select_dtypes(include=["object", "string"]).columns:
        out[col] = (out[col].astype("string")
                    .map(lambda v: unicodedata.normalize("NFC", v)
                         if isinstance(v, str) else v)
                    .str.replace(INVISIBLE, " ", regex=True)
                    .str.replace(r"\s+", " ", regex=True)
                    .str.strip())
        out[col] = out[col].mask(out[col].isin(NA_TEXTS), pd.NA)

    return out


def parse_fee(text):
    """요금 텍스트 → (가격, 무료여부, 신뢰도)."""
    if pd.isna(text) or not str(text).strip():
        return np.nan, np.nan, "unknown"

    s = re.sub(r"\([^)]*\)", " ", str(text))

    has_digit = bool(re.search(r"\d", s))
    has_free_word = any(w in s for w in FREE_WORDS)

    if has_free_word and not has_digit:
        return 0.0, True, "free"

    adult = re.search(r"(?:성인|어른|일반|대인)\D{0,8}([\d,]+)\s*원", s)
    if adult:
        return float(adult.group(1).replace(",", "")), False, "exact"

    nums = re.findall(r"([\d,]{3,})\s*원", s)
    if nums:
        return float(nums[0].replace(",", "")), False, "exact"

    if has_free_word:
        return 0.0, True, "free"

    return np.nan, np.nan, "unknown"


def parse_fees(df):
    """② 요금 파싱."""
    out = df.copy()

    if "fee_text" not in out.columns:
        out["price"] = np.nan
        out["is_free"] = pd.NA
        out["fee_confidence"] = "unknown"
        return out

    parsed = out["fee_text"].map(parse_fee)
    out["price"] = [p[0] for p in parsed]
    out["is_free"] = [p[1] for p in parsed]
    out["fee_confidence"] = [p[2] for p in parsed]

    return out


def drop_duplicates_smart(df, key="id"):
    """③ 중복 제거 — 키 기준 + 좌표 근접 후보 표시."""
    out = df.copy()

    # 키 중복
    if key in out.columns:
        out = out.drop_duplicates(subset=[key], keep="first")

    # 이름+좌표 반올림 중복 (같은 장소 다른 등록)
    if {"name", "lat", "lng"} <= set(out.columns):
        out["_dup_key"] = (out["name"].astype(str).str.replace(r"\s", "",
                                                                regex=True)
                           + "_" + out["lat"].round(3).astype(str)
                           + "_" + out["lng"].round(3).astype(str))
        out = out.drop_duplicates(subset=["_dup_key"], keep="first")
        out = out.drop(columns="_dup_key")

    return out.reset_index(drop=True)


def handle_outliers(df):
    """④ 도메인 규칙 위반 → NaN + 플래그."""
    out = df.copy()

    for col, (lo, hi) in DOMAIN_RULES.items():
        if col not in out.columns:
            continue

        mask = ((out[col] < lo) | (out[col] > hi)).fillna(False)

        if mask.any():
            out[f"{col}_was_invalid"] = mask
            out.loc[mask, col] = np.nan

    return out


def handle_missing(df):
    """⑤ 결측 처리 — 플래그 먼저, 채우지 않는 것도 전략."""
    out = df.copy()

    for col in ("price", "station_km", "walk_min"):
        if col in out.columns and out[col].isna().any():
            out[f"{col}_was_na"] = out[col].isna()

    # 이미지·전화는 존재 여부만
    for col in ("image", "tel"):
        if col in out.columns:
            out[f"has_{col}"] = (out[col].notna()
                                 & (out[col].astype(str).str.strip() != ""))

    # price 는 채우지 않음 — "정보 없음" ≠ "무료"
    return out


def add_features(df):
    """⑥ 파생 변수."""
    out = df.copy()

    # ── 접근성 ──
    if "station_km" in out.columns:
        out["walk_min"] = (out["station_km"] / WALK_SPEED_KMH * 60).round(1)

        out["access_grade"] = pd.cut(
            out["walk_min"],
            bins=[-np.inf, 10, 20, 30, np.inf],
            labels=["매우좋음", "좋음", "보통", "나쁨"])

        out["access_grade"] = out["access_grade"].astype(
            pd.CategoricalDtype(["매우좋음", "좋음", "보통", "나쁨"],
                                ordered=True))

    # ── 가격 등급 ──
    if "price" in out.columns:
        out["price_grade"] = pd.cut(
            out["price"],
            bins=[-np.inf, 0, 5_000, 15_000, np.inf],
            labels=["무료", "저가", "중가", "고가"])

    # ── 정보 충실도 (평점 대체 지표) ──
    info_cols = [c for c in ("has_image", "has_tel") if c in out.columns]
    if info_cols:
        out["info_score"] = out[info_cols].sum(axis=1) / len(info_cols)

    # ── 수정 시각 ──
    if "modified" in out.columns:
        out["modified_at"] = pd.to_datetime(
            out["modified"], format="%Y%m%d%H%M%S", errors="coerce")
        out["modified_year"] = out["modified_at"].dt.year

    return out


def filter_scope(df):
    """⑦ 분석 범위로 한정."""
    out = df.copy()

    before = len(out)

    if "category" in out.columns:
        out = out[out["category"].isin(USE_CATEGORIES)]

    if {"lat", "lng"} <= set(out.columns):
        out = out[out["lat"].notna() & out["lng"].notna()]

    return out.reset_index(drop=True), before - len(out)


# ══════════ 파이프라인 ══════════

def preprocess(df, verbose=True):
    """원본 → 분석용. 처리 이력을 함께 반환."""
    out = df.copy()
    history = []

    steps = [
        ("① 문자열 정제", clean_strings),
        ("② 요금 파싱", parse_fees),
        ("③ 중복 제거", drop_duplicates_smart),
        ("④ 이상치 처리", handle_outliers),
        ("⑤ 결측 처리", handle_missing),
        ("⑥ 파생 변수", add_features),
    ]

    for label, fn in steps:
        before = (len(out), out.shape[1], int(out.isna().sum().sum()))
        out = fn(out)
        after = (len(out), out.shape[1], int(out.isna().sum().sum()))

        history.append({
            "단계": label,
            "행": f"{before[0]:,}→{after[0]:,}",
            "열": f"{before[1]}→{after[1]}",
            "결측": f"{before[2]:,}→{after[2]:,}",
        })

    out, dropped = filter_scope(out)
    history.append({
        "단계": "⑦ 범위 한정",
        "행": f"{len(out)+dropped:,}→{len(out):,}",
        "열": f"{out.shape[1]}→{out.shape[1]}",
        "결측": f"{dropped}건 제외",
    })

    hist = pd.DataFrame(history)

    if verbose:
        print("=" * 76)
        print("                       전처리 실행")
        print("=" * 76)
        print(f"\n{hist.to_string(index=False)}")

    return out, hist


# ══════════ 실행 ══════════
if __name__ == "__main__":
    raw = pd.read_parquet(INTERIM / "raw_merged.parquet")
    print(f"[원본] {raw.shape}")

    clean, history = preprocess(raw)

    # ── 요금 파싱 결과 ──
    print(f"\n[요금 파싱 결과]")
    conf = clean["fee_confidence"].value_counts()
    for c, n in conf.items():
        label = {"exact": "숫자 명시", "free": "무료 명시",
                 "unknown": "정보 없음"}.get(c, c)
        bar = "■" * int(n / len(clean) * 40)
        print(f"  {label:<12}{n:>6,} ({n/len(clean)*100:>5.1f}%)  {bar}")

    known = clean["is_free"].notna().sum()
    print(f"\n  무료 여부 판정 가능: {known:,}건 "
          f"({known/len(clean)*100:.1f}%)")
    print(f"  그중 무료          : {int(clean['is_free'].sum()):,}건")
    print(f"  가격 숫자 확보     : {clean['price'].notna().sum():,}건 "
          f"({clean['price'].notna().mean()*100:.1f}%)")

    print(f"\n  ⚠ 가격 채움률이 낮아 '무료 여부'를 가성비 지표로 사용합니다")

    # ── 접근성 ──
    if "walk_min" in clean.columns:
        print(f"\n[접근성 분포]")
        acc = clean["access_grade"].value_counts().sort_index()
        for g, n in acc.items():
            bar = "■" * int(n / len(clean) * 40)
            print(f"  {str(g):<10}{n:>6,} ({n/len(clean)*100:>5.1f}%)  {bar}")

    # ── 저장 ──
    clean.to_parquet(PROCESSED / "analysis_table.parquet", index=False)
    clean.to_csv(PROCESSED / "analysis_table.csv",
                 index=False, encoding="utf-8-sig")
    history.to_csv(PROCESSED / "preprocess_log.csv",
                   index=False, encoding="utf-8-sig")

    print(f"\n[저장]")
    print(f"  data/processed/analysis_table.parquet  "
          f"({len(clean):,}행 × {clean.shape[1]}열)")
    print(f"  data/processed/preprocess_log.csv")

    # ── 멱등성 확인 ──
    twice, _ = preprocess(clean, verbose=False)
    same = clean.shape == twice.shape
    print(f"\n[멱등성] {'✅ 통과' if same else '❌ 실패'} "
          f"({clean.shape} vs {twice.shape})")

    print("\n" + "=" * 76)
```

```python
print(f"\n[① 지표 후보 평가]")
print(f"  {'지표':<14}{'채움률':>9}{'고유값':>8}{'변별력':>9}  판정")

decisions = []

for name, info in CANDIDATES.items():
    col = info["열"]
    if col not in df.columns:
        continue

    fill = df[col].notna().mean()
    nuniq = df[col].nunique()

    s = df[col].dropna()
    if pd.api.types.is_bool_dtype(s) or nuniq <= 2:
        p = s.astype(float).mean() if len(s) else 0
        power = 1 - abs(p - 0.5) * 2          # 0.5 에 가까울수록 변별력 ↑
    else:
        power = (s.rank(pct=True).std() * 2) if len(s) > 1 else 0

    # 판정
    if fill < 0.3:
        verdict, use = "❌ 채움률 부족", False
    elif power < 0.2:
        verdict, use = "❌ 변별력 없음", False
    else:
        verdict, use = "✅ 사용", True

    decisions.append({"지표": name, "채움률": round(fill, 3),
                      "변별력": round(power, 3), "사용": use})

    print(f"  {name:<14}{fill*100:>8.1f}%{nuniq:>8}{power:>9.3f}  {verdict}")

print(f"  {'지표':<18}{'가중치':>8}{'원천':<14}근거")
print("  " + "-" * 76)
for name, (w, col, reason) in WEIGHTS.items():
```

## ⑤ 자주 하는 실수

### 결측을 0으로 채움

요금 `NaN` 을 0으로 채우면 **유료 시설이 무료가 됩니다.**

**해결** — "정보 없음"으로 남기고 플래그를 만듭니다.

### 파생 관계를 발견으로 착각

`walk_min` 과 `station_km` 의 r=1.00 은 **계산으로 만든 관계**입니다.

**해결** — 계산식으로 만든 열끼리의 상관은 제외합니다.

### 채움률이 낮은 열을 그대로 사용

가격 13.5%로 가성비 점수를 만들면 **86%가 점수를 못 받습니다.**

**해결** — 대체 지표를 찾거나 제외합니다.

### EDA 가 그래프 모음으로 끝남

**해결** — 발견마다 **"→ 결정"** 을 붙입니다.

### 편중을 못 보고 넘어감

역세권 54%가 음식점인 걸 모르면 **추천이 전부 식당**이 됩니다.

**해결** — 필터 후 그룹 구성을 확인합니다.

### 기준을 감으로 정함

"도보 15분"을 근거 없이 정하면 후보가 20%뿐입니다.

**해결** — 여러 기준의 해당 건수를 표로 비교합니다.

### 전처리 이력을 안 남김

**해결** — 단계별 행·열·결측 변화를 표로 저장합니다.

### 멱등성 확인 안 함

**해결** — 두 번 돌려 같은 결과인지 확인합니다.

### 컬럼 정의서를 갱신 안 함

초안(월요일)과 실제 열이 달라집니다.

**해결** — 오늘 실제 데이터로 갱신합니다.

### 원본을 덮어씀

**해결** — `raw/` 는 읽기만, 결과는 `processed/` 로.

## ⑥ 확인 문제

**1.** 오늘 만든 분석용 테이블이 내일 쓸 수 있는 상태인지 검증하세요.

```python
"""분석용 테이블 준비도 검증."""

import json
from pathlib import Path

import numpy as np
import pandas as pd


def verify_ready(df, design_path="docs/recommend_design.json", verbose=True):
    """
    확정한 추천 설계를 실행할 수 있는 상태인지 검증합니다.
    """
    design = json.loads(Path(design_path).read_text(encoding="utf-8"))

    issues = []
    line = "=" * 78

    if verbose:
        print(line)
        print("                    분석용 테이블 준비도 검증")
        print(line)
        print(f"\n[테이블]  {len(df):,}행 × {df.shape[1]}열")

    # ══════════ ① 설계에 필요한 열이 있는가 ══════════
    needed = {v["원천"] for v in design["지표"].values()}
    needed |= {"id", "name", "lat", "lng", "category"}

    missing = [c for c in needed if c not in df.columns]

    if verbose:
        print(f"\n[① 필요 열]")
        for c in sorted(needed):
            present = c in df.columns
            fill = (df[c].notna().mean() * 100) if present else 0
            print(f"  {'✓' if present else '✗'} {c:<16}"
                  f"{fill:>6.1f}%" if present else f"  ✗ {c:<16}  없음")

    if missing:
        issues.append(("치명", f"설계에 필요한 열 누락: {missing}"))

    # ══════════ ② 지표별 채움률 ══════════
    if verbose:
        print(f"\n[② 지표 채움률]")

    for name, spec in design["지표"].items():
        col = spec["원천"]
        if col not in df.columns:
            continue

        fill = df[col].notna().mean()
        w = spec["가중치"]

        # 가중치가 높은데 채움률이 낮으면 문제
        risk = w * (1 - fill)

        if verbose:
            mark = "⚠" if risk > 0.15 else " "
            print(f" {mark}{name:<18}{col:<14}채움 {fill*100:>5.1f}% · "
                  f"가중치 {w:.2f} · 위험도 {risk:.3f}")

        if risk > 0.15:
            issues.append(("경고",
                           f"'{name}' 가중치 {w} 인데 채움률 {fill*100:.0f}% "
                           f"— {(1-fill)*100:.0f}% 가 점수를 못 받음"))

    # ══════════ ③ 필터 통과 건수 ══════════
    pool = df.copy()

    cats = design["필터"].get("카테고리", [])
    if cats and "category" in pool.columns:
        pool = pool[pool["category"].isin(cats)]

    walk_filter = design["필터"].get("접근성", "")
    minutes = None
    if "walk_min" in pool.columns:
        import re
        m = re.search(r"(\d+)\s*분", walk_filter)
        if m:
            minutes = int(m.group(1))
            pool = pool[pool["walk_min"] <= minutes]

    if verbose:
        print(f"\n[③ 필터 통과]")
        print(f"  전체        : {len(df):,}건")
        print(f"  카테고리 후 : {len(df[df['category'].isin(cats)]):,}건")
        print(f"  접근성 후   : {len(pool):,}건 "
              f"({len(pool)/len(df)*100:.1f}%)")

    if len(pool) < 50:
        issues.append(("치명", f"필터 통과 {len(pool)}건 — "
                               f"추천 후보가 부족합니다"))
    elif len(pool) < 150:
        issues.append(("경고", f"필터 통과 {len(pool)}건 — "
                               f"조건 완화를 검토하세요"))

    # ══════════ ④ 다양성 확보 가능한가 ══════════
    if "category" in pool.columns and len(pool):
        cat_counts = pool["category"].value_counts()
        n_cats = len(cat_counts)
        top_share = cat_counts.iloc[0] / len(pool)

        if verbose:
            print(f"\n[④ 후보 다양성]")
            print(f"  카테고리 종류 : {n_cats}개")
            print(f"  최다 비중     : {cat_counts.index[0]} "
                  f"{top_share*100:.1f}%")
            print(f"  구성:")
            for c, n in cat_counts.items():
                bar = "■" * int(n / cat_counts.max() * 20)
                print(f"    {c:<10}{n:>5,} ({n/len(pool)*100:>5.1f}%)  {bar}")

        if n_cats < 3:
            issues.append(("치명", f"후보에 카테고리가 {n_cats}종뿐 — "
                                   f"다양성 확보 불가"))
        if top_share > 0.7:
            issues.append(("경고", f"'{cat_counts.index[0]}' 가 "
                                   f"{top_share*100:.0f}% — 다양성 강도를 "
                                   f"더 높이세요"))

    # ══════════ ⑤ 점수 계산 시뮬레이션 ══════════
    if verbose:
        print(f"\n[⑤ 점수 계산 시뮬레이션]")

    try:
        sim = pool.copy()
        parts = {}

        for name, spec in design["지표"].items():
            col = spec["원천"]
            if col not in sim.columns:
                continue

            s = sim[col]

            if pd.api.types.is_bool_dtype(s) or s.dropna().isin([True, False]).all():
                score = s.fillna(False).astype(float)
            elif pd.api.types.is_numeric_dtype(s):
                score = s.rank(pct=True)
                if "낮을수록" in str(spec.get("근거", "")) or col == "walk_min":
                    score = 1 - score
            else:
                continue

            parts[name] = score * spec["가중치"]

        if parts:
            sim["score"] = sum(parts.values())

            valid_score = sim["score"].notna().mean()
            in_range = sim["score"].dropna().between(0, 1.001).all()
            spread = sim["score"].std()

            if verbose:
                print(f"  점수 계산 성공 : {valid_score*100:.1f}%")
                print(f"  0~1 범위       : {'✓' if in_range else '✗'}")
                print(f"  분산(표준편차) : {spread:.3f}")
                print(f"  상위 3건:")
                for _, r in sim.nlargest(3, "score").iterrows():
                    print(f"    {r['score']:.3f}  {r['name'][:24]:<26}"
                          f"{r['category']}  도보 {r['walk_min']:.0f}분")

            if valid_score < 0.9:
                issues.append(("경고", f"점수를 계산할 수 없는 행 "
                                       f"{(1-valid_score)*100:.0f}%"))
            if not in_range:
                issues.append(("치명", "점수가 0~1 범위를 벗어남"))
            if spread < 0.05:
                issues.append(("치명", f"점수 분산 {spread:.3f} — "
                                       f"변별력이 없어 순위가 무의미"))
        else:
            issues.append(("치명", "점수를 계산할 수 있는 지표가 없음"))

    except Exception as e:
        issues.append(("치명", f"점수 계산 실패: {type(e).__name__}: {e}"))
        if verbose:
            print(f"  🔴 {type(e).__name__}: {e}")

    # ══════════ 결과 ══════════
    fatal = sum(1 for l, _ in issues if l == "치명")
    passed = fatal == 0

    if verbose:
        print(f"\n{line}")
        print(f"  판정: {'✅ 내일 구현 가능' if passed else '❌ 오늘 보완 필요'}"
              f"   (치명 {fatal} · 경고 {len(issues)-fatal})")
        print(line)

        for level, msg in issues:
            print(f"  {'🔴' if level == '치명' else '🟡'} {msg}")

        if not issues:
            print("  문제 없음")

        print(f"\n  [Day 074 준비 체크리스트]")
        for item in [
            "analysis_table.parquet 이 저장됐는가",
            "recommend_design.json 이 확정됐는가",
            "컬럼 정의서를 실제 열로 갱신했는가",
            "발견 3가지를 findings.csv 에 기록했는가",
            "EDA 그래프를 output/eda/ 에 저장했는가",
            "전처리 멱등성을 확인했는가",
        ]:
            print(f"    □ {item}")

        print(line)

    return {"passed": passed, "issues": issues, "pool_size": len(pool)}


# ══════════ 사용 ══════════
if __name__ == "__main__":
    verify_ready(df)
```

**검증 5단계**

| # | 검증 | 잡아내는 문제 |
|---|---|---|
| ① | 필요 열 존재 | 설계와 데이터 불일치 |
| ② | **가중치 × 결측 위험도** | **높은 가중치인데 결측 많음** ⭐ |
| ③ | 필터 통과 건수 | 후보 부족 |
| ④ | 후보 다양성 | 한 카테고리 독점 |
| ⑤ | **점수 시뮬레이션** | **분산 없음 = 순위 무의미** ⭐ |

> **②번의 "위험도 = 가중치 × (1 - 채움률)"** 이 유용한 지표입니다.
>
> ```
> is_free: 가중치 0.30 × 결측 38% = 위험도 0.115
> ```
>
> **38%의 항목이 가성비 점수를 못 받습니다.**
> 이들은 자동으로 불리해집니다.
>
> **대응 2가지**
> - 결측을 중립값(0.5)으로 채워 불이익을 없앤다
> - 가중치를 낮춘다 (0.30 → 0.20)
>
> 내일 구현할 때 **어느 쪽을 택할지 오늘 정해 두세요.**
>
> **⑤번의 "분산"도 중요합니다.**
> 점수 표준편차가 0.187이면 충분히 퍼져 있습니다.
> 0.05 미만이면 **모두 비슷한 점수**라 1위와 100위가 사실상 같습니다.

## 오늘의 정리

| 항목 | 핵심 |
|---|---|
| 전처리 순서 | 문자열 → 파싱 → 중복 → **이상치 → 결측** → 파생 → 검증 |
| 요금 파싱 | 가격 13.5% → **무료 여부 61.8%** |
| 결측 원칙 | **"정보 없음" ≠ "무료"** — 0으로 채우지 않기 |
| 중복 | 키 기준 + **이름+좌표 반올림** |
| 파생 관계 | `walk_min` ↔ `station_km` r=1.00 → **하나만 사용** |
| EDA 목표 | 쓸 만한가 · 발견은 · **추천 기준은** |
| 발견 형식 | **관찰(숫자) → 해석 → 결정** |
| 기준 결정 | 여러 값의 **해당 건수를 표로 비교** |
| 다양성 강도 | 최다 카테고리 비중으로 결정 |
| 준비도 검증 | 필요 열 · **가중치×결측 위험도** · 점수 분산 |

**오늘 반드시 기억할 세 가지**
> ① **결측을 0으로 채우지 마세요.** "정보 없음"과 "무료"는 다릅니다.
> ② **발견마다 "→ 결정"을 붙이세요.** EDA 는 그래프 모음이 아닙니다.
> ③ **오늘 추천 기준을 확정하세요.** 내일 아침에 고민하면 늦습니다.
