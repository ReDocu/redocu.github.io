## ③ 개념 설명

### 저장 시점 설계

```
[PLAYING]
    │ 목숨 0
    ▼
[GAME_OVER] ─── 신기록 아님 ──→ [RANKING] ──→ [TITLE]
    │
    │ 신기록
    ▼
[NAME_INPUT] ── 이름 입력 완료 ─→ 저장 ─→ [RANKING] ──→ [TITLE]
```

**상태를 하나 더 추가**했습니다. `NAME_INPUT`, `RANKING`.

| 상태 | 하는 일 | DB 접근 |
|---|---|---|
| `TITLE` | 시작 대기 | 최고 기록 조회 (1회) |
| `PLAYING` | 게임 진행 | **없음** |
| `GAME_OVER` | 결과 표시 | 신기록 여부 확인 (1회) |
| `NAME_INPUT` | 이름 입력 | 없음 |
| `RANKING` | 랭킹 표시 | 랭킹 조회 (1회) |

> **원칙 — DB 접근은 상태가 바뀌는 순간에만.** 매 프레임 조회하면 안 됩니다.

### pygame 텍스트 입력

pygame 에는 입력창 위젯이 없습니다. 직접 만듭니다.

```python
class TextInput:
    """간단한 텍스트 입력 상자."""

    def __init__(self, max_length=12):
        self.text = ""
        self.max_length = max_length
        self.cursor_timer = 0

    def handle_event(self, event):
        """반환: 엔터를 눌러 입력이 확정되면 True."""
        if event.type != pygame.KEYDOWN:
            return False

        if event.key == pygame.K_RETURN:
            return bool(self.text.strip())

        if event.key == pygame.K_BACKSPACE:
            self.text = self.text[:-1]
            return False

        # 일반 문자 입력 — event.unicode 사용
        if event.unicode and len(self.text) < self.max_length:
            if event.unicode.isprintable():
                self.text += event.unicode

        return False

    def update(self):
        self.cursor_timer = (self.cursor_timer + 1) % 60

    @property
    def display_text(self):
        """커서 깜빡임을 포함한 표시 문자열."""
        cursor = "|" if self.cursor_timer < 30 else " "
        return self.text + cursor
```

> **`event.unicode`** 는 실제로 입력된 문자입니다. Shift 조합·특수문자를 자동으로 처리합니다.
> `event.key` 는 물리적 키라서 `a` 와 `A` 를 구분하지 못합니다.

> ⚠️ **한글 입력은 pygame 기본 기능으로 어렵습니다.**
> IME(입력기) 처리가 복잡해, 이 과정에서는 **영문·숫자 이름만** 받습니다.
> 한글이 필요하면 게임 시작 전 콘솔에서 `input()` 으로 받는 방법도 있습니다.

### Step 1. 텍스트 입력 위젯

`text_input.py`

```python
"""pygame 텍스트 입력 위젯."""

import pygame

WIDTH, HEIGHT = 800, 600
BG = (18, 20, 30)
WHITE = (240, 240, 240)
GRAY = (110, 115, 130)
YELLOW = (241, 196, 15)
BOX = (35, 38, 52)


class TextInput:
    """엔터로 확정하는 한 줄 입력 상자."""

    def __init__(self, max_length=12, allowed=None):
        self.text = ""
        self.max_length = max_length
        self.allowed = allowed          # None 이면 출력 가능한 모든 문자
        self.cursor_timer = 0

    def handle_event(self, event):
        """엔터로 입력이 확정되면 True 를 반환합니다."""
        if event.type != pygame.KEYDOWN:
            return False

        if event.key == pygame.K_RETURN:
            return bool(self.text.strip())

        if event.key == pygame.K_BACKSPACE:
            self.text = self.text[:-1]
            return False

        if event.key == pygame.K_ESCAPE:
            self.text = ""
            return False

        ch = event.unicode
        if not ch or not ch.isprintable():
            return False
        if len(self.text) >= self.max_length:
            return False
        if self.allowed is not None and ch not in self.allowed:
            return False

        self.text += ch
        return False

    def update(self):
        self.cursor_timer = (self.cursor_timer + 1) % 60

    @property
    def value(self):
        return self.text.strip()

    def draw(self, surface, font, center, width=360, height=52):
        """입력 상자를 그립니다."""
        cx, cy = center
        rect = pygame.Rect(0, 0, width, height)
        rect.center = (cx, cy)

        pygame.draw.rect(surface, BOX, rect, border_radius=8)
        pygame.draw.rect(surface, YELLOW, rect, 2, border_radius=8)

        cursor = "|" if self.cursor_timer < 30 else ""
        display = self.text + cursor

        if not self.text:
            t = font.render("이름을 입력하세요" + cursor, True, GRAY)
        else:
            t = font.render(display, True, WHITE)

        surface.blit(t, t.get_rect(center=rect.center))

        # 글자 수 표시
        count = pygame.font.SysFont("malgungothic", 14).render(
            f"{len(self.text)}/{self.max_length}", True, GRAY)
        surface.blit(count, (rect.right - 44, rect.bottom + 6))


if __name__ == "__main__":
    pygame.init()
    screen = pygame.display.set_mode((WIDTH, HEIGHT))
    pygame.display.set_caption("텍스트 입력 위젯 테스트")
    clock = pygame.time.Clock()

    font = pygame.font.SysFont("malgungothic", 26)
    small = pygame.font.SysFont("malgungothic", 16)
    big = pygame.font.SysFont("malgungothic", 40)

    # 영문·숫자만 허용
    ALLOWED = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789_ "
    text_input = TextInput(max_length=12, allowed=ALLOWED)

    submitted = []

    running = True
    while running:
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                running = False
            elif event.type == pygame.KEYDOWN and event.key == pygame.K_F1:
                running = False
            else:
                if text_input.handle_event(event):
                    submitted.append(text_input.value)
                    submitted = submitted[-5:]
                    text_input.text = ""

        text_input.update()

        screen.fill(BG)

        title = big.render("NEW RECORD!", True, YELLOW)
        screen.blit(title, title.get_rect(center=(WIDTH // 2, 140)))

        text_input.draw(screen, font, (WIDTH // 2, 240))

        hint = small.render("영문·숫자만 · ENTER 로 확정 · ESC 로 지우기 · F1 종료",
                            True, GRAY)
        screen.blit(hint, hint.get_rect(center=(WIDTH // 2, 320)))

        if submitted:
            screen.blit(small.render("입력 기록:", True, GRAY), (280, 380))
            for i, s in enumerate(submitted):
                screen.blit(small.render(f"  {i + 1}. {s}", True, WHITE),
                            (280, 406 + i * 24))

        pygame.display.flip()
        clock.tick(60)

    pygame.quit()
```

```python
import pygame

WIDTH, HEIGHT = 800, 600

MEDAL_COLORS = {0: GOLD, 1: SILVER, 2: BRONZE}


def draw_text(surface, text, font, color, pos, center=False):
    t = font.render(text, True, color)
    rect = t.get_rect(center=pos) if center else t.get_rect(topleft=pos)
    surface.blit(t, rect)
    return rect


def draw_ranking(surface, entries, fonts, new_record_id=None):
```

### Step 3. 게임 + DB 완전 연동 (오늘의 종합 예제)

`game/scenes/manager.py` — 상태 전환에 DB 를 연결합니다.

```python
"""게임 상태 관리 — DB 연동 포함."""

import pygame

from .. import db
from ..settings import (SCREEN_WIDTH, SCREEN_HEIGHT, COLOR_BG, COLOR_TEXT,
                        COLOR_TEXT_DIM, COLOR_BULLET)
from ..utils import make_font, draw_text, draw_overlay

# ── 상태 ──
TITLE = "TITLE"
PLAYING = "PLAYING"
PAUSED = "PAUSED"
GAME_OVER = "GAME_OVER"
NAME_INPUT = "NAME_INPUT"
RANKING = "RANKING"

ALLOWED_CHARS = ("abcdefghijklmnopqrstuvwxyz"
                 "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789_ ")


class TextInput:
    """이름 입력 위젯."""

    def __init__(self, max_length=12):
        self.text = ""
        self.max_length = max_length
        self.cursor_timer = 0

    def handle_event(self, event):
        if event.type != pygame.KEYDOWN:
            return False

        if event.key == pygame.K_RETURN:
            return bool(self.text.strip())
        if event.key == pygame.K_BACKSPACE:
            self.text = self.text[:-1]
            return False

        ch = event.unicode
        if ch and ch in ALLOWED_CHARS and len(self.text) < self.max_length:
            self.text += ch
        return False

    def update(self):
        self.cursor_timer = (self.cursor_timer + 1) % 60

    @property
    def value(self):
        return self.text.strip() or "PLAYER"


class GameManager:
    """게임 전체 상태를 관리하고 DB 와 연결합니다."""

    def __init__(self, screen):
        self.screen = screen
        self.state = TITLE

        self.fonts = {
            "big": make_font(42),
            "normal": make_font(22),
            "small": make_font(15),
        }

        self.text_input = TextInput()

        # ── DB 조회 결과 캐시 (상태 진입 시 1회만 갱신) ──
        self.cached_ranking = []
        self.cached_best = None
        self.new_record_id = None

        db.init()
        self._refresh_title_info()

        # 게임 객체는 start_game() 에서 생성
        self.player = None
        self.stats = None
        self.play_frames = 0

    # ══════════ DB 접근 (상태 전환 시에만) ══════════

    def _refresh_title_info(self):
        """타이틀 진입 시 최고 기록을 가져옵니다."""
        ranking = db.get_ranking(1)
        self.cached_best = ranking[0] if ranking else None

    def _refresh_ranking(self):
        """랭킹 화면 진입 시 목록을 가져옵니다."""
        self.cached_ranking = db.get_ranking(10)

    def _save_record(self):
        """기록을 저장하고 순위를 반환합니다."""
        try:
            record_id, rank = db.save_score(
                self.text_input.value,
                self.stats.score,
                kills=self.stats.kills,
                max_combo=self.stats.max_combo,
                shots=self.stats.shots,
                hits=self.stats.hits,
                play_seconds=self.play_frames // 60,
                stage=getattr(self.stats, "stage", 1),
            )
            self.new_record_id = record_id
            return rank
        except Exception as e:
            print(f"[경고] 기록 저장 실패: {e}")
            self.new_record_id = None
            return None

    # ══════════ 상태 전환 ══════════

    def _change_state(self, new_state):
        """상태를 바꾸고 진입 처리를 수행합니다."""
        self.state = new_state

        if new_state == TITLE:
            self._refresh_title_info()
            self.new_record_id = None
        elif new_state == RANKING:
            self._refresh_ranking()
        elif new_state == NAME_INPUT:
            self.text_input.text = ""

    def start_game(self):
        """새 게임을 시작합니다."""
        from ..entities import Player          # 순환 참조 방지를 위한 지역 import
        from ..stats import GameStats

        self.player = Player(SCREEN_WIDTH // 2, SCREEN_HEIGHT - 70)
        self.stats = GameStats(lives=3)
        self.play_frames = 0
        # (스프라이트 그룹 초기화는 실제 구현에서 추가)

        self._change_state(PLAYING)

    # ══════════ 이벤트 ══════════

    def handle_event(self, event):
        """이벤트를 처리합니다. 프로그램을 계속하면 True."""
        if event.type != pygame.KEYDOWN:
            return True

        if event.key == pygame.K_ESCAPE and self.state != NAME_INPUT:
            return False

        if self.state == TITLE:
            if event.key == pygame.K_RETURN:
                self.start_game()
            elif event.key == pygame.K_r:
                self._change_state(RANKING)

        elif self.state == PLAYING:
            if event.key == pygame.K_p:
                self._change_state(PAUSED)

        elif self.state == PAUSED:
            if event.key == pygame.K_p:
                self.state = PLAYING
            elif event.key == pygame.K_q:
                self._change_state(TITLE)

        elif self.state == GAME_OVER:
            if event.key == pygame.K_RETURN:
                # 신기록이면 이름 입력, 아니면 바로 랭킹
                if db.is_new_record(self.stats.score):
                    self._change_state(NAME_INPUT)
                else:
                    self._change_state(RANKING)

        elif self.state == NAME_INPUT:
            if self.text_input.handle_event(event):
                self._save_record()
                self._change_state(RANKING)

        elif self.state == RANKING:
            if event.key == pygame.K_RETURN:
```

> - **DB 접근이 `_refresh_*` / `_save_record` 세 함수에만** 있습니다. 그리기 함수는 캐시된 데이터만 씁니다.

> - `new_record_id` 로 방금 저장한 기록을 랭킹에서 강조합니다.
> - 타이틀에 **최고 기록을 표시**해 동기를 부여합니다.

```python
    ...
```

**증상** — DB 파일이 순식간에 수백 MB가 됩니다.
**해결** — 게임 오버 시 **한 번만** 저장합니다.

### 상태를 직접 대입해 진입 처리를 놓침

```python
self.state = RANKING                    # 캐시 갱신 안 됨 → 빈 화면
```

**해결** — `self._change_state(RANKING)` 를 통해서만 바꿉니다.

```python

    self.text += chr(event.key)         # Shift, 특수문자 처리 안 됨
```

**해결** — `event.unicode` 를 씁니다.

### 이름 검증을 안 함

```python
db.save_score(self.text_input.text, ...)     # 빈 문자열이 저장됨
```

**해결** — 빈 값이면 기본 이름을 씁니다. `self.text or "PLAYER"`

### 한글 입력이 안 된다고 당황

pygame 은 IME 를 기본 지원하지 않습니다. **영문·숫자만 받는 것이 정상**입니다.
필요하면 `pygame.key.start_text_input()` 과 `TEXTINPUT` 이벤트를 검토하되, 이 과정에서는 범위 밖입니다.

### 저장 실패 시 게임이 죽음

```python
db.save_score(...)                      # 예외가 그대로 전파됨
```

**해결** — `try/except` 로 감싸고 로그만 남깁니다. 점수 저장 실패로 게임이 죽으면 안 됩니다.

### 신기록 판정을 저장 후에 함

```python
db.save_score(...)                      # 먼저 저장하고
if db.is_new_record(score):             # 확인하면 항상 True
```

**해결** — **저장 전에** 확인합니다.

```python
def get_player_history(player_name, limit=20):
    """플레이어의 기록 이력을 반환합니다."""
    with session() as conn:
        rows = conn.execute("""
            SELECT id, score, kills, stage, play_seconds, created_at,
                   CASE WHEN shots > 0 THEN hits * 100.0 / shots ELSE 0 END
                   AS accuracy
            FROM scores
            WHERE player_name = ?
            ORDER BY created_at DESC
            LIMIT ?
        """, (player_name, limit)).fetchall()

    return [dict(r) for r in rows]


def get_player_stats(player_name):
    """플레이어의 요약 통계를 반환합니다. 기록이 없으면 None."""
    with session() as conn:
        row = conn.execute("""
            SELECT COUNT(*)          AS games,
                   MAX(score)        AS best_score,
                   ROUND(AVG(score)) AS avg_score,
                   SUM(kills)        AS total_kills,
                   MAX(stage)        AS best_stage,
                   SUM(play_seconds) AS total_seconds,
                   CASE WHEN SUM(shots) > 0
                        THEN SUM(hits) * 100.0 / SUM(shots) ELSE 0 END
                   AS accuracy
            FROM scores
            WHERE player_name = ?
        """, (player_name,)).fetchone()

        if row is None or row["games"] == 0:
            return None

        result = dict(row)

        # 전체 순위 (최고 점수 기준)
        result["rank"] = conn.execute(
            "SELECT COUNT(*) + 1 AS r FROM scores WHERE score > ?",
            (result["best_score"],),
        ).fetchone()["r"]

        # 전체 플레이어 수
        result["total_players"] = conn.execute(
            "SELECT COUNT(DISTINCT player_name) AS c FROM scores"
        ).fetchone()["c"]

    return result


def get_player_names(keyword="", limit=20):
    """플레이어 이름 목록을 반환합니다 (자동완성용)."""
    with session() as conn:
        rows = conn.execute("""
            SELECT DISTINCT player_name FROM scores
            WHERE player_name LIKE ?
            ORDER BY player_name LIMIT ?
        """, (f"%{keyword}%", limit)).fetchall()

    return [r["player_name"] for r in rows]
```

```python
class GameManager:
    def __init__(self, screen):
        self.screen = screen
        self.state = TITLE
        self.font = make_font(22)

        self.text_input = TextInput(max_length=12)
        self.cached_ranking = []          # ← 캐시
        self.new_record_id = None

        db.init()

    def _change_state(self, new_state):
        """상태 전환 + 진입 처리."""
        self.state = new_state

        if new_state == RANKING:
            self.cached_ranking = db.get_ranking(10)      # ← 여기서 1회만
        elif new_state == NAME_INPUT:
            self.text_input.text = ""
        elif new_state == TITLE:
            self.new_record_id = None

    def _save_record(self):
        """기록을 저장합니다. 실패해도 게임은 계속됩니다."""
        try:
            record_id, rank = db.save_score(
                self.text_input.value,
                self.stats.score,
                kills=self.stats.kills,
                shots=self.stats.shots,
                hits=self.stats.hits,
            )
            self.new_record_id = record_id
            return rank
        except Exception as e:
            print(f"[경고] 기록 저장 실패: {e}")
            return None

    def handle_event(self, event):
        if event.type != pygame.KEYDOWN:      # ← 타입 확인
            return True

        if self.state == GAME_OVER:
            if event.key == pygame.K_RETURN:
                # 신기록일 때만 이름 입력
                if db.is_new_record(self.stats.score):
                    self._change_state(NAME_INPUT)
                else:
                    self._change_state(RANKING)

        elif self.state == NAME_INPUT:
            if self.text_input.handle_event(event):
                self._save_record()
                self._change_state(RANKING)

        elif self.state == RANKING:
            if event.key == pygame.K_RETURN:
                self._change_state(TITLE)

        return True

    def update(self):
        if self.state == NAME_INPUT:
            self.text_input.update()

    def draw(self):
        if self.state == RANKING:
            for i, r in enumerate(self.cached_ranking):    # ← 캐시 사용
                color = YELLOW if r["id"] == self.new_record_id else WHITE
                draw_text(self.screen, f"{i+1}. {r['player_name']}",
                          self.font, color, (200, 150 + i * 30))
                draw_text(self.screen, f"{r['score']:,}",
                          self.font, color, (450, 150 + i * 30))
```

**개선 효과**

| 항목 | 전 | 후 |
|---|---|---|
| 랭킹 화면 DB 조회 | 초당 60회 | 진입 시 1회 |
| 저장 대상 | 모든 기록 | 신기록만 |
| 이름 | 미정의 | 입력 화면에서 받음 |
| 저장 실패 | 게임 종료 | 로그만 남기고 계속 |
| 잘못된 이벤트 | AttributeError | 무시 |

> **교훈** — **`draw()` 안에서 파일·DB·네트워크에 접근하지 마세요.**
> `draw()` 는 1초에 60번 호출됩니다. 여기서 하는 모든 일에 60을 곱해야 합니다.
>
> 같은 원칙이 [Day 019](day-019.md)의 "폰트를 루프 안에서 만들지 마라",
> [Day 020](day-020.md)의 "이미지를 루프 안에서 변환하지 마라" 와 이어집니다.
